class Solution {
    public int[] nodesBetweenCriticalPoints(ListNode head) {
        if(head==null || head.next==null || head.next.next==null) return new int[] {-1, -1};
        List<Integer> points = new ArrayList<>();
        ListNode prev = head, curr = head.next, nxt = head.next.next;
        int pos = 2;
        while(nxt!=null) {
            boolean maxima = curr.val>prev.val && curr.val>nxt.val;
            boolean minima = curr.val<prev.val && curr.val<nxt.val;
            if(maxima || minima) points.add(pos);
            prev = curr;
            curr = nxt;
            nxt = nxt.next;
            pos++;
        }
        if(points.size()<2) return new int[] {-1, -1};
        int minDist = Integer.MAX_VALUE;
        for(int i=1;i<points.size();i++) minDist = Math.min(minDist, points.get(i)-points.get(i-1));
        return new int[] {minDist, points.get(points.size()-1)-points.get(0)};
    }
}

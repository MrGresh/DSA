class Solution {
    public int pairSum(ListNode head) {
        ListNode slow = head, fast = head;
        while(fast!=null) {
            slow = slow.next;
            fast = fast.next.next;
        }
        ListNode prev = null, nxt = null;
        while(slow!=null) {
            nxt = slow.next;
            slow.next = prev;
            prev = slow;
            slow = nxt;
        }
        ListNode curr = head;
        int ans = Integer.MIN_VALUE;
        while(prev!=null) {
            ans = Math.max(ans, curr.val + prev.val);
            curr = curr.next;
            prev = prev.next;
        }
        return ans;
    }
}

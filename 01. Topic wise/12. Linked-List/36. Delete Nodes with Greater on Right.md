class Solution {
    Node compute(Node head) {
        head = rev(head);
        int max = head.data;
        Node curr = head;
        while(curr.next!=null) {
            if(curr.next.data>=max) {
                max = curr.next.data;
                curr = curr.next;
            }
            else curr.next = curr.next.next;
        }
        return rev(head);
    }
    Node rev(Node curr) {
        Node nxt = null, prev = null;
        while(curr!=null) {
            nxt = curr.next;
            curr.next = prev;
            prev = curr;
            curr = nxt;
        }
        return prev;
    }
}

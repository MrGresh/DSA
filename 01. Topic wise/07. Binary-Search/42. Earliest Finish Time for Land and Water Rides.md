class Solution {
    public int earliestFinishTime(int[] landStartTime, int[] landDuration, int[] waterStartTime, int[] waterDuration) {
        long minTime = Long.MAX_VALUE;

        // Try Land Ride first, then Water Ride
        minTime = Math.min(minTime, solveOneWay(landStartTime, landDuration, waterStartTime, waterDuration));
        
        // Try Water Ride first, then Land Ride (Symmetric)
        minTime = Math.min(minTime, solveOneWay(waterStartTime, waterDuration, landStartTime, landDuration));

        return (int) minTime;
    }

    private long solveOneWay(int[] start1, int[] duration1, int[] start2, int[] duration2) {
        int n = start1.length;
        int m = start2.length;

        // Create pairs for the second category (Ride 2) and sort by start time
        Ride[] rides2 = new Ride[m];
        for (int i = 0; i < m; i++) {
            rides2[i] = new Ride(start2[i], duration2[i]);
        }
        Arrays.sort(rides2, (a, b) -> Integer.compare(a.start, b.start));

        // Precompute prefix minimums of duration for Case A (start2 <= finish1)
        int[] prefixMinDuration = new int[m];
        prefixMinDuration[0] = rides2[0].duration;
        for (int i = 1; i < m; i++) {
            prefixMinDuration[i] = Math.min(prefixMinDuration[i - 1], rides2[i].duration);
        }

        // Precompute suffix minimums of total finish time for Case B (start2 > finish1)
        int[] suffixMinTotal = new int[m];
        suffixMinTotal[m - 1] = rides2[m - 1].start + rides2[m - 1].duration;
        for (int i = m - 2; i >= 0; i--) {
            suffixMinTotal[i] = Math.min(suffixMinTotal[i + 1], rides2[i].start + rides2[i].duration);
        }

        long bestTotalFinish = Long.MAX_VALUE;

        // Iterate through each ride in the first category
        for (int i = 0; i < n; i++) {
            int finish1 = start1[i] + duration1[i];

            // Binary search to find the first ride2 where ride2.start > finish1
            int idx = binarySearch(rides2, finish1);

            // Case A: ride2.start <= finish1 (Indices from 0 to idx - 1)
            if (idx > 0) {
                long currentFinish = (long) finish1 + prefixMinDuration[idx - 1];
                bestTotalFinish = Math.min(bestTotalFinish, currentFinish);
            }

            // Case B: ride2.start > finish1 (Indices from idx to m - 1)
            if (idx < m) {
                long currentFinish = suffixMinTotal[idx];
                bestTotalFinish = Math.min(bestTotalFinish, currentFinish);
            }
        }

        return bestTotalFinish;
    }

    // Returns the first index where ride.start > target
    private int binarySearch(Ride[] rides, int target) {
        int low = 0, high = rides.length;
        while (low < high) {
            int mid = low + (high - low) / 2;
            if (rides[mid].start > target) {
                high = mid;
            } else {
                low = mid + 1;
            }
        }
        return low;
    }

    private static class Ride {
        int start;
        int duration;

        Ride(int start, int duration) {
            this.start = start;
            this.duration = duration;
        }
    }
}

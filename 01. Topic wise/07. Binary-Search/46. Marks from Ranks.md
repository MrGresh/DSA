### Brute Force
```java
class Solution {
    public ArrayList<Integer> getMarks(int[] l, int[] r, int[] rank) {
        List<Integer> nums = new ArrayList<Integer>();
        for(int i=0;i<l.length;i++) {
            for(int e=l[i];e<=r[i];e++) nums.add(e);
        }
        ArrayList<Integer> ans = new ArrayList<Integer>();
        for(int i=0;i<rank.length;i++) ans.add(nums.get(rank[i]-1));
        return ans;
    }
}
```

### Binary Search
```java
class Solution {
    public ArrayList<Integer> getMarks(int[] l, int[] r, int[] rank) {
        List<Integer> nums = new ArrayList<Integer>();
        nums.add(r[0]-l[0]+1);
        for(int i=1;i<l.length;i++) nums.add(r[i]-l[i]+1+nums.get(i-1));
        ArrayList<Integer> ans = new ArrayList<Integer>();
        for(int i=0;i<rank.length;i++) {
            int L=0, R=nums.size()-1;
            while(L<=R) {
                int m = L+(R-L)/2;
                if(nums.get(m)<rank[i]) L = m+1;
                else {
                    int n = r[m]-l[m]+1;
                    int firstRank = (nums.get(m)-n+1);
                    if(rank[i]>=firstRank) {
                        int firstEle = l[m];
                        int diff = rank[i] - firstRank;
                        ans.add(firstEle+diff);
                        break;
                    } else R = m-1;
                }
            }
        }
        return ans;
    }
}
```

```java
class Solution {
    public int kthElement(int a[], int b[], int k) {
        int i=-1, j=-1, pos=0;
        while(i+1<a.length && j+1<b.length) {
            ++pos;
            if(a[i+1]<b[j+1]) {
                i++;
                if(pos==k) return a[i];
            } else {
                j++;
                if(pos==k) return b[j];
            }
        }
        while(i+1<a.length) {
            ++pos;
            ++i;
            if(pos==k) return a[i];
        }
        while(j+1<b.length) {
            ++pos;
            ++j;
            if(pos==k) return b[j];
        }
        return -1;
    }
}
```

```java
class Solution {
    public int kthElement(int a[], int b[], int k) {
        int n = a.length;
        int m = b.length;

        if (n > m) return kthElement(b, a, k);

        int low = Math.max(0, k - m);
        int high = Math.min(k, n);

        while (low <= high) {
            int cut1 = (low + high) / 2;
            int cut2 = k - cut1;

            int l1 = (cut1 == 0) ? Integer.MIN_VALUE : a[cut1 - 1];
            int l2 = (cut2 == 0) ? Integer.MIN_VALUE : b[cut2 - 1];
            int r1 = (cut1 == n) ? Integer.MAX_VALUE : a[cut1];
            int r2 = (cut2 == m) ? Integer.MAX_VALUE : b[cut2];

            if (l1 <= r2 && l2 <= r1) {
                return Math.max(l1, l2);
            } else if (l1 > r2) {
                high = cut1 - 1;
            } else {
                low = cut1 + 1;
            }
        }
        return -1;
    }
}
```

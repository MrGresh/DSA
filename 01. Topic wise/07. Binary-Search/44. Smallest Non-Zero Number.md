class Solution {
    public int find(int[] arr) {
        int l=1, r=10000;
        int ans = 10000;
        while(l<=r) {
            int m = l+(r-l)/2;
            if(isPossible(arr, m)) {
                ans = m;
                r = m - 1;
            } else l = m + 1;
        }
        return ans;
    }
    boolean isPossible(int[] arr, int x) {
        for(int e: arr) {
            x = 2 * x - e;
            if (x >= 10000) return true;
            if(x<0) return false;
        }
        return true;
    }
}

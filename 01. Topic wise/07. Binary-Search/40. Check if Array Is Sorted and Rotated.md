class Solution {
    public boolean check(int[] nums) {
        int inc = 0, prev = 0;
        for(int i=0;i<nums.length;i++) {
            if(i==0) inc++;
            else if(nums[i]<nums[i-1]) {
                inc++;
                if(inc==3) return false;
            }
        }
        if(inc==2 && nums[0]<nums[nums.length-1]) return false;
        return true;
    }
}

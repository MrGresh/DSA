class Solution {
    int maxPeopleDefeated(int p) {
        int ans = 0;
        int l = 1, r = 100000;
        while(l<=r) {
            int m = l+(r-l)/2;
            long val = (((long)m)*(m+1)*(2*m+1))/6;
            if(val<=(long)p) {
                ans = m;
                l = m+1;
            } else r = m-1;
        }
        return ans;
    }
};

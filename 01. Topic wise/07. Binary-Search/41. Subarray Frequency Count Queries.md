class Solution {
    public ArrayList<Integer> freqInRange(int[] arr, int[][] queries) {
        // Map to store element -> list of indices where it appears
        Map<Integer, ArrayList<Integer>> indexMap = new HashMap<>();
        
        for (int i = 0; i < arr.length; i++) {
            indexMap.putIfAbsent(arr[i], new ArrayList<>());
            indexMap.get(arr[i]).add(i);
        }
        
        ArrayList<Integer> result = new ArrayList<>();
        
        // Process each query
        for (int[] query : queries) {
            int l = query[0];
            int r = query[1];
            int x = query[2];
            
            if (!indexMap.containsKey(x)) {
                result.add(0);
                continue;
            }
            
            ArrayList<Integer> indices = indexMap.get(x);
            
            // Find the count of indices between l and r (inclusive)
            int count = getFrequency(indices, l, r);
            result.add(count);
        }
        
        return result;
    }
    
    private int getFrequency(ArrayList<Integer> indices, int l, int r) {
        int startIndex = lowerBound(indices, l);
        int endIndex = upperBound(indices, r);
        
        return endIndex - startIndex;
    }
    
    // Finds the first index in 'indices' that is >= target
    private int lowerBound(ArrayList<Integer> indices, int target) {
        int low = 0, high = indices.size();
        while (low < high) {
            int mid = low + (high - low) / 2;
            if (indices.get(mid) >= target) {
                high = mid;
            } else {
                low = mid + 1;
            }
        }
        return low;
    }
    
    // Finds the first index in 'indices' that is > target
    private int upperBound(ArrayList<Integer> indices, int target) {
        int low = 0, high = indices.size();
        while (low < high) {
            int mid = low + (high - low) / 2;
            if (indices.get(mid) > target) {
                high = mid;
            } else {
                low = mid + 1;
            }
        }
        return low;
    }
}

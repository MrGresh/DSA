### Memoization
```java
class Solution {
    int MOD = 1000000007;
    Integer[] memo;
    public int countWays(int n, int m) {
        memo = new Integer[n+1];
        return rec(n, m);
    }
    int rec(int n, int m) {
        if(n<m) return 1;
        if(n==m) return 2;
        if(memo[n]!=null) return memo[n];
        int ans = rec(n-1, m);
        if(n-m>=0) ans = (ans + rec(n-m, m)) % MOD;
        return memo[n]=ans;
    }
}
```
### Tabulation
```java
class Solution {
    public int countWays(int N, int m) {
        int MOD = 1000000007;
        int[] dp = new int[N+1];
        for(int n=0;n<=N;n++) {
            if(n<m) dp[n] = 1;
            else if(n==m) dp[n] = 2;
            else {
                int ans = dp[n-1];
                if(n-m>=0) ans = (ans + dp[n-m]) % MOD;
                dp[n]=ans;
            }
        }
        return dp[N];
    }
}
```

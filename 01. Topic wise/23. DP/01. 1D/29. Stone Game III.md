### Memoization
```java
class Solution {
    Integer[] memo;
    public String stoneGameIII(int[] stoneValue) {
        memo = new Integer[stoneValue.length];
        int val = rec(stoneValue, 0);
        if(val==0) return "Tie";
        if(val>0) return "Alice";
        return "Bob";
    }
    int rec(int[] stoneValue, int i) {
        if(i==stoneValue.length) return 0;
        if(memo[i]!=null) return memo[i];
        int val = stoneValue[i] - rec(stoneValue, i + 1);
        if(i+1<stoneValue.length) val = Math.max(val, stoneValue[i]+stoneValue[i+1] - rec(stoneValue, i + 2));
        if(i+2<stoneValue.length) val = Math.max(val, stoneValue[i]+stoneValue[i+1]+stoneValue[i+2] - rec(stoneValue, i + 3));
        return memo[i] = val;
    }
}
```

### Tabulation
```java
class Solution {
    public String stoneGameIII(int[] stoneValue) {
        int[] dp = new int[stoneValue.length+1];
        for(int i=stoneValue.length;i>=0;i--) {
            if(i==stoneValue.length) dp[i] = 0;
            else {
                int val = stoneValue[i] - dp[i+1];
                if(i+1<stoneValue.length) val = Math.max(val, stoneValue[i]+stoneValue[i+1] - dp[i+2]);
                if(i+2<stoneValue.length) val = Math.max(val, stoneValue[i]+stoneValue[i+1]+stoneValue[i+2] - dp[i+3]);
                dp[i] = val;
            }
        }
        int val = dp[0];
        if(val==0) return "Tie";
        if(val>0) return "Alice";
        return "Bob";
    }
}
```

### Space Optimization
```java
class Solution {
    public String stoneGameIII(int[] stoneValue) {
        int n = stoneValue.length;
        int[] dp = new int[4]; // dp[i % 4] holds the value for index i

        dp[n % 4] = 0; // base case: dp[n] = 0

        for (int i = n - 1; i >= 0; i--) {
            int val = stoneValue[i] - dp[(i + 1) % 4];
            if (i + 1 < n) val = Math.max(val, stoneValue[i] + stoneValue[i + 1] - dp[(i + 2) % 4]);
            if (i + 2 < n) val = Math.max(val, stoneValue[i] + stoneValue[i + 1] + stoneValue[i + 2] - dp[(i + 3) % 4]);
            dp[i % 4] = val;
        }

        int val = dp[0]; // dp[0 % 4] == dp[0], since we just computed index 0 last
        if (val == 0) return "Tie";
        if (val > 0) return "Alice";
        return "Bob";
    }
}
```

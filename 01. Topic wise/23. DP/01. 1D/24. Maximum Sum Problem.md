### Memoization
```java
class Solution {
    Integer[] memo;
    public int maxSum(int n) {
        memo = new Integer[n + 1];
        return rec(n);
    }
    int rec(int n) {
        if (n == 0) return 0;
        if (memo[n] != null) return memo[n];
        int splitSum = rec(n / 2) + rec(n / 3) + rec(n / 4);
        return memo[n] = Math.max(n, splitSum);
    }
}
```
### Tabulation
```java
class Solution {
    public int maxSum(int N) {
        int[] dp = new int[N + 1];
        for (int n = 0; n <= N; n++) {
            if(n==0) dp[n] = 0;
            else {
                int splitSum = dp[n / 2] + dp[n / 3] + dp[n / 4];
                dp[n] = Math.max(n, splitSum);
            }
        }
        return dp[N];
    }
}
```

### Memoization
```java
class Solution {
    Integer[] memo;
    public int maxProduct(int n) {
        memo = new Integer[n+1];
        return rec(n, n);
    }
    int rec(int N, int n) {
        if(n==0) return 1;
        if(memo[n] != null) return memo[n];
        int ans = Integer.MIN_VALUE;
        for(int cut=1;cut<=n;cut++) {
            if(cut==N) continue;
            ans = Math.max(ans, cut*rec(N, n-cut));
        }
        return memo[n] = ans;
    }
}
```
### Tabulation
```java
class Solution {
    public int maxProduct(int N) {
        int[] dp = new int[N+1];
        for(int n=0;n<=N;n++) {
            if(n==0) dp[n] = 1;
            else {
                int ans = Integer.MIN_VALUE;
                for(int cut=1;cut<=n;cut++) {
                    if(cut==N) continue;
                    ans = Math.max(ans, cut*dp[n-cut]);
                }
                dp[n] = ans;
            }
        }
        return dp[N];
    }
}
```

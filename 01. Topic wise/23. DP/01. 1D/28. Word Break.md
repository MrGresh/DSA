### Trie + Backtracking (TLE)
```java
class TreeNode {
    TreeNode[] childs = new TreeNode[26];
    boolean isEnd = false;
}
class Solution {
    TreeNode root;
    Boolean[] memo;
    public boolean wordBreak(String s, List<String> wordDict) {
        root = new TreeNode();
        memo = new Boolean[s.length() + 1];
        for(String word: wordDict) {
            TreeNode node = root;
            for(int i=0;i<word.length();i++) {
                int idx = word.charAt(i)-'a';
                if(node.childs[idx]==null) node.childs[idx] = new TreeNode();
                node = node.childs[idx];
            }
            node.isEnd = true;
        }
        return rec(s, root, 0);
    }
    boolean rec(String s, TreeNode node, int i) {
        if(i==s.length()) return node.isEnd;
        int idx = s.charAt(i)-'a';
        if(node.childs[idx]==null) return false;
        boolean res = rec(s, node.childs[idx], i+1);
        if(!res && node.childs[idx].isEnd) res = rec(s, root, i+1);
        return res;
    }
}
```

### Memoization
```java
class TreeNode {
    TreeNode[] childs = new TreeNode[26];
    boolean isEnd = false;
}

class Solution {
    TreeNode root;
    Boolean[] memo;
    public boolean wordBreak(String s, List<String> wordDict) {
        root = new TreeNode();
        memo = new Boolean[s.length()];
        for (String word : wordDict) {
            TreeNode node = root;
            for (int i = 0; i < word.length(); i++) {
                int idx = word.charAt(i) - 'a';
                if (node.childs[idx] == null) {
                    node.childs[idx] = new TreeNode();
                }
                node = node.childs[idx];
            }
            node.isEnd = true;
        }
        return rec(s, 0);
    }
    private boolean rec(String s, int start) {
        if (start == s.length()) return true;
        if (memo[start] != null) return memo[start];
        TreeNode node = root;
        for (int i = start; i < s.length(); i++) {
            int idx = s.charAt(i) - 'a';
            if (node.childs[idx] == null) break;
            node = node.childs[idx];
            if (node.isEnd && rec(s, i + 1)) return memo[start] = true;
        }
        return memo[start] = false;
    }
}
```

### Tabulation
```java
class TreeNode {
    TreeNode[] childs = new TreeNode[26];
    boolean isEnd = false;
}

class Solution {
    public boolean wordBreak(String s, List<String> wordDict) {
        TreeNode root = new TreeNode();
        Boolean[] dp = new Boolean[s.length()+1];
        for (String word : wordDict) {
            TreeNode node = root;
            for (int i = 0; i < word.length(); i++) {
                int idx = word.charAt(i) - 'a';
                if (node.childs[idx] == null) {
                    node.childs[idx] = new TreeNode();
                }
                node = node.childs[idx];
            }
            node.isEnd = true;
        }
        for(int start=s.length();start>=0;start--) {
            if (start == s.length()) dp[start] = true;
            else {
                TreeNode node = root;
                for (int i = start; i < s.length(); i++) {
                    int idx = s.charAt(i) - 'a';
                    if (node.childs[idx] == null) break;
                    node = node.childs[idx];
                    if (node.isEnd && dp[i+1]) dp[start] = true;
                }
                if(dp[start]==null) dp[start] = false;
            }
        }
        return dp[0];
    }
}
```

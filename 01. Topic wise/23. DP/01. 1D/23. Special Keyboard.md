### Raw Recursion
```java
class Solution {
    public int optimalKeys(int n) {
        return rec(n, 0, 0);
    }
    int rec(int n, int curr, int buffer) {
        if(n==0) return curr;
        int ans = rec(n-1, curr+1, buffer);
        ans = Math.max(ans, rec(n-1, curr+buffer, buffer));
        if(n-2>=0) ans = Math.max(ans, rec(n-2, curr, curr));
        if(n-3>=0) ans = Math.max(ans, rec(n-3, 2*curr, curr));
        return ans;
    }
}
```
### Memoization
```java
class Solution {
    Integer[] memo;
    public int optimalKeys(int n) {
        memo = new Integer[n + 1];
        return rec(n);
    }
    int rec(int n) {
        if (n <= 6) return n;
        if (memo[n] != null) return memo[n];
        int maxA = rec(n - 1) + 1;
        for (int j = n - 3; j >= 1; j--) {
            int curr = rec(j) * (n - j - 1);
            maxA = Math.max(maxA, curr);
        }
        return memo[n] = maxA;
    }
}
```
### Tabulation
```java
class Solution {
    public int optimalKeys(int N) {
        int[] dp = new int[N + 1];
        for(int n=0;n<=N;n++) {
            if(n<=6) dp[n] =  n;
            else {
                int maxA = dp[n-1] + 1;
                for (int j = n - 3; j >= 1; j--) {
                    int curr = dp[j] * (n - j - 1);
                    maxA = Math.max(maxA, curr);
                }
                dp[n] = maxA;
            }
        }
        return dp[N];
    }
}
```

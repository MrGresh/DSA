### Memoization
```java
class Solution {
    Integer[] memo;
    public int maximumJumps(int[] nums, int target) {
        memo = new Integer[nums.length];
        int ans = rec(nums, target, 0);
        return ans == Integer.MIN_VALUE ? -1 : ans;
    }
    int rec(int[] nums, int target, int i) {
        if(i==nums.length-1) return 0;
        if(memo[i]!=null) return memo[i];
        int val = Integer.MIN_VALUE;
        for(int j=i+1;j<nums.length;j++) {
            int calc = nums[j] - nums[i];
            if((-target <= calc) && (calc <= target)) {
                int result = rec(nums, target, j);
                if(result!=Integer.MIN_VALUE) val = Math.max(val, 1+result);
            }
        }
        return memo[i] = val;
    }
}
```

### Tabulation
```java
class Solution {
    public int maximumJumps(int[] nums, int target) {
        int[] dp = new int[nums.length];
        for(int i=nums.length-1;i>=0;i--) {
            if(i==nums.length-1) dp[i] = 0;
            else {
                int val = Integer.MIN_VALUE;
                for(int j=i+1;j<nums.length;j++) {
                    int calc = nums[j] - nums[i];
                    if((-target <= calc) && (calc <= target)) {
                        int result = dp[j];
                        if(result!=Integer.MIN_VALUE) val = Math.max(val, 1+result);
                    }
                }
                dp[i] = val;
            }
        }
        int ans = dp[0];
        return ans == Integer.MIN_VALUE ? -1 : ans;
    }
}
```

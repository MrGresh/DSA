### Memoization
```java
class Solution {
    Integer[] memo;
    public int minimumCost(int x, int s, int m, int l, int cs, int cm, int cl) {
        memo = new Integer[x+1];
        return rec(x, s, m, l, cs, cm, cl);
    }
    int rec(int x, int s, int m, int l, int cs, int cm, int cl) {
        if(x<=0) return 0;
        if(memo[x]!=null) return memo[x];
        int val = cs + rec(x-s, s, m, l, cs, cm, cl);
        val = Math.min(val, cm + rec(x-m, s, m, l, cs, cm, cl));
        val = Math.min(val, cl + rec(x-l, s, m, l, cs, cm, cl));
        return memo[x]=val;
    }
}
```

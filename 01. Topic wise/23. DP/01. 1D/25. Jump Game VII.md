### BFS
```java
class Solution {
    public boolean canReach(String s, int minJump, int maxJump) {
        int n = s.length();
        if (s.charAt(n - 1) == '1') return false;

        Queue<Integer> queue = new LinkedList<>();
        queue.offer(0);
        
        int furthestExplored = 0; 

        while (!queue.isEmpty()) {
            int i = queue.poll();

            if (i == n - 1) return true;

            int start = Math.max(i + minJump, furthestExplored + 1);
            int end = Math.min(i + maxJump, n - 1);

            for (int j = start; j <= end; j++) {
                if (s.charAt(j) == '0') {
                    queue.offer(j);
                }
            }
            
            furthestExplored = Math.max(furthestExplored, end);
        }

        return false;
    }
}
```

### Memoization
```java
class Solution {
    Boolean[] memo;
    public boolean canReach(String s, int minJump, int maxJump) {
        if(s.charAt(s.length()-1)=='1') return false;
        memo = new Boolean[s.length()];
        return rec(s, minJump, maxJump, 0);
    }
    boolean rec(String s, int minJump, int maxJump, int i) {
        if(i==s.length()-1) return true;
        if(memo[i]!=null) return memo[i];

        int start = i + minJump;
        int end = Math.min(i + maxJump, s.length() - 1);

        for(int j=start;j<=end;j++) {
            if(s.charAt(j)=='0' && rec(s, minJump, maxJump, j)) return memo[i] = true;
        }
        return memo[i] = false;
    }
}
```

### Tabulation
```java
class Solution {
    public boolean canReach(String s, int minJump, int maxJump) {
        if(s.charAt(s.length()-1)=='1') return false;
        boolean[] dp = new boolean[s.length()];
        for(int i=s.length()-1;i>=0;i--) {
            if(i==s.length()-1) dp[i] = true;
            else {
                int start = i + minJump;
                int end = Math.min(i + maxJump, s.length() - 1);

                for(int j=start;j<=end;j++) {
                    if(s.charAt(j)=='0' && dp[j]) {
                        dp[i] = true;
                        break;
                    }
                }
            }
        }
        return dp[0];
    }
}
```

### Memoization
```java
class Solution {
    Integer[] memo;
    public int minOperation(int n) {
        memo = new Integer[n];
        return rec(n, 0);
    }
    int rec(int n, int i) {
        if(i==n) return 0;
        if(memo[i]!=null) return memo[i];
        int steps = Integer.MAX_VALUE;
        if(i*2<=n && i!=0) steps = rec(n, i*2);
        steps = Math.min(steps, rec(n, i+1));
        return memo[i]=1+steps;
    }
}
```
### Tabulation
```java
class Solution {
    public int minOperation(int n) {
        int[] dp = new int[n+1];
        for(int i=n;i>=0;i--) {
            if(i==n) dp[i] = 0;
            else {
                int steps = Integer.MAX_VALUE;
                if(i*2<=n && i!=0) steps = dp[i*2];
                steps = Math.min(steps, dp[i+1]);
                dp[i]=1+steps;
            }
        }
        return dp[0];
    }
}
```
### Greedy (Maths)
```java
class Solution {
    public int minOperation(int n) {
        int ops = 0;
        while (n > 0) {
            if (n % 2 == 0) n /= 2;
            else n--;
            ops++;
        }
        return ops;
    }
}
```

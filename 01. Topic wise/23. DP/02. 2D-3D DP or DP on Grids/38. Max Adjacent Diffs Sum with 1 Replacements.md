### Memoization
```java
class Solution {
    private Long[][] memo;
    public int maxDiffSum(int[] arr) {
        int n = arr.length;
        if (n <= 1) return 0;
        
        this.memo = new Long[n][2];
        
        return (int) Math.max(solve(arr, n - 1, 0), solve(arr, n - 1, 1));
    }
    private long solve(int[] arr, int i, int v) {
        if (i == 0) return 0;
        if (memo[i][v]!=null) return memo[i][v];
        long curVal = (v == 0) ? arr[i] : 1;
        
        long fromPrev0 = Math.abs(curVal - arr[i - 1]) + solve(arr, i - 1, 0);
        long fromPrev1 = Math.abs(curVal - 1L) + solve(arr, i - 1, 1);
        
        long result = Math.max(fromPrev0, fromPrev1);
        memo[i][v] = result;
        return result;
    }
}
```

### Tabulation
```java
class Solution {
    public int maxDiffSum(int[] arr) {
        int n = arr.length;
        if (n <= 1) return 0;
        
        long[][] dp = new long[n][2];
        for(int i=0;i<n;i++) {
            for(int v=0;v<=1;v++) {
                if (i == 0) dp[i][v] = 0;
                else {
                    long curVal = (v == 0) ? arr[i] : 1;
                    
                    long fromPrev0 = Math.abs(curVal - arr[i - 1]) + dp[i - 1][0];
                    long fromPrev1 = Math.abs(curVal - 1L) + dp[i - 1][1];
                    
                    long result = Math.max(fromPrev0, fromPrev1);
                    dp[i][v] = result;
                }
            }
        }
        return (int) Math.max(dp[n - 1][0], dp[n - 1][1]);
    }
}
```

### Space Optimization
```java
class Solution {
    public int maxDiffSum(int[] arr) {
        int n = arr.length;
        if (n <= 1) return 0;
        
        long[] prev = new long[2];
        for(int i=0;i<n;i++) {
            long[] curr = new long[2];
            for(int v=0;v<=1;v++) {
                if (i == 0) curr[v] = 0;
                else {
                    long curVal = (v == 0) ? arr[i] : 1;
                    
                    long fromPrev0 = Math.abs(curVal - arr[i - 1]) + prev[0];
                    long fromPrev1 = Math.abs(curVal - 1L) + prev[1];
                    
                    long result = Math.max(fromPrev0, fromPrev1);
                    curr[v] = result;
                }
            }
            prev = curr;
        }
        return (int) Math.max(prev[0], prev[1]);
    }
}
```

### Memoization
```java
class Solution {
    Integer[][] memo;
    int MOD = 1000000007;
    public int prefixStrings(int n) {
        memo = new Integer[n+1][n+1];
        return rec(n, n);
    }
    int rec(int ones, int zeros) {
        if(ones==0 && zeros==0) return 1;
        if(memo[ones][zeros]!=null) return memo[ones][zeros];
        int cnt = 0;
        if(ones>0) cnt = (cnt + rec(ones-1, zeros)) % MOD;
        if(zeros>0 && zeros>ones) cnt = (cnt + rec(ones, zeros-1)) % MOD;
        return memo[ones][zeros]=cnt;
    }
}
```
### Tabulation
```java
class Solution {
    public int prefixStrings(int n) {
        int MOD = 1000000007;
        int[][] dp = new int[n+1][n+1];
        for(int ones=0;ones<=n;ones++) {
            for(int zeros=0;zeros<=n;zeros++) {
                if(ones==0 && zeros==0) dp[ones][zeros] = 1;
                else {
                    int cnt = 0;
                    if(ones>0) cnt = (cnt + dp[ones-1][zeros]) % MOD;
                    if(zeros>0 && zeros>ones) cnt = (cnt + dp[ones][zeros-1]) % MOD;
                    dp[ones][zeros]=cnt;
                }
            }
        }
        return dp[n][n];
    }
}
```
### Space Optimization
```java
class Solution {
    public int prefixStrings(int n) {
        int MOD = 1000000007;
        int[] prev = new int[n+1];
        for(int ones=0;ones<=n;ones++) {
            int[] curr = new int[n+1];
            for(int zeros=0;zeros<=n;zeros++) {
                if(ones==0 && zeros==0) curr[zeros] = 1;
                else {
                    int cnt = 0;
                    if(ones>0) cnt = (cnt + prev[zeros]) % MOD;
                    if(zeros>0 && zeros>ones) cnt = (cnt + curr[zeros-1]) % MOD;
                    curr[zeros]=cnt;
                }
            }
            prev = curr;
        }
        return prev[n];
    }
}
```

### Memoization
```java
class Solution {
    Integer[][] memo;
    public int validGroups(String s) {
        int n = s.length();
        memo = new Integer[n][(9*n) + 1];
        return rec(s, 0, 0);
    }
    int rec(String s, int idx, int prevSum) {
        if(idx==s.length()) return 1;
        if(memo[idx][prevSum] != null) return memo[idx][prevSum];
        int currSum = 0, cnt = 0;
        for(int i=idx;i<s.length();i++) {
            currSum += s.charAt(i)-'0';
            if(currSum>=prevSum) cnt += rec(s, i+1, currSum);
        }
        return memo[idx][prevSum] = cnt;
    }
}
```
### Tabulation (TLE)
```java
class Solution {
    public int validGroups(String s) {
        int n = s.length();
        int[][] dp = new int[n+1][(9*n) + 1];
        for(int idx=n;idx>=0;idx--) {
            for(int prevSum=(9*n);prevSum>=0;prevSum--) {
                if(idx==s.length()) dp[idx][prevSum] = 1;
                else {
                    int currSum = 0, cnt = 0;
                    for(int i=idx;i<s.length();i++) {
                        currSum += s.charAt(i)-'0';
                        if(currSum>=prevSum) cnt += dp[i+1][currSum];
                    }
                    dp[idx][prevSum] = cnt;
                }
            }
        }
        return dp[0][0];
    }
}
```

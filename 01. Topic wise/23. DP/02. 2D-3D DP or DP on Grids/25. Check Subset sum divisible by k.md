### Memoization
```java
class Solution {
    Integer[][] memo;
    public boolean divisibleByK(int[] arr, int k) {
        int sum = 0;
        for(int x: arr) sum += x;
        memo = new Integer[arr.length][sum+1];
        return rec(arr, k, 0, 0);
    }
    boolean rec(int[] arr, int k, int i, int sum) {
        if(i==arr.length) return sum==0 ? false : sum%k==0;
        if(memo[i][sum]!=null) return memo[i][sum]==1;
        memo[i][sum] = (rec(arr, k, i+1, sum) || rec(arr, k, i+1, sum+arr[i])) ? 1 : 0;
        return memo[i][sum] == 1;
    }
}

### Tabulation
class Solution {
    public boolean divisibleByK(int[] arr, int k) {
        int Sum = 0;
        for(int x: arr) Sum += x;
        int[][] dp = new int[arr.length+1][Sum+1];
        for(int i=arr.length;i>=0;i--) {
            int sum = Sum;
            while(sum >= 0) {
                if(i==arr.length) dp[i][sum] = (sum==0 ? 0 : (sum%k==0 ? 1 : 0));
                else dp[i][sum] = (dp[i+1][sum]==1 || (sum+arr[i]>Sum ? false : dp[i+1][sum+arr[i]]==1)) ? 1 : 0;
                sum--;
            }
        }
        return dp[0][0]==1;
    }
}

### Space Optimization
```java
class Solution {
    public boolean divisibleByK(int[] arr, int k) {
        int Sum = 0;
        for(int x: arr) Sum += x;
        int[] prev = new int[Sum+1];
        for(int i=arr.length;i>=0;i--) {
            int[] curr = new int[Sum+1];
            int sum = Sum;
            while(sum >= 0) {
                if(i==arr.length) curr[sum] = (sum==0 ? 0 : (sum%k==0 ? 1 : 0));
                else curr[sum] = (prev[sum]==1 || (sum+arr[i]>Sum ? false : prev[sum+arr[i]]==1)) ? 1 : 0;
                sum--;
            }
            prev = curr;
        }
        return prev[0]==1;
    }
}

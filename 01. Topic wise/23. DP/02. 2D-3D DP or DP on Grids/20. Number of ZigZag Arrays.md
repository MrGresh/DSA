### Memoization
```java
class Solution {
    Long[][][] memo;
    long MOD = 1000000007;
    public int zigZagArrays(int n, int l, int r) {
        memo = new Long[n+1][r+1][3];
        return (int) (rec(l, r, n, 0, 0) % MOD);
    }
    long rec(int l, int r, int n, int p1, int p2) {
        if(n==0) return 1;
        if(memo[n][p1][p2] != null) return memo[n][p1][p2];
        long ans = 0;
        for(int i=l;i<=r;i++) {
            boolean c1 = ((p1!=0 ? i>p1 : false) && (p2 == 1 ? true : false));
            boolean c2 = ((p1!=0 ? i<p1 : false) && (p2 == 2 ? true : false));
            boolean c3 = ((p1!=0) ? (i==p1) : false);
            if(!(c1 || c2 || c3)) ans = (ans + rec(l, r, n-1, i, p1==0?0:(i>p1?1:2))) % MOD;
        }
        return memo[n][p1][p2] = ans;
    }
}
```

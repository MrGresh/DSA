### Memoization
```java
class Solution {
    int MOD = 1000000007;
    Integer[][][] memo;
    public int countStrings(int n, int k) {
        memo = new Integer[n+1][k+1][3];
        return rec(n, k, 2);
    }
    int rec(int n, int k, int last) {
        if(n==0) return k==0 ? 1 : 0;
        if(memo[n][k][last] != null) return memo[n][k][last];
        int ans = rec(n-1, k, 0);
        if(last==0 || last==2) {
            ans = (ans + rec(n-1, k, 1)) % MOD;
        } else {
            if(k>0) ans = (ans + rec(n-1, k-1, 1)) % MOD;
        }
        return memo[n][k][last] = ans;
    }
}
```

### Tabulation
```java
class Solution {
    public int countStrings(int N, int K) {
        int MOD = 1000000007;
        int[][][] dp = new int[N+1][K+1][3];
        for(int n=0;n<=N;n++) {
            for(int k=0;k<=K;k++) {
                for(int last=0;last<3;last++) {
                    if(n==0) dp[n][k][last] = (k==0 ? 1 : 0);
                    else {
                        int ans = dp[n-1][k][0];
                        if(last==0 || last==2) {
                            ans = (ans + dp[n-1][k][1]) % MOD;
                        } else {
                            if(k>0) ans = (ans + dp[n-1][k-1][1]) % MOD;
                        }
                        dp[n][k][last] = ans;
                    }
                }
            }
        }
        return dp[N][K][2];
    }
}
```

### Space Optimization
```java
class Solution {
    public int countStrings(int N, int K) {
        int MOD = 1000000007;
        int[][] prev = new int[K+1][3];
        for(int n=0;n<=N;n++) {
            int[][] curr = new int[K+1][3];
            for(int k=0;k<=K;k++) {
                for(int last=0;last<3;last++) {
                    if(n==0) curr[k][last] = (k==0 ? 1 : 0);
                    else {
                        int ans = prev[k][0];
                        if(last==0 || last==2) {
                            ans = (ans + prev[k][1]) % MOD;
                        } else {
                            if(k>0) ans = (ans + prev[k-1][1]) % MOD;
                        }
                        curr[k][last] = ans;
                    }
                }
            }
            prev = curr;
        }
        return prev[K][2];
    }
}
```

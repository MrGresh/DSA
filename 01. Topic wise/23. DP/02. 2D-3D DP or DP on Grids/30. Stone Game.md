### Raw Recursion
```java
class Solution {
    public boolean stoneGame(int[] nums) {
        return rec(nums, 0, nums.length-1, 1, 0, 0);
    }
    boolean rec(int[] nums, int l, int r, int turn, int s1, int s2) {
        if(l>r) return s1>=s2;
        if(turn==1) {
            if(rec(nums, l+1, r, 2, s1+nums[l], s2)) return true;;
            return rec(nums, l, r-1, 2, s1+nums[r], s2);
        } else {
            if(!rec(nums, l+1, r, 1, s1, s2+nums[l])) return false;
            return rec(nums, l, r-1, 1, s1, s2+nums[r]);
        }
    }
}
```

### Memoization
```java
class Solution {
    Integer[][] memo;
    public boolean stoneGame(int[] piles) {
        int n = piles.length;
        memo = new Integer[n][n];
        return rec(piles, 0, n - 1) >= 0;
    }
    int rec(int[] piles, int l, int r) {
        if (l > r) return 0;
        if (memo[l][r] != null) return memo[l][r];

        int takeLeft = piles[l] - rec(piles, l + 1, r);
        int takeRight = piles[r] - rec(piles, l, r - 1);

        int best = Math.max(takeLeft, takeRight);
        memo[l][r] = best;
        return best;
    }
}
```
### Tabulation
```java
class Solution {
    public boolean stoneGame(int[] piles) {
        int n = piles.length;
        int[][] dp = new int[n][n];
        for(int l=n-1;l>=0;l--) {
            for(int r=0;r<n;r++) {
                if (l > r) dp[l][r] = 0;
                else {
                    int takeLeft = l<n-1 ? piles[l] - dp[l + 1][r] : Integer.MIN_VALUE;
                    int takeRight = r>0 ? piles[r] - dp[l][r - 1] : Integer.MIN_VALUE;

                    int best = Math.max(takeLeft, takeRight);
                    dp[l][r] = best;
                }
            }
        }
        return dp[0][n-1] >= 0;
    }
}
```

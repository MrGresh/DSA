### Memoization
```java
class Solution {
    public int maxProduct(int[] arr, int k) {
        Arrays.sort(arr);
        int n = arr.length;
        // memo[i][j][k] -> best product using k picks from window [i, j]
        Long[][][] memo = new Long[n][n][k + 1];
        long ans = solve(arr, 0, n - 1, k, memo);
        return (int) ans; // see note below about overflow
    }

    private long solve(int[] arr, int i, int j, int k, Long[][][] memo) {
        if (k == 0) return 1L;
        if (k == 1) return arr[j]; // arr sorted ascending -> arr[j] is the max in [i,j]

        if (memo[i][j][k] != null) return memo[i][j][k];

        // Option 1: take the two smallest (most negative) from the left
        long takeLeft = (long) arr[i] * arr[i + 1] * solve(arr, i + 2, j, k - 2, memo);

        // Option 2: take the two largest from the right
        long takeRight = (long) arr[j] * arr[j - 1] * solve(arr, i, j - 2, k - 2, memo);

        long res = Math.max(takeLeft, takeRight);
        memo[i][j][k] = res;
        return res;
    }
}
```

### Tabulation
```java
class Solution {
    public int maxProduct(int[] arr, int K) {
        Arrays.sort(arr);
        int n = arr.length;
        final long NEG_INF = Long.MIN_VALUE / 2;
        long[][][] dp = new long[n][n][K + 1];
        for (long[][] mat : dp) for (long[] row : mat) Arrays.fill(row, NEG_INF);

        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < n; j++) {
                if (i > j) continue;          // invalid window
                dp[i][j][0] = 1L;             // 0 picks -> product 1
                for (int k = 1; k <= K; k++) {
                    if (j - i + 1 < k) continue; // not enough elements
                    if (k == 1) { dp[i][j][k] = arr[j]; continue; }

                    long takeLeft = NEG_INF, takeRight = NEG_INF;
                    if (i + 1 <= j) {
                        long sub = (i + 2 > j) ? (k - 2 == 0 ? 1L : NEG_INF) : dp[i + 2][j][k - 2];
                        if (sub != NEG_INF) takeLeft = (long) arr[i] * arr[i + 1] * sub;
                    }
                    if (j - 1 >= i) {
                        long sub = (j - 2 < i) ? (k - 2 == 0 ? 1L : NEG_INF) : dp[i][j - 2][k - 2];
                        if (sub != NEG_INF) takeRight = (long) arr[j] * arr[j - 1] * sub;
                    }
                    dp[i][j][k] = Math.max(takeLeft, takeRight);
                }
            }
        }
        return (int) dp[0][n - 1][K];
    }
}
```

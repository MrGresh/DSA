### Memoization
```java
class Solution {
    Integer[][] memo;
    public int maxTask(int[] h, int[] l) {
        memo = new Integer[h.length][2];
        return rec(h, l, 0, 0);
    }
    int rec(int[] h, int[] l, int i, int flg) {
        if(i==h.length) return 0;
        if(memo[i][flg]!=null) return memo[i][flg];
        int ans = rec(h, l, i+1, 0);
        ans = Math.max(ans, l[i]+rec(h, l, i+1, 1));
        if(i==0 || flg==0) ans = Math.max(ans, h[i]+rec(h, l, i+1, 1));
        return memo[i][flg]=ans;
    }
}
```

### Tabulation
```java
class Solution {
    public int maxTask(int[] h, int[] l) {
        int[][] dp = new int[h.length+1][2];
        for(int i=h.length;i>=0;i--) {
            for(int flg=0;flg<=1;flg++) {
                if(i==h.length) dp[i][flg] = 0;
                else {
                    int ans = dp[i+1][0];
                    ans = Math.max(ans, l[i]+dp[i+1][1]);
                    if(i==0 || flg==0) ans = Math.max(ans, h[i]+dp[i+1][1]);
                    dp[i][flg]=ans;
                }
            }
        }
        return dp[0][0];
    }
}
```

### Space Optimization
```java
class Solution {
    public int maxTask(int[] h, int[] l) {
        int[] prev = new int[2];
        for(int i=h.length;i>=0;i--) {
            int[] curr = new int[2];
            for(int flg=0;flg<=1;flg++) {
                if(i==h.length) curr[flg] = 0;
                else {
                    int ans = prev[0];
                    ans = Math.max(ans, l[i]+prev[1]);
                    if(i==0 || flg==0) ans = Math.max(ans, h[i]+prev[1]);
                    curr[flg]=ans;
                }
            }
            prev = curr;
        }
        return prev[0];
    }
}
```

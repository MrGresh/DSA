### Memoization
```java
class Solution {
    Integer[][] memo;
    public int countWays(int n, int sum) {
        memo = new Integer[n+1][sum+1];
        int ans = rec(sum, n, 0);
        return ans == 0 ? -1 : ans;
    }
    int rec(int sum, int n, int cnt) {
        if(cnt==n) return sum==0 ? 1 : 0;
        if(memo[cnt][sum]!=null) return memo[cnt][sum];
        int ans = 0;
        int digit= (cnt==0) ? 1 : 0;
        while(digit<=9) {
            if(sum-digit>=0) ans += rec(sum-digit, n, cnt+1);
            else break;
            digit++;
        }
        return memo[cnt][sum]=ans;
    }
};
```
### Tabulation
```java
class Solution {
    public int countWays(int n, int Sum) {
        int[][] dp = new int[n+1][Sum+1];
        for(int cnt=n;cnt>=0;cnt--) {
            for(int sum=0;sum<=Sum;sum++) {
                if(cnt==n) dp[cnt][sum] = sum==0 ? 1 : 0;
                else {
                    int ans = 0;
                    int digit= (cnt==0) ? 1 : 0;
                    while(digit<=9) {
                        if(sum-digit>=0) ans += dp[cnt+1][sum-digit];
                        else break;
                        digit++;
                    }
                    dp[cnt][sum]=ans;
                }
            }
        }
        int ans = dp[0][Sum];
        return ans == 0 ? -1 : ans;
    }
};
```
### Space Optimization
```java
class Solution {
    public int countWays(int n, int Sum) {
        int[] prev = new int[Sum+1];
        for(int cnt=n;cnt>=0;cnt--) {
            int[] curr = new int[Sum+1];
            for(int sum=0;sum<=Sum;sum++) {
                if(cnt==n) curr[sum] = sum==0 ? 1 : 0;
                else {
                    int ans = 0;
                    int digit= (cnt==0) ? 1 : 0;
                    while(digit<=9) {
                        if(sum-digit>=0) ans += prev[sum-digit];
                        else break;
                        digit++;
                    }
                    curr[sum]=ans;
                }
            }
            prev = curr;
        }
        int ans = prev[Sum];
        return ans == 0 ? -1 : ans;
    }
};
```

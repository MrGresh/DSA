### Memoization
```java
class Solution {
    Integer[][] memo;
    public int minimumCost(int[] cost, int w) {
        memo = new Integer[cost.length][w+1];
        int ans = Integer.MAX_VALUE;
        ans = rec(cost, w, 0);
        return ans == Integer.MAX_VALUE ? -1 : ans;
    }
    int rec(int[] cost, int w, int i) {
        if(w==0) return 0;
        if(i==cost.length) return Integer.MAX_VALUE;
        if(memo[i][w] != null) return memo[i][w];
        int ans = rec(cost, w, i+1);
        if(w>=(i+1) && cost[i] != -1) {
            int val = rec(cost, w-(i+1), i);
            if(val != Integer.MAX_VALUE) ans = Math.min(ans, cost[i] + val);
        }
        return memo[i][w] = ans;
    }
}
```

### Tabulation
```java
class Solution {
    public int minimumCost(int[] cost, int W) {
        int[][] dp = new int[cost.length+1][W+1];
        for(int i=cost.length;i>=0;i--) {
            for(int w=0;w<=W;w++) {
                if(w==0) dp[i][w] = 0;
                else if(i==cost.length) dp[i][w] = Integer.MAX_VALUE;
                else {
                    int ans = dp[i+1][w];
                    if(w>=(i+1) && cost[i] != -1) {
                        int val = dp[i][w-(i+1)];
                        if(val != Integer.MAX_VALUE) ans = Math.min(ans, cost[i] + val);
                    }
                    dp[i][w] = ans;
                }
            }
        }
        int ans = Integer.MAX_VALUE;
        ans = dp[0][W];
        return ans == Integer.MAX_VALUE ? -1 : ans;
    }
}
```

### Space Optimization
```java
class Solution {
    public int minimumCost(int[] cost, int W) {
        int[] prev = new int[W+1];
        for(int i=cost.length;i>=0;i--) {
            int[] curr = new int[W+1];
            for(int w=0;w<=W;w++) {
                if(w==0) curr[w] = 0;
                else if(i==cost.length) curr[w] = Integer.MAX_VALUE;
                else {
                    int ans = prev[w];
                    if(w>=(i+1) && cost[i] != -1) {
                        int val = curr[w-(i+1)];
                        if(val != Integer.MAX_VALUE) ans = Math.min(ans, cost[i] + val);
                    }
                    curr[w] = ans;
                }
            }
            prev = curr;
        }
        int ans = Integer.MAX_VALUE;
        ans = prev[W];
        return ans == Integer.MAX_VALUE ? -1 : ans;
    }
}
```

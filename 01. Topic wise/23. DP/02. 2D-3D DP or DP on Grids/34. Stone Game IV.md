### Memoization
```java
class Solution {
    Boolean[][] memo;
    public boolean winnerSquareGame(int n) {
        memo = new Boolean[n+1][2];
        return rec(n, 0);
    }
    boolean rec(int n, int turn) {
        if(memo[n][turn]!=null) return memo[n][turn];
        int stone = 1;
        boolean ans = turn==0 ? false : true;
        if(turn==0) {
            while((stone*stone)<=n) {
                if(rec(n-(stone*stone), 1-turn)) {
                    ans = true;
                    break;
                }
                stone++;
            }
        } else {
            while((stone*stone)<=n) {
                if(!rec(n-(stone*stone), 1-turn)) {
                    ans = false;
                    break;
                }
                stone++;
            }
        }
        return memo[n][turn]=ans;
    }
}
```

### Tabulation
```java
class Solution {
    public boolean winnerSquareGame(int N) {
        boolean[][] dp = new boolean[N+1][2];
        for(int n=0;n<=N;n++) {
            for(int turn=0;turn<=1;turn++) {int stone = 1;
                boolean ans = turn==0 ? false : true;
                if(turn==0) {
                    while((stone*stone)<=n) {
                        if(dp[n-(stone*stone)][1-turn]) {
                            ans = true;
                            break;
                        }
                        stone++;
                    }
                } else {
                    while((stone*stone)<=n) {
                        if(!dp[n-(stone*stone)][1-turn]) {
                            ans = false;
                            break;
                        }
                        stone++;
                    }
                }
                dp[n][turn]=ans;
            }
        }
        return dp[N][0];
    }
}
```

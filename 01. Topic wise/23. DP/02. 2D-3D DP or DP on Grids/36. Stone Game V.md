### Memoization
```java
class Solution {
    int[] pref;
    Integer[][] memo;
    public int stoneGameV(int[] stoneValue) {
        pref = new int[stoneValue.length];
        memo = new Integer[stoneValue.length][stoneValue.length];
        pref[0] = stoneValue[0];
        for(int i=1;i<stoneValue.length;i++) pref[i] = pref[i-1]+stoneValue[i];
        return rec(stoneValue, 0, stoneValue.length-1);
    }
    int rec(int[] stoneValue, int l, int r) {
        if(l==r) return 0;
        if(memo[l][r]!=null) return memo[l][r];
        int ans = 0;
        for(int i=l;i<r;i++) {
            int sum1 = pref[i]-pref[l]+stoneValue[l], sum2 = pref[r]-pref[i+1]+stoneValue[i+1];
            if(sum1>sum2) ans = Math.max(ans, sum2+rec(stoneValue, i+1, r));
            else if(sum2>sum1) ans = Math.max(ans, sum1+rec(stoneValue, l, i));
            else {
                ans = Math.max(ans, sum2+rec(stoneValue, i+1, r));
                ans = Math.max(ans, sum1+rec(stoneValue, l, i));
            }
        }
        return memo[l][r]=ans;
    }
}
```

### Tabulation
```java
class Solution {
    public int stoneGameV(int[] stoneValue) {
        int[] pref = new int[stoneValue.length];
        pref[0] = stoneValue[0];
        for(int i=1;i<stoneValue.length;i++) pref[i] = pref[i-1]+stoneValue[i];
        int[][] dp = new int[stoneValue.length][stoneValue.length];
        for(int l=stoneValue.length-1;l>=0;l--) {
            for(int r=l;r<stoneValue.length;r++) {
                if(l==r) dp[l][r] = 0;
                else {
                    int ans = 0;
                    for(int i=l;i<r;i++) {
                        int sum1 = pref[i]-pref[l]+stoneValue[l], sum2 = pref[r]-pref[i+1]+stoneValue[i+1];
                        if(sum1>sum2) ans = Math.max(ans, sum2+dp[i+1][r]);
                        else if(sum2>sum1) ans = Math.max(ans, sum1+dp[l][i]);
                        else {
                            ans = Math.max(ans, sum2+dp[i+1][r]);
                            ans = Math.max(ans, sum1+dp[l][i]);
                        }
                    }
                    dp[l][r]=ans;
                }
            }
        }
        return dp[0][stoneValue.length-1];
    }
}
```

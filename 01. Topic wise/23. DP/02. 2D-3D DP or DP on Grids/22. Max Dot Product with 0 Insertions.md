### Memoization
```java
class Solution {
    Integer[][][] memo;
    public int maxDotProduct(int[] a, int[] b) {
        memo = new Integer[a.length][b.length+1][a.length-b.length+1];
        return rec(a, b, 0, 0, a.length-b.length);
    }
    int rec(int[] a, int[] b, int i, int j, int diff) {
        if(i==a.length) return 0;
        if(memo[i][j][diff]!=null) return memo[i][j][diff];
        int ans = 0;
        if(j<b.length) ans = a[i]*b[j] + rec(a, b, i+1, j+1, diff);
        if(diff>0) ans = Math.max(ans, a[i]*0 + rec(a, b, i+1, j, diff-1));
        return memo[i][j][diff] = ans;
    }
}
```

### Tabulation (TLE)
```java
class Solution {
    public int maxDotProduct(int[] a, int[] b) {
        int[][][] dp = new int[a.length+1][b.length+1][a.length-b.length+1];
        for(int i=a.length;i>=0;i--) {
            for(int j=b.length;j>=0;j--) {
                for(int diff=0;diff<=(a.length-b.length);diff++) {
                    if(i==a.length) dp[i][j][diff] = 0;
                    else {
                        int ans = 0;
                        if(j<b.length) ans = a[i]*b[j] + dp[i+1][j+1][diff];
                        if(diff>0) ans = Math.max(ans, a[i]*0 + dp[i+1][j][diff-1]);
                        dp[i][j][diff] = ans;
                    }
                }
            }
        }
        return dp[0][0][a.length-b.length];
    }
}
```

### Space Optimization (TLE)
```java
class Solution {
    public int maxDotProduct(int[] a, int[] b) {
        int[][] prev = new int[b.length+1][a.length-b.length+1];
        for(int i=a.length;i>=0;i--) {
            int[][] curr = new int[b.length+1][a.length-b.length+1];
            for(int j=b.length;j>=0;j--) {
                for(int diff=0;diff<=(a.length-b.length);diff++) {
                    if(i==a.length) curr[j][diff] = 0;
                    else {
                        int ans = 0;
                        if(j<b.length) ans = a[i]*b[j] + prev[j+1][diff];
                        if(diff>0) ans = Math.max(ans, a[i]*0 + prev[j][diff-1]);
                        curr[j][diff] = ans;
                    }
                }
            }
            prev = curr;
        }
        return prev[0][a.length-b.length];
    }
}
```

### Memoization
```java
class Solution {
    private int MOD = 1_000_000_007;
    private Long[][][] memo;
    
    public int subsequencePairCount(int[] nums) {
        int n = nums.length;
        memo = new Long[n][201][201];
        return (int) (solve(0, 0, 0, nums) % MOD);
    }
    
    private long solve(int idx, int g1, int g2, int[] nums) {
        if (idx == nums.length) return (g1 > 0 && g1 == g2) ? 1 : 0;
        if (memo[idx][g1][g2] != null) return memo[idx][g1][g2];
        
        long res = solve(idx + 1, g1, g2, nums);
        
        int nextG1 = (g1 == 0) ? nums[idx] : gcd(g1, nums[idx]);
        res = (res + solve(idx + 1, nextG1, g2, nums)) % MOD;
        
        int nextG2 = (g2 == 0) ? nums[idx] : gcd(g2, nums[idx]);
        res = (res + solve(idx + 1, g1, nextG2, nums)) % MOD;
        
        return memo[idx][g1][g2] = res;
    }
    
    private int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```

### Tabulation (TLE)
```java
class Solution {
    public int subsequencePairCount(int[] nums) {
        int n = nums.length;
        int MOD = 1_000_000_007;
        Long[][][] dp = new Long[n+1][201][201];
        for(int idx = nums.length;idx>=0;idx--) {
            for(int g1=200;g1>=0;g1--) {
                for(int g2=200;g2>=0;g2--) {
                    if (idx == nums.length) dp[idx][g1][g2] = (g1 > 0 && g1 == g2) ? 1L : 0L;
                    else {
                        long res = dp[idx + 1][g1][g2];
    
                        int nextG1 = (g1 == 0) ? nums[idx] : gcd(g1, nums[idx]);
                        res = (res + dp[idx + 1][nextG1][g2]) % MOD;
                        
                        int nextG2 = (g2 == 0) ? nums[idx] : gcd(g2, nums[idx]);
                        res = (res + dp[idx + 1][g1][nextG2]) % MOD;
                        
                        dp[idx][g1][g2] = res;
                    }
                }
            }
        }
        return (int) (dp[0][0][0] % MOD);
    }
    
    private int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```

### Space Optimization (TLE)
```java
class Solution {
    public int subsequencePairCount(int[] nums) {
        int n = nums.length;
        int MOD = 1_000_000_007;
        Long[][] prev = new Long[201][201];
        for(int idx = nums.length;idx>=0;idx--) {
            Long[][] curr = new Long[201][201];
            for(int g1=200;g1>=0;g1--) {
                for(int g2=200;g2>=0;g2--) {
                    if (idx == nums.length) curr[g1][g2] = (g1 > 0 && g1 == g2) ? 1L : 0L;
                    else {
                        long res = prev[g1][g2];
    
                        int nextG1 = (g1 == 0) ? nums[idx] : gcd(g1, nums[idx]);
                        res = (res + prev[nextG1][g2]) % MOD;
                        
                        int nextG2 = (g2 == 0) ? nums[idx] : gcd(g2, nums[idx]);
                        res = (res + prev[g1][nextG2]) % MOD;
                        
                        curr[g1][g2] = res;
                    }
                }
            }
            prev = curr;
        }
        return (int) (prev[0][0] % MOD);
    }
    
    private int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```

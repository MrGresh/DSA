### Memoization
```java
class Solution {
    Integer[][] memo;
    int MOD = 1000000007;
    public int ways(int x, int y) {
        memo = new Integer[x+1][y+1];
        return rec(x, y);
    }
    int rec(int x, int y) {
        if(x==0 && y==0) return 1;
        if(memo[x][y]!=null) return memo[x][y];
        int cnt = 0;
        if(x-1>=0) cnt = (cnt + rec(x-1, y)) % MOD;
        if(y-1>=0) cnt = (cnt + rec(x, y-1)) % MOD;
        return memo[x][y] = cnt;
    }
}
```
### Tabulation
```java
class Solution {
    public int ways(int X, int Y) {
        int MOD = 1000000007;
        int[][] dp = new int[X+1][Y+1];
        for(int x=0;x<=X;x++) {
            for(int y=0;y<=Y;y++) {
                if(x==0 && y==0) dp[x][y] = 1;
                else {
                    int cnt = 0;
                    if(x-1>=0) cnt = (cnt + dp[x-1][y]) % MOD;
                    if(y-1>=0) cnt = (cnt + dp[x][y-1]) % MOD;
                    dp[x][y] = cnt;
                }
            }
        }
        return dp[X][Y];
    }
}
```
### Space Optimization
```java
class Solution {
    public int ways(int X, int Y) {
        int MOD = 1000000007;
        int[] prev = new int[Y+1];
        for(int x=0;x<=X;x++) {
            int[] curr = new int[Y+1];
            for(int y=0;y<=Y;y++) {
                if(x==0 && y==0) curr[y] = 1;
                else {
                    int cnt = 0;
                    if(x-1>=0) cnt = (cnt + prev[y]) % MOD;
                    if(y-1>=0) cnt = (cnt + curr[y-1]) % MOD;
                    curr[y] = cnt;
                }
            }
            prev = curr;
        }
        return prev[Y];
    }
}
```

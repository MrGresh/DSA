### Memoization
```java
class Solution {
    Integer[][] memo;
    Integer[][] memo2;
    int MOD = 1000000007;
    public ArrayList<Integer> findWays(int[][] grid) {
        memo = new Integer[grid.length][grid[0].length];
        memo2 = new Integer[grid.length][grid[0].length];
        int maxAdventure = rec(grid, 0, 0);
        int totalPaths = rec2(grid, 0, 0);
        if(maxAdventure==Integer.MIN_VALUE) maxAdventure = 0;
        return new ArrayList<>(Arrays.asList(totalPaths, maxAdventure));
    }
    int rec(int[][] grid, int i, int j) {
        if(i==grid.length-1 && j==grid[0].length-1) return grid[grid.length-1][grid[0].length-1];
        if(memo[i][j]!=null) return memo[i][j];
        int ans = Integer.MIN_VALUE;
        if(grid[i][j]==1) {
            if(j<grid[0].length-1) {
                int val = rec(grid, i, j+1);
                if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
            }
        } else if(grid[i][j]==2) {
            if(i<grid.length-1) {
                int val = rec(grid, i+1, j);
                if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
            }
        } else {
            if(i<grid.length-1) {
                int val = rec(grid, i+1, j);
                if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
            }
            if(j<grid[0].length-1) {
                int val = rec(grid, i, j+1);
                if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
            }
        }
        return memo[i][j]=ans;
    }
    int rec2(int[][] grid, int i, int j) {
        if(i==grid.length-1 && j==grid[0].length-1) return 1;
        if(memo2[i][j]!=null) return memo2[i][j];
        int ans = 0;
        if(grid[i][j]==1) {
            if(j<grid[0].length-1) ans = (ans + rec2(grid, i, j+1)) % MOD;
        } else if(grid[i][j]==2) {
            if(i<grid.length-1) ans = (ans + rec2(grid, i+1, j)) % MOD;
        } else {
            if(i<grid.length-1) ans = (ans + rec2(grid, i+1, j)) % MOD;
            if(j<grid[0].length-1) ans = (ans + rec2(grid, i, j+1)) % MOD;
        }
        return memo2[i][j]=ans;
    }
}
```

### Tabulation
```java
class Solution {
    public ArrayList<Integer> findWays(int[][] grid) {
        int MOD = 1000000007;
        int[][] dp = new int[grid.length][grid[0].length];
        int[][] dp2 = new int[grid.length][grid[0].length];
        for(int i=grid.length-1;i>=0;i--) {
            for(int j=grid[0].length-1;j>=0;j--) {
                if(i==grid.length-1 && j==grid[0].length-1) dp[i][j] = grid[i][j];
                else {
                    int ans = Integer.MIN_VALUE;
                    if(grid[i][j]==1) {
                        if(j<grid[0].length-1) {
                            int val = dp[i][j+1];
                            if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
                        }
                    } else if(grid[i][j]==2) {
                        if(i<grid.length-1) {
                            int val = dp[i+1][j];
                            if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
                        }
                    } else {
                        if(i<grid.length-1) {
                            int val = dp[i+1][j];
                            if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
                        }
                        if(j<grid[0].length-1) {
                            int val = dp[i][j+1];
                            if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
                        }
                    }
                    dp[i][j]=ans;
                }
            }
        }
        for(int i=grid.length-1;i>=0;i--) {
            for(int j=grid[0].length-1;j>=0;j--) {
                if(i==grid.length-1 && j==grid[0].length-1) dp2[i][j] = 1;
                else {
                    int ans = 0;
                    if(grid[i][j]==1) {
                        if(j<grid[0].length-1) ans = (ans + dp2[i][j+1]) % MOD;
                    } else if(grid[i][j]==2) {
                        if(i<grid.length-1) ans = (ans + dp2[i+1][j]) % MOD;
                    } else {
                        if(i<grid.length-1) ans = (ans + dp2[i+1][j]) % MOD;
                        if(j<grid[0].length-1) ans = (ans + dp2[i][j+1]) % MOD;
                    }
                    dp2[i][j]=ans;
                }
            }
        }
        int maxAdventure = dp[0][0];
        int totalPaths = dp2[0][0];
        if(maxAdventure==Integer.MIN_VALUE) maxAdventure = 0;
        return new ArrayList<>(Arrays.asList(totalPaths, maxAdventure));
    }
}
```

```java
class Solution {
    public ArrayList<Integer> findWays(int[][] grid) {
        int MOD = 1000000007;
        int[] prev = new int[grid[0].length];
        int[] prev2 = new int[grid[0].length];
        for(int i=grid.length-1;i>=0;i--) {
            int[] curr = new int[grid[0].length];
            for(int j=grid[0].length-1;j>=0;j--) {
                if(i==grid.length-1 && j==grid[0].length-1) curr[j] = grid[i][j];
                else {
                    int ans = Integer.MIN_VALUE;
                    if(grid[i][j]==1) {
                        if(j<grid[0].length-1) {
                            int val = curr[j+1];
                            if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
                        }
                    } else if(grid[i][j]==2) {
                        if(i<grid.length-1) {
                            int val = prev[j];
                            if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
                        }
                    } else {
                        if(i<grid.length-1) {
                            int val = prev[j];
                            if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
                        }
                        if(j<grid[0].length-1) {
                            int val = curr[j+1];
                            if(val!=Integer.MIN_VALUE) ans = Math.max(ans, grid[i][j]+val);
                        }
                    }
                    curr[j]=ans;
                }
            }
            prev = curr;
        }
        for(int i=grid.length-1;i>=0;i--) {
            int[] curr = new int[grid[0].length];
            for(int j=grid[0].length-1;j>=0;j--) {
                if(i==grid.length-1 && j==grid[0].length-1) curr[j] = 1;
                else {
                    int ans = 0;
                    if(grid[i][j]==1) {
                        if(j<grid[0].length-1) ans = (ans + curr[j+1]) % MOD;
                    } else if(grid[i][j]==2) {
                        if(i<grid.length-1) ans = (ans + prev2[j]) % MOD;
                    } else {
                        if(i<grid.length-1) ans = (ans + prev2[j]) % MOD;
                        if(j<grid[0].length-1) ans = (ans + curr[j+1]) % MOD;
                    }
                    curr[j]=ans;
                }
            }
            prev2 = curr;
        }
        int maxAdventure = prev[0];
        int totalPaths = prev2[0];
        if(maxAdventure==Integer.MIN_VALUE) maxAdventure = 0;
        return new ArrayList<>(Arrays.asList(totalPaths, maxAdventure));
    }
}
```

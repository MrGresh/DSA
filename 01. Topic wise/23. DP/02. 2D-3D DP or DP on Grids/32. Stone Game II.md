### Memoization
```java
class Solution {
    Integer[][][] memo;
    public int stoneGameII(int[] piles) {
        memo = new Integer[piles.length][2][2*piles.length];
        return rec(piles, 0, 1, 1);
    }
    int rec(int[] piles, int idx, int turn, int M) {
        if(idx==piles.length) return 0;
        if(memo[idx][turn][M]!=null) return memo[idx][turn][M];
        int val = turn==1 ? 0 : Integer.MAX_VALUE;
        if(turn == 1) {
            int sum = 0;
            for(int i=idx;i<=(idx+2*M-1) && i<piles.length;i++) {
                sum += piles[i];
                val = Math.max(val, sum+rec(piles, i+1, 0, Math.max(M, i-idx+1)));
            }
        } else {
            for(int i=idx;i<=(idx+2*M-1) && i<piles.length;i++) {
                val = Math.min(val, rec(piles, i+1, 1, Math.max(M, i-idx+1)));
            }
        }
        return memo[idx][turn][M] = val;
    }
}
```

### Tabulation
```java
class Solution {
    public int stoneGameII(int[] piles) {
        int[][][] dp = new int[piles.length+1][2][2*piles.length+1];
        for(int idx=piles.length;idx>=0;idx--) {
            for(int turn=1;turn>=0;turn--) {
                for(int M=2*piles.length;M>=1;M--) {
                    if(idx==piles.length) dp[idx][turn][M] = 0;
                    else {
                        int val = turn==1 ? 0 : Integer.MAX_VALUE;
                        if(turn == 1) {
                            int sum = 0;
                            for(int i=idx;i<=(idx+2*M-1) && i<piles.length;i++) {
                                sum += piles[i];
                                val = Math.max(val, sum+dp[i+1][0][Math.max(M, i-idx+1)]);
                            }
                        } else {
                            for(int i=idx;i<=(idx+2*M-1) && i<piles.length;i++) {
                                val = Math.min(val, dp[i+1][1][Math.max(M, i-idx+1)]);
                            }
                        }
                        dp[idx][turn][M] = val;
                    }
                }
            }
        }
        return dp[0][1][1];
    }
}
```

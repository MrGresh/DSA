### Memoization
```java
class Solution {
    Integer[][] memo;
    public int maxPathSum(int[] a, int[] b) {
        memo = new Integer[Math.max(a.length, b.length)][2];
        return Math.max(rec(a, b, 0, 0), rec(a, b, 0, 1));
    }
    int rec(int[] a, int[] b, int idx, int array) {
        if((array==0 && idx==a.length) || (array==1 && idx==b.length)) return 0;
        if(memo[idx][array]!=null) return memo[idx][array];
        int val = rec(a, b, idx+1, array);
        if(array==0) {
            int l = 0, r = b.length-1;
            while(l<=r) {
                int m = l+(r-l)/2;
                if(b[m]==a[idx]) {
                    val = Math.max(val, rec(a, b, m+1, 1));
                    break;
                } else if(b[m]<a[idx]) l=m+1;
                else r=m-1;
            }
        } else {
            int l = 0, r = a.length-1;
            while(l<=r) {
                int m = l+(r-l)/2;
                if(a[m]==b[idx]) {
                    val = Math.max(val, rec(a, b, m+1, 0));
                    break;
                } else if(a[m]<b[idx]) l=m+1;
                else r=m-1;
            }
        }
        return memo[idx][array] = val+(array==0 ? a[idx] : b[idx]);
    }
}
```

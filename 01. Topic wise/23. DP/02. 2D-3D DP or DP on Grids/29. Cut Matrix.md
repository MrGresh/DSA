### Memoization
```java
class Solution {
    int MOD = 1000000007;
    Integer[][][] memo;
    public int findWays(int[][] mat, int k) {
        for(int j=mat[0].length-2;j>=0;j--) mat[mat.length-1][j] += mat[mat.length-1][j+1];
        for(int i=mat.length-2;i>=0;i--) mat[i][mat[0].length-1] += mat[i+1][mat[0].length-1];
        for(int i=mat.length-2;i>=0;i--) {
            for(int j=mat[0].length-2;j>=0;j--) mat[i][j] += (mat[i+1][j] + mat[i][j+1] - mat[i+1][j+1]);
        }
        memo = new Integer[mat.length][mat[0].length][k+1];
        if(mat[0][0]==0) return 0;
        return rec(mat, k , 0, 0);
    }
    int rec(int[][] mat, int k, int i, int j) {
        if(k==1) return 1;
        if(memo[i][j][k] != null) return memo[i][j][k];
        int ans = 0;
        for(int col=j+1;col<mat[0].length;col++) {
            int newCutOnes = mat[i][col];
            int prevCutOnes = mat[i][j]-mat[i][col];
            if(newCutOnes>0 && prevCutOnes>0) ans = (ans + rec(mat, k-1, i, col)) % MOD;
        }
        for(int row=i+1;row<mat.length;row++) {
            int newCutOnes = mat[row][j];
            int prevCutOnes = mat[i][j]-mat[row][j];
            if(newCutOnes>0 && prevCutOnes>0) ans = (ans + rec(mat, k-1, row, j)) % MOD;
        }
        return memo[i][j][k] = ans;
    }
}
```

```java
class Solution {
    public int findWays(int[][] mat, int K) {
        int MOD = 1000000007;
        for(int j=mat[0].length-2;j>=0;j--) mat[mat.length-1][j] += mat[mat.length-1][j+1];
        for(int i=mat.length-2;i>=0;i--) mat[i][mat[0].length-1] += mat[i+1][mat[0].length-1];
        for(int i=mat.length-2;i>=0;i--) {
            for(int j=mat[0].length-2;j>=0;j--) mat[i][j] += (mat[i+1][j] + mat[i][j+1] - mat[i+1][j+1]);
        }
        int[][][] dp = new int[K+1][mat.length][mat[0].length];
        if(mat[0][0]==0) return 0;
        for(int k=1;k<=K;k++) {
            for(int i=mat.length-1;i>=0;i--) {
                for(int j=mat[0].length-1;j>=0;j--) {
                    if(k==1) dp[k][i][j] = (mat[i][j] > 0 ? 1 : 0);
                    else {
                        int ans = 0;
                        for(int col=j+1;col<mat[0].length;col++) {
                            int newCutOnes = mat[i][col];
                            int prevCutOnes = mat[i][j]-mat[i][col];
                            if(newCutOnes>0 && prevCutOnes>0) ans = (ans + dp[k-1][i][col]) % MOD;
                        }
                        for(int row=i+1;row<mat.length;row++) {
                            int newCutOnes = mat[row][j];
                            int prevCutOnes = mat[i][j]-mat[row][j];
                            if(newCutOnes>0 && prevCutOnes>0) ans = (ans + dp[k-1][row][j]) % MOD;
                        }
                        dp[k][i][j] = ans;
                    }
                }
            }
        }
        return dp[K][0][0];
    }
}
```

```java
        }
        int[][] prev = new int[mat.length][mat[0].length];
        if(mat[0][0]==0) return 0;
        for(int k=1;k<=K;k++) {
            int[][] curr = new int[mat.length][mat[0].length];
            for(int i=mat.length-1;i>=0;i--) {
                for(int j=mat[0].length-1;j>=0;j--) {
                    if(k==1) curr[i][j] = (mat[i][j] > 0 ? 1 : 0);
                    else {
                        int ans = 0;
                        for(int col=j+1;col<mat[0].length;col++) {
                            int newCutOnes = mat[i][col];
                            int prevCutOnes = mat[i][j]-mat[i][col];
                            if(newCutOnes>0 && prevCutOnes>0) ans = (ans + prev[i][col]) % MOD;
                        }
                        for(int row=i+1;row<mat.length;row++) {
                            int newCutOnes = mat[row][j];
                            int prevCutOnes = mat[i][j]-mat[row][j];
                            if(newCutOnes>0 && prevCutOnes>0) ans = (ans + prev[row][j]) % MOD;
                        }
                        curr[i][j] = ans;
                    }
                }
            }
            prev = curr;
        }
        return prev[0][0];
    }
}
```

### Memoization
```java
class Solution {
    Integer[][] memo;
    public int count(int n, int m) {
        memo = new Integer[n+1][12];
        return rec(m, n, 0);
    }
    int rec(int m, int n, int prev) {
        if(n==0) return 1;
        if(memo[n][prev] != null) return memo[n][prev];
        int ans = 0;
        for(int j=1;j<=m;j++) {
            if(prev==0 || prev%j==0 || j%prev==0) ans += rec(m, n-1, j);
        }
        return memo[n][prev] = ans;
    }
}
```
### Tabulation
```java
class Solution {
    public int count(int N, int m) {
        int[][] dp = new int[N+1][12];
        for(int n=0;n<=N;n++) {
            for(int prev=11;prev>=0;prev--) {
                if(n==0) dp[n][prev] = 1;
                else {
                    int ans = 0;
                    for(int j=1;j<=m;j++) {
                        if(prev==0 || prev%j==0 || j%prev==0) ans += dp[n-1][j];
                    }
                    dp[n][prev] = ans;
                }
            }
        }
        return dp[N][0];
    }
}
```
### Space Optimization
```java
class Solution {
    public int count(int N, int m) {
        int[] prv = new int[12];
        for(int n=0;n<=N;n++) {
            int[] curr = new int[12];
            for(int prev=11;prev>=0;prev--) {
                if(n==0) curr[prev] = 1;
                else {
                    int ans = 0;
                    for(int j=1;j<=m;j++) {
                        if(prev==0 || prev%j==0 || j%prev==0) ans += prv[j];
                    }
                    curr[prev] = ans;
                }
            }
            prv = curr;
        }
        return prv[0];
    }
}
```

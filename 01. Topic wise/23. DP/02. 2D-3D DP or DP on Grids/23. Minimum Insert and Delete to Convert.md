### Memoization
```java
class Solution {
    Integer[][] memo;
    public int minInsAndDel(int[] a, int[] b) {
        memo = new Integer[a.length][b.length];
        return rec(a, b, 0, 0);
    }
    int rec(int[] a, int[] b, int i, int j) {
        if(j==b.length) return a.length-i;
        if(i==a.length) return b.length-j;
        if(memo[i][j]!=null) return memo[i][j];
        int ans = Integer.MAX_VALUE;
        if(a[i]==b[j]) ans = rec(a, b, i+1, j+1);
        else ans = Math.min(1+rec(a, b, i+1, j), 1+rec(a, b, i, j+1));
        return memo[i][j] = ans;
    }
}
```

### Tabulation
```java
class Solution {
    public int minInsAndDel(int[] a, int[] b) {
        int[][] dp = new int[a.length+1][b.length+1];
        for(int i=a.length;i>=0;i--) {
            for(int j=b.length;j>=0;j--) {
                if(j==b.length) dp[i][j] = a.length-i;
                else if(i==a.length) dp[i][j] = b.length-j;
                else {
                    int ans = Integer.MAX_VALUE;
                    if(a[i]==b[j]) ans = dp[i+1][j+1];
                    else ans = Math.min(1+dp[i+1][j], 1+dp[i][j+1]);
                    dp[i][j] = ans;
                }
            }
        }
        return dp[0][0];
    }
}
```

### Space Optimization
```java
class Solution {
    public int minInsAndDel(int[] a, int[] b) {
        int[] prev = new int[b.length+1];
        for(int i=a.length;i>=0;i--) {
            int[] curr = new int[b.length+1];
            for(int j=b.length;j>=0;j--) {
                if(j==b.length) curr[j] = a.length-i;
                else if(i==a.length) curr[j] = b.length-j;
                else {
                    int ans = Integer.MAX_VALUE;
                    if(a[i]==b[j]) ans = prev[j+1];
                    else ans = Math.min(1+prev[j], 1+curr[j+1]);
                    curr[j] = ans;
                }
            }
            prev = curr;
        }
        return prev[0];
    }
}
```

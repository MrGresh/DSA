### Memoization
```java
class Solution {
    Integer[][] memo;
    public int numDistinct(String s, String t) {
        memo = new Integer[s.length()][t.length()];
        return rec(s, t, 0, 0);
    }
    int rec(String s, String t, int i, int j) {
        if(j==t.length()) return 1;
        if(i==s.length()) return 0;
        if(memo[i][j]!=null) return memo[i][j];
        int cnt = rec(s, t, i+1, j);
        if(s.charAt(i)==t.charAt(j)) cnt += rec(s, t, i+1, j+1);
        return memo[i][j] = cnt;
    }
}
```
### Tabulation
```java
class Solution {
    public int numDistinct(String s, String t) {
        int[][] dp = new int[s.length()+1][t.length()+1];
        for(int i=s.length();i>=0;i--) {
            for(int j=t.length();j>=0;j--) {
                if(j==t.length()) dp[i][j] = 1;
                else if(i==s.length()) dp[i][j] = 0;
                else {
                    int cnt = dp[i+1][j];
                    if(s.charAt(i)==t.charAt(j)) cnt += dp[i+1][j+1];
                    dp[i][j] = cnt;
                }
            }
        }
        return dp[0][0];
    }
}
```

### Space Optimization
```java
class Solution {
    public int numDistinct(String s, String t) {
        int[] prev = new int[t.length()+1];
        for(int i=s.length();i>=0;i--) {
            int[] curr = new int[t.length()+1];
            for(int j=t.length();j>=0;j--) {
                if(j==t.length()) curr[j] = 1;
                else if(i==s.length()) curr[j] = 0;
                else {
                    int cnt = prev[j];
                    if(s.charAt(i)==t.charAt(j)) cnt += prev[j+1];
                    curr[j] = cnt;
                }
            }
            prev = curr;
        }
        return prev[0];
    }
}
```

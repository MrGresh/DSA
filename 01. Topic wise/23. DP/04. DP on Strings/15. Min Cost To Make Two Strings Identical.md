### Memoization
```java
class Solution {
    Integer[][] memo;
    public int findMinCost(String s1, String s2, int costS1, int costS2) {
        memo = new Integer[s1.length()+1][s2.length()+1];
        return rec(s1, s2, costS1, costS2, 0, 0);
    }
    int rec(String s1, String s2, int costS1, int costS2, int i, int j) {
        if(i==s1.length()) return costS2*(s2.length()-j);
        if(j==s2.length()) return costS1*(s1.length()-i);
        if(memo[i][j]!=null) return memo[i][j];
        if(s1.charAt(i)==s2.charAt(j)) {
            return memo[i][j] = rec(s1, s2, costS1, costS2, i+1, j+1);
        } else {
            return memo[i][j] = Math.min(
                costS2 + rec(s1, s2, costS1, costS2, i, j+1),
                costS1 + rec(s1, s2, costS1, costS2, i+1, j)
                );
        }
    }
}
```

### Tabulation
```java
class Solution {
    public int findMinCost(String s1, String s2, int costS1, int costS2) {
        int[][] dp = new int[s1.length()+1][s2.length()+1];
        for(int i=s1.length();i>=0;i--) {
            for(int j=s2.length();j>=0;j--) {
                if(i==s1.length()) dp[i][j] = costS2*(s2.length()-j);
                else if(j==s2.length()) dp[i][j] = costS1*(s1.length()-i);
                else {
                    if(s1.charAt(i)==s2.charAt(j)) {
                        dp[i][j] = dp[i+1][j+1];
                    } else {
                        dp[i][j] = Math.min(costS1 + dp[i+1][j], costS2 + dp[i][j+1]);
                    }
                }
            }
         }
        return dp[0][0];
    }
}
```

### Space Optimization
```java
class Solution {
    public int findMinCost(String s1, String s2, int costS1, int costS2) {
        int[] prev = new int[s2.length()+1];
        for(int i=s1.length();i>=0;i--) {
            int[] curr = new int[s2.length()+1];
            for(int j=s2.length();j>=0;j--) {
                if(i==s1.length()) curr[j] = costS2*(s2.length()-j);
                else if(j==s2.length()) curr[j] = costS1*(s1.length()-i);
                else {
                    if(s1.charAt(i)==s2.charAt(j)) {
                        curr[j] = prev[j+1];
                    } else {
                        curr[j] = Math.min(costS1 + prev[j], costS2 + curr[j+1]);
                    }
                }
            }
            prev = curr;
         }
        return prev[0];
    }
}
```

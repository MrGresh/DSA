### Memoization
```java
class Solution {
    Integer[] memo;
    public int maxJumps(int[] arr, int d) {
        memo = new Integer[arr.length];
        int ans = 0;
        for(int i=0;i<arr.length;i++) {
            ans = Math.max(ans, rec(arr, d, i));
        }
        return ans;
    }
    int rec(int[] arr, int d, int i) {
        if(memo[i]!=null) return memo[i];
        int ans = 0;
        for(int x=1;x<=d;x++) {
            if((i+x<arr.length) && (arr[i]>arr[i+x])) {
                ans = Math.max(ans, rec(arr, d, i+x));
            } else break;
        }
        for(int x=1;x<=d;x++) {
            if((i-x>=0) && (arr[i]>arr[i-x])) {
                ans = Math.max(ans, rec(arr, d, i-x));
            } else break;
        }
        return memo[i] = 1 + ans;
    }
}
```
### Tabulation
```java
class Solution {
    public int maxJumps(int[] arr, int d) {
        int n = arr.length;
        int[] dp = new int[n];

        Integer[] indices = new Integer[n];
        for (int i = 0; i < n; i++) indices[i] = i;
        Arrays.sort(indices, (a, b) -> Integer.compare(arr[a], arr[b]));

        int ans = 0;
        for(int idx=0;idx<n;idx++) {
            int i = indices[idx];
            int val = 0;
            for(int x=1;x<=d;x++) {
                if((i+x<arr.length) && (arr[i]>arr[i+x])) {
                    val = Math.max(val, dp[i+x]);
                } else break;
            }
            for(int x=1;x<=d;x++) {
                if((i-x>=0) && (arr[i]>arr[i-x])) {
                    val = Math.max(val, dp[i-x]);
                } else break;
            }
            dp[i] = 1 + val;
            ans = Math.max(ans, dp[i]);
        }
        return ans;
    }
}
```

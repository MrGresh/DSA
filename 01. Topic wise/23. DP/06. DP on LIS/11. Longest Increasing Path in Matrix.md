### Memoization
```java
class Solution {
    Integer[][] memo;
    public int longIncPath(int[][] mat, int r, int c) {
        memo = new Integer[r][c];
        int ans = 1;
        for(int i=0;i<r;i++) {
            for(int j=0;j<c;j++) {
                ans = Math.max(ans, rec(mat, r, c, i, j));
            }
        }
        return ans;
    }
    int rec(int[][] mat, int r, int c, int i, int j) {
        if(memo[i][j]!=null) return memo[i][j];
        int val = 0;
        if(i-1>=0 && mat[i-1][j]>mat[i][j]) val = rec(mat, r, c, i-1, j);
        if(j-1>=0 && mat[i][j-1]>mat[i][j]) val = Math.max(val, rec(mat, r, c, i, j-1));
        if(i+1<r && mat[i+1][j]>mat[i][j]) val = Math.max(val, rec(mat, r, c, i+1, j));
        if(j+1<c && mat[i][j+1]>mat[i][j]) val = Math.max(val, rec(mat, r, c, i, j+1));
        return memo[i][j]=1+val;
    }
}
```

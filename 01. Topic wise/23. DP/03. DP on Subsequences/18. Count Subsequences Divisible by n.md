### Memoization
```java
class Solution {
    int MOD = 1_000_000_007;
    int n;
    String s;
    Integer[][] memo;

    public int countSubsequences(String s, int n) {
        this.s = s;
        this.n = n;
        memo = new Integer[s.length()][n];
        int total = solve(0, 0);
        return (int) (((long) total - 1 + MOD) % MOD);
    }

    int solve(int i, int rem) {
        if (i == s.length()) {
            return rem == 0 ? 1 : 0;
        }
        if (memo[i][rem] != null) return memo[i][rem];

        int skip = solve(i + 1, rem);

        int digit = s.charAt(i) - '0';
        int newRem = (rem * 10 + digit) % n;
        int include = solve(i + 1, newRem);

        int result = (int) (((long) skip + include) % MOD);
        memo[i][rem] = result;
        return result;
    }
}
```

### Tabulation
```java
class Solution {
    public int countSubsequences(String s, int n) {
        int MOD = 1_000_000_007;
        int[][] dp = new int[s.length()+1][n];
        for(int i=s.length();i>=0;i--) {
            for(int rem=0;rem<n;rem++) {
                if (i == s.length()) dp[i][rem] = rem == 0 ? 1 : 0;
                else {
                    int skip = dp[i+1][rem];
            
                    int digit = s.charAt(i) - '0';
                    int newRem = (rem * 10 + digit) % n;
                    int include = dp[i+1][newRem];
            
                    int result = (int) (((long) skip + include) % MOD);
                    dp[i][rem] = result;
                }
            }
        }
        int total = dp[0][0];
        return (int) (((long) total - 1 + MOD) % MOD);
    }
}
```

### Space Optimization
```java
class Solution {
    public int countSubsequences(String s, int n) {
        int MOD = 1_000_000_007;
        int[] prev = new int[n];
        for(int i=s.length();i>=0;i--) {
            int[] curr = new int[n];
            for(int rem=0;rem<n;rem++) {
                if (i == s.length()) curr[rem] = rem == 0 ? 1 : 0;
                else {
                    int skip = prev[rem];
            
                    int digit = s.charAt(i) - '0';
                    int newRem = (rem * 10 + digit) % n;
                    int include = prev[newRem];
            
                    int result = (int) (((long) skip + include) % MOD);
                    curr[rem] = result;
                }
            }
            prev = curr;
        }
        int total = prev[0];
        return (int) (((long) total - 1 + MOD) % MOD);
    }
}
```

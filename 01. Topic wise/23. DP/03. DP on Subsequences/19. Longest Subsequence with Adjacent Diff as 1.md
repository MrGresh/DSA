### Memoization
```java
class Solution {
    Integer[][] memo;
    public int longestSubseq(int[] arr) {
        memo = new Integer[arr.length][arr.length+1];
        return rec(arr, 0, -1);
    }
    int rec(int[] arr, int i, int prevIdx) {
        if(i==arr.length) return 0;
        int key = prevIdx+1;
        if(memo[i][key]!=null) return memo[i][key];
        int val = rec(arr, i+1, prevIdx);
        if(prevIdx==-1 || Math.abs(arr[i]-arr[prevIdx])==1) {
            val = Math.max(val, 1+rec(arr, i+1, i));
        }
        return memo[i][key]=val;
    }
}
```

### Tabulation
```java
class Solution {
    public int longestSubseq(int[] arr) {
        int[][] dp = new int[arr.length+1][arr.length+1];
        for(int i=arr.length;i>=0;i--) {
            for(int prevIdx=arr.length-1;prevIdx>=-1;prevIdx--) {
                int key = prevIdx+1;
                if(i==arr.length) dp[i][key] = 0;
                else {
                    int val = dp[i+1][key];
                    if(prevIdx==-1 || Math.abs(arr[i]-arr[prevIdx])==1) {
                        val = Math.max(val, 1+dp[i+1][i+1]);
                    }
                    dp[i][key]=val;
                }
            }
        }
        return dp[0][0];
    }
}
```

### Space Optimization
```java
class Solution {
    public int longestSubseq(int[] arr) {
        int[] prev = new int[arr.length+1];
        for(int i=arr.length;i>=0;i--) {
            int[] curr = new int[arr.length+1];
            for(int prevIdx=arr.length-1;prevIdx>=-1;prevIdx--) {
                int key = prevIdx+1;
                if(i==arr.length) curr[key] = 0;
                else {
                    int val = prev[key];
                    if(prevIdx==-1 || Math.abs(arr[i]-arr[prevIdx])==1) {
                        val = Math.max(val, 1+prev[i+1]);
                    }
                    curr[key]=val;
                }
            }
            prev = curr;
        }
        return prev[0];
    }
}
```

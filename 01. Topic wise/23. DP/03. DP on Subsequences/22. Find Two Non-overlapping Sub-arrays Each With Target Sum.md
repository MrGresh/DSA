### Memoization
```java
class Solution {
    int[] windowStart;
    Integer[] memo;

    public int minSumOfLengths(int[] arr, int target) {
        int n = arr.length;
        windowStart = new int[n];
        Arrays.fill(windowStart, -1);
        memo = new Integer[n];

        // Step 1: sliding window — precompute, for each right, the left index
        // such that arr[left..right] sums to target (or -1 if none).
        int left = 0, sum = 0;
        for (int right = 0; right < n; right++) {
            sum += arr[right];
            while (sum > target) {
                sum -= arr[left];
                left++;
            }
            if (sum == target) {
                windowStart[right] = left;
            }
        }

        // Step 2: for every valid window, pair it with the best (shortest)
        // valid subarray strictly before it — found via memoized recursion.
        int ans = Integer.MAX_VALUE;
        for (int right = 0; right < n; right++) {
            if (windowStart[right] != -1) {
                int len = right - windowStart[right] + 1;
                int prevBest = bestUpTo(windowStart[right] - 1);
                if (prevBest != Integer.MAX_VALUE) {
                    ans = Math.min(ans, len + prevBest);
                }
            }
        }

        return ans == Integer.MAX_VALUE ? -1 : ans;
    }

    // Memoized: shortest valid subarray fully contained within arr[0..i]
    private int bestUpTo(int i) {
        if (i < 0) return Integer.MAX_VALUE;
        if (memo[i] != null) return memo[i];

        int result = bestUpTo(i - 1); // carry forward best from before i

        if (windowStart[i] != -1) {
            result = Math.min(result, i - windowStart[i] + 1);
        }

        memo[i] = result;
        return result;
    }
}
```

### Raw Recursion (TLE)
```java
class Solution {
    public static int countWays(String s1, String s2) {
        return rec(s1, s2, 0, new StringBuilder());
    }
    static int rec(String s1, String s2, int i, StringBuilder curr) {
        if(i==s1.length()) {
            return curr.toString().equals(s2) ? 1 : 0;
        }
        curr.append(s1.charAt(i));
        int ans = rec(s1, s2, i+1, curr);
        curr.deleteCharAt(curr.length()-1);
        ans += rec(s1, s2, i+1, curr);
        return ans;
    }
}
```

### Memoization
```java
class Solution {
    static int MOD = 1000000007;
    static Integer[][] memo;
    public static int countWays(String s1, String s2) {
        memo = new Integer[s1.length()][s2.length()];
        return rec(s1, s2, 0, 0);
    }
    static int rec(String s1, String s2, int i, int j) {
        if(j==s2.length()) return 1;
        if(i==s1.length() || ((s2.length()-j)-(s1.length()-i)>0)) return 0;
        if(memo[i][j]!=null) return memo[i][j];
        int ans = 0;
        if(s1.charAt(i)==s2.charAt(j)) ans += rec(s1, s2, i+1, j+1);
        ans = (ans + rec(s1, s2, i+1, j)) % MOD;
        return memo[i][j] = ans;
    }
}
```

### Tabulation
```java
class Solution {
    public static int countWays(String s1, String s2) {
        int MOD = 1000000007;
        int[][] dp = new int[s1.length()+1][s2.length()+1];
        for(int i=s1.length();i>=0;i--) {
            for(int j=s2.length();j>=0;j--) {
                if(j==s2.length()) dp[i][j] = 1;
                else if(i==s1.length() || ((s2.length()-j)-(s1.length()-i)>0)) dp[i][j] = 0;
                else {
                    int ans = 0;
                    if(s1.charAt(i)==s2.charAt(j)) ans += dp[i+1][j+1];
                    ans = (ans + dp[i+1][j]) % MOD;
                    dp[i][j] = ans;
                }
            }
        }
        return dp[0][0];
    }
}
```

### Space Optimization
```java
class Solution {
    public static int countWays(String s1, String s2) {
        int MOD = 1000000007;
        int[] prev = new int[s2.length()+1];
        for(int i=s1.length();i>=0;i--) {
            int[] curr = new int[s2.length()+1];
            for(int j=s2.length();j>=0;j--) {
                if(j==s2.length()) curr[j] = 1;
                else if(i==s1.length() || ((s2.length()-j)-(s1.length()-i)>0)) curr[j] = 0;
                else {
                    int ans = 0;
                    if(s1.charAt(i)==s2.charAt(j)) ans += prev[j+1];
                    ans = (ans + prev[j]) % MOD;
                    curr[j] = ans;
                }
            }
            prev = curr;
        }
        return prev[0];
    }
}
```

### Memoization
```java
class Solution {
    int n;
    int[] arr;
    Integer[][][] memo;

    public int minCount(int[] arr) {
        this.n = arr.length;
        this.arr = arr;
        memo = new Integer[n][n][n];

        return n - solve(0, 0, 0);
    }

    int solve(int i, int p, int q) {
        if (i == n) return 0;
        if (memo[i][p][q] != null) return memo[i][p][q];

        int best = solve(i + 1, p, q);

        if (p==0 || (arr[i] > arr[p - 1])) {
            best = Math.max(best, 1 + solve(i + 1, i + 1, q));
        }

        if (q==0 || (arr[i] < arr[q - 1])) {
            best = Math.max(best, 1 + solve(i + 1, p, i + 1));
        }

        return memo[i][p][q] = best;
    }
}
```

### Tabulation
```java
class Solution {
    public int minCount(int[] arr) {
        int n = arr.length;
        int[][][] dp = new int[n+1][n+1][n+1];
        for(int i=n;i>=0;i--) {
            for(int p=n;p>=0;p--) {
                for(int q=n;q>=0;q--) {
                    if (i == n) dp[i][p][q] = 0;
                    else {
                        int best = dp[i + 1][p][q];

                        if (p==0 || (arr[i] > arr[p - 1])) {
                            best = Math.max(best, 1 + dp[i + 1][i + 1][q]);
                        }
                
                        if (q==0 || (arr[i] < arr[q - 1])) {
                            best = Math.max(best, 1 + dp[i + 1][p][i + 1]);
                        }
                
                        dp[i][p][q] = best;
                    }
                }
            }
        }
        return n - dp[0][0][0];
    }
}
```

### Space Optimization
```java
class Solution {
    public int minCount(int[] arr) {
        int n = arr.length;
        int[][] prev = new int[n+1][n+1];
        for(int i=n;i>=0;i--) {
            int[][] curr = new int[n+1][n+1];
            for(int p=n;p>=0;p--) {
                for(int q=n;q>=0;q--) {
                    if (i == n) curr[p][q] = 0;
                    else {
                        int best = prev[p][q];

                        if (p==0 || (arr[i] > arr[p - 1])) {
                            best = Math.max(best, 1 + prev[i + 1][q]);
                        }
                
                        if (q==0 || (arr[i] < arr[q - 1])) {
                            best = Math.max(best, 1 + prev[p][i + 1]);
                        }
                
                        curr[p][q] = best;
                    }
                }
            }
            prev = curr;
        }
        return n - prev[0][0];
    }
}
```

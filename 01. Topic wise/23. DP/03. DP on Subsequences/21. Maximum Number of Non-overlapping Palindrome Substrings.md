### Memoization
```java
class Solution {
    int n, k;
    boolean[][] isPal;
    Integer[] memo;
    String s;

    public int maxPalindromes(String s, int k) {
        this.s = s;
        this.n = s.length();
        this.k = k;
        this.isPal = new boolean[n][n];
        this.memo = new Integer[n + 1];

        // Precompute palindrome table
        for (int i = n - 1; i >= 0; i--) {
            isPal[i][i] = true;
            for (int j = i + 1; j < n; j++) {
                if (s.charAt(i) == s.charAt(j) && (j - i == 1 || isPal[i + 1][j - 1])) {
                    isPal[i][j] = true;
                }
            }
        }

        return solve(0);
    }

    private int solve(int i) {
        if (i >= n) return 0;
        if (memo[i] != null) return memo[i];

        // Option 1: skip this starting index
        int best = solve(i + 1);

        // Option 2: try every valid palindromic substring starting at i
        for (int j = i + k - 1; j < n; j++) {
            if (isPal[i][j]) best = Math.max(best, 1 + solve(j + 1));
        }

        return memo[i] = best;
    }
}
```

### Tabulation
```java
class Solution {
    public int maxPalindromes(String s, int k) {
        int n = s.length();
        boolean[][] isPal = new boolean[n][n];
        int[] dp = new int[n + 1];
        
        for (int i = n - 1; i >= 0; i--) {
            isPal[i][i] = true;
            for (int j = i + 1; j < n; j++) {
                if (s.charAt(i) == s.charAt(j) && (j - i == 1 || isPal[i + 1][j - 1])) {
                    isPal[i][j] = true;
                }
            }
        }


        for(int i=n;i>=0;i--) {
            if(i==n) dp[i] = 0;
            else {
                int best = dp[i+1];
                for (int j = i + k - 1; j < n; j++) {
                    if (isPal[i][j]) best = Math.max(best, 1 + dp[j+1]);
                }
                dp[i] = best;
            }
        }

        return dp[0];
    }
}
```

class Solution {
    public int maxAmount(int[] arr, int k) {
        PriorityQueue<Integer> pq = new PriorityQueue<>((a, b) -> Integer.compare(arr[b], arr[a]));
        for (int i = 0; i < arr.length; i++) pq.offer(i);
        int ans = 0;
        while(k-->0 && !pq.isEmpty()) {
            int i = pq.poll();
            ans = (ans + arr[i])%1000000007;
            if(arr[i]>0) {
                arr[i] -= 1;
                pq.offer(i);
            }
        }
        return ans;
    }
}

class Solution {
    public String lexicographicallySmallest(String s, int k) {
        int n = s.length();
        if((n&(n-1))==0) k /= 2;
        else k *= 2;
        if(k==0) return s;
        if(k>=n) return "-1";
        StringBuilder ans = new StringBuilder();
        ans.append(s.charAt(0));
        for(int i=1;i<n;i++) {
            while(ans.length()>0 && s.charAt(i)<ans.charAt(ans.length()-1) && k>0) {
                k--;
                ans.deleteCharAt(ans.length()-1);
            }
            ans.append(s.charAt(i));
        }
        while(k>0 && ans.length()>0) {
            ans.deleteCharAt(ans.length()-1);
            k--;
        }
        return ans.length()>0 ? ans.toString() : "-1";
    }
}

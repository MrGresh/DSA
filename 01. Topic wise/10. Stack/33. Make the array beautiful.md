class Solution {
    List<Integer> makeBeautiful(int[] arr) {
        Stack<Integer> st = new Stack<>();
        for(int x: arr) {
            if(!st.isEmpty() && ((st.peek()>=0 && x<0) || (st.peek()<0 && x>=0))) st.pop();
            else st.push(x);
        }
        return new ArrayList<>(st);
    }
}

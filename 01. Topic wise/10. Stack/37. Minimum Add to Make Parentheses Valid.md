class Solution {
    public int minAddToMakeValid(String s) {
        int open = 0;  // unmatched '('
        int add = 0;   // unmatched ')' (each needs an inserted '(')

        for (char c : s.toCharArray()) {
            if (c == '(') {
                open++;
            } else {
                if(open > 0) open--;
                else add++;
            }
        }

        return add + open;
    }
}

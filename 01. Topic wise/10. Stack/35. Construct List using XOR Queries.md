class Solution {
    public ArrayList<Integer> constructList(int[][] queries) {
        ArrayList<Integer> ans = new ArrayList<>();
        int currXor = 0;
        for(int i=queries.length-1;i>=0;i--) {
            int[] val = queries[i];
            if(val[0]==1) currXor ^= val[1];
            else ans.add(currXor ^ val[1]);
        }
        ans.add(currXor);
        Collections.sort(ans);
        return ans;
    }
}

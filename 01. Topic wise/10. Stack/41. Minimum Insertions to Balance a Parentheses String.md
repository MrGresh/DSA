class Solution {
    public int minInsertions(String s) {
        int open = 0;   // unmatched '('
        int ins = 0;    // insertions made
        int n = s.length();

        for (int i=0;i<n;i++) {
            if (s.charAt(i) == '(') open++;
            else {
                if (i + 1 < n && s.charAt(i + 1) == ')') i++;
                else ins++; // )

                if (open > 0) open--;
                else ins++; // (
            }
        }
        return ins + 2 * open;
    }
}

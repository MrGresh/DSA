class Solution {
    public int minInsertions(String s) {
        int open = 0;   // unmatched '('
        int ins = 0;    // insertions made
        int i = 0, n = s.length();

        while (i < n) {
            if (s.charAt(i) == '(') {
                open++;
                i++;
            } else {
                // found a ')', check whether it's part of '))'
                if (i + 1 < n && s.charAt(i + 1) == ')') {
                    i += 2;
                } else {
                    ins++;      // insert the missing ')'
                    i++;
                }
                // this closing pair needs a '('
                if (open > 0) open--;
                else ins++;     // insert a '('
            }
        }
        return ins + 2 * open;
    }
}

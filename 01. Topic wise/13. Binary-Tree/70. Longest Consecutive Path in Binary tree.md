class Solution {
    int ans;
    public int longestConsecutive(Node root) {
        ans = 0;
        rec(root, -1, 0);
        return ans > 1 ? ans : -1;
    }
    void rec(Node root, int prev, int currLen) {
        if (root == null) return;

        // Check if consecutive
        if (prev != -1 && root.data == prev + 1) {
            currLen = currLen + 1;
        } else {
            currLen = 1; // Restart sequence with current node included
        }

        ans = Math.max(ans, currLen);

        rec(root.left, root.data, currLen);
        rec(root.right, root.data, currLen);
    }
}

class Solution {
    public int numberOfTurns(Node root, int p, int q) {
        StringBuilder pathP = new StringBuilder();
        StringBuilder pathQ = new StringBuilder();
        
        findPath(root, p, pathP);
        findPath(root, q, pathQ);
        
        String sp = pathP.toString();
        String sq = pathQ.toString();
        
        // find LCA divergence point
        int i = 0;
        while (i < sp.length() && i < sq.length() && sp.charAt(i) == sq.charAt(i)) {
            i++;
        }
        
        // part of p's path below LCA needs to be reversed (going up)
        // since going up retraces the same edges, direction stays same 
        // as it was on the way down (edge direction doesn't flip)
        String upPart = sp.substring(i);
        StringBuilder reversedUp = new StringBuilder(upPart).reverse();
        
        String downPart = sq.substring(i);
        
        String combined = reversedUp.toString() + downPart;
        
        if (combined.length() < 2) return -1;
        
        int turns = 0;
        for (int j = 1; j < combined.length(); j++) {
            if (combined.charAt(j) != combined.charAt(j - 1)) {
                turns++;
            }
        }
        
        return turns == 0 ? -1 : turns;
    }
    
    private boolean findPath(Node node, int target, StringBuilder path) {
        if (node == null) return false;
        if (node.data == target) return true;
        
        path.append('L');
        if (findPath(node.left, target, path)) return true;
        path.deleteCharAt(path.length() - 1);
        
        path.append('R');
        if (findPath(node.right, target, path)) return true;
        path.deleteCharAt(path.length() - 1);
        
        return false;
    }
}

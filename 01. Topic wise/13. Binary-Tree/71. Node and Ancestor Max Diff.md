class Solution {
    int ans;
    int maxDiff(Node root) {
        ans = Integer.MIN_VALUE;
        rec(root);
        return ans;
    }
    int rec(Node root) {
        if(root==null) return Integer.MAX_VALUE;
        int n1 = rec(root.left);
        int n2 = rec(root.right);
        if(n1!=Integer.MAX_VALUE || n2!=Integer.MAX_VALUE) ans = Math.max(ans, root.data-Math.min(n1, n2));
        return Math.min(root.data, Math.min(n1, n2));
    }
}

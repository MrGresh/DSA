class Solution {
    public void toSumTree(Node root) {
        rec(root);
    }
    int rec(Node root) {
        if(root==null) return 0;
        int lVal = rec(root.left);
        int rVal = rec(root.right);
        int val = root.data + lVal + rVal;
        root.data = lVal + rVal;
        return val;
    }
}

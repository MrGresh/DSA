### BFS
```java
class Solution {
    public int getCount(Node root, int k) {
        int level = 0, cnt = 0;
        Queue<Node> q = new LinkedList<>();
        q.add(root);
        while(!q.isEmpty()) {
            ++level;
            int size = q.size();
            while(size-->0) {
                Node n = q.poll();
                if(n.left==null && n.right==null) {
                    if(level<=k) {
                        k -= level;
                        cnt++;
                        if(k<level) return cnt++;
                    }
                } else {
                    if(n.left!=null) q.offer(n.left);
                    if(n.right!=null) q.offer(n.right);
                }
            }
        }
        return cnt;
    }
}
```

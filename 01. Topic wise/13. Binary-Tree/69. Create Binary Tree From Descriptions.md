class Solution {
    public TreeNode createBinaryTree(int[][] descriptions) {
        Map<Integer, TreeNode> map = new HashMap<>();
        Set<Integer> childs = new HashSet<>();
        for(int[] d: descriptions) {
            if(!map.containsKey(d[0])) map.put(d[0], new TreeNode(d[0]));
            if(!map.containsKey(d[1])) map.put(d[1], new TreeNode(d[1]));
            TreeNode n1 = map.get(d[0]), n2 = map.get(d[1]);
            if(d[2]==1) n1.left = n2;
            else n1.right = n2;
            childs.add(n2.val);
        }
        for(Integer node: map.keySet()) {
            if(!childs.contains(node)) return map.get(node);
        }
        return null;
    }
}

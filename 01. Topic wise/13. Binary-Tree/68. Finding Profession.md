### BFS (TLE)
```java
class Solution {
    public String profession(int level, int pos) {
        Queue<Character> q = new LinkedList<>();
        q.add('E');
        int currLevel = 0;
        while(!q.isEmpty()) {
            ++currLevel;
            int size = q.size();
            for(int i=1;i<=size;i++) {
                Character currProf = q.poll();
                if(i==pos && currLevel==level) {
                    if(currProf == 'E') return "Engineer";
                    else return "Doctor";
                }
                if(currProf == 'E') {
                    q.add('E');
                    q.add('D');
                } else {
                    q.add('D');
                    q.add('E');
                }
            }
        }
        return "Null";
    }
}
```
### DFS (TLE)
```java
class Solution {
    public String profession(int level, int pos) {
        char curr = 'E';
        
        long leftBoundary = 1;
        long rightBoundary = (long) Math.pow(2, level - 1);
        
        for (int i = 1; i < level; i++) {
            long mid = leftBoundary + (rightBoundary - leftBoundary) / 2;
            
            if (pos <= mid) rightBoundary = mid;
            else {
                curr = (curr == 'E') ? 'D' : 'E';
                leftBoundary = mid + 1;
            }
        }
        
        return curr == 'E' ? "Engineer" : "Doctor";
    }
}
```
### Recursion
```java
class Solution {
    public String profession(int level, int pos) {
        if (pos == 1) return "Engineer";
        
        String parentProfession = profession(level - 1, (pos + 1) / 2);
        
        if (pos % 2 != 0) return parentProfession;
        else return parentProfession.equals("Engineer") ? "Doctor" : "Engineer";
    }
}
```
### Bit Magic
```java
class Solution {
    public String profession(int level, int pos) {
        int count = Integer.bitCount(pos - 1);
        
        return (count % 2 != 0) ? "Doctor" : "Engineer";
    }
}
```

class Solution {
    public boolean areAnagrams(Node root1, Node root2) {
        if(root1.data!=root2.data) return false;
        Queue<Node> q1 = new LinkedList<>();
        q1.add(root1);
        Queue<Node> q2 = new LinkedList<>();
        q2.add(root2);
        while(!q1.isEmpty()) {
            int size1 = q1.size();
            int size2 = q2.size();
            Map<Integer, Integer> map = new HashMap<>();
            for(int i=1;i<=size1;i++) {
                Node n = q1.poll();
                if(n.left!=null) {
                    q1.add(n.left);
                    map.put(n.left.data, map.getOrDefault(n.left.data, 0)+1);
                }
                if(n.right!=null) {
                    q1.add(n.right);
                    map.put(n.right.data, map.getOrDefault(n.right.data, 0)+1);
                }
            }
            for(int i=1;i<=size2;i++) {
                Node n = q2.poll();
                if(n.left!=null) {
                    if(!map.containsKey(n.left.data)) return false;
                    if(map.get(n.left.data)==1) map.remove(n.left.data);
                    else map.put(n.left.data, map.get(n.left.data)-1);
                    q2.add(n.left);
                }
                if(n.right!=null) {
                    if(!map.containsKey(n.right.data)) return false;
                    if(map.get(n.right.data)==1) map.remove(n.right.data);
                    else map.put(n.right.data, map.get(n.right.data)-1);
                    q2.add(n.right);
                }
            }
        }
        return true;
    }
}

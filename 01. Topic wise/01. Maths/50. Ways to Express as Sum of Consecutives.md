class Solution {
    public int getCount(int n) {
        int count = 0;
        
        // Multiply by 2 as per the derived formula
        long target = 2 * (long) n; 
        
        // k is the number of consecutive terms. It must be >= 2.
        // The loop runs while k * k < target
        for (int k = 2; (long) k * k < target; k++) {
            if (target % k == 0) {
                long remainder = (target / k) - k + 1;
                // Check if the numerator for 'a' is positive and even
                if (remainder > 0 && remainder % 2 == 0) {
                    count++;
                }
            }
        }
        
        return count;
    }
}

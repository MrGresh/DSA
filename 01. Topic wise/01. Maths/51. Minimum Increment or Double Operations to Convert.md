class Solution {
    public int countMinOperations(int arr[]) {
        int n = arr.length;
        long ans = 0;
        
        while (true) {
            boolean allZero = true;
            int oddCount = 0;
            
            for (int i = 0; i < n; i++) {
                if (arr[i] != 0) allZero = false;
                if ((arr[i] & 1) == 1) oddCount++;
            }
            
            if (allZero) break;
            
            ans += oddCount;
            for (int i = 0; i < n; i++) {
                if ((arr[i] & 1) == 1) arr[i]--;
            }
            
            allZero = true;
            for (int i = 0; i < n; i++) {
                if (arr[i] != 0) {
                    allZero = false;
                    break;
                }
            }
            if (allZero) break;
            
            for (int i = 0; i < n; i++) arr[i] >>= 1;
            ans++;
        }
        
        return (int) ans;
    }
}

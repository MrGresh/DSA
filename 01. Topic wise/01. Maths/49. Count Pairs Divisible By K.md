class Solution {
    public int countKdivPairs(int[] arr, int k) {
        // Step 1: Create a frequency map for remainders
        int[] rem = new int[k];
        for (int x : arr) {
            rem[x % k]++;
        }
        
        int cnt = 0;
        
        // Step 2: Handle elements perfectly divisible by k (remainder 0)
        // Ways to pick 2 elements from rem[0] = n * (n - 1) / 2
        cnt += (rem[0] * (rem[0] - 1)) / 2;
        
        // Step 3: Pair remainders i and k - i
        // Go up to k/2. We handle the exact midpoint separately if k is even.
        for (int i = 1; i < (k + 1) / 2; i++) {
            cnt += rem[i] * rem[k - i];
        }
        
        // Step 4: If k is even, handle the exact midpoint (remainder k/2)
        if (k % 2 == 0) {
            cnt += (rem[k / 2] * (rem[k / 2] - 1)) / 2;
        }
        
        return cnt;
    }
}

class Solution {
    int ans, prev;
    public int absDiff(Node root) {
        ans = Integer.MAX_VALUE;
        prev = -1;
        rec(root);
        return ans;
    }
    void rec(Node root) {
        if(root==null) return;
        rec(root.left);
        if(prev!=-1) ans = Math.min(ans, Math.abs(root.data-prev));
        prev = root.data;
        rec(root.right);
    }
}

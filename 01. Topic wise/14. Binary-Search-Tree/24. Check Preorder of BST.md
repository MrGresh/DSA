class Solution {
    int currentIndex;
    int cnt;
    public boolean canRepresentBST(List<Integer> arr) {
        currentIndex = 0;
        cnt = 0;
        traversePreorderBST(arr, Integer.MIN_VALUE, Integer.MAX_VALUE);
        return (cnt==arr.size());
    }
    void traversePreorderBST(List<Integer>  pre, int min, int max) {
        if (currentIndex > pre.size() - 1) return;
        int val = pre.get(currentIndex);
        if (val < min || val > max) return;
        cnt++;
        currentIndex++;
        traversePreorderBST(pre, min, val);
        traversePreorderBST(pre, val, max);
    }
}

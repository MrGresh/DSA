class Solution {
    public boolean wifiRange(String s, int x) {
        boolean[] isCovered = new boolean[s.length()];
        for(int i=0;i<s.length();i++) {
            if(s.charAt(i)=='1') {
                isCovered[i] = true;
                for(int j=i+1;j<=i+x;j++) {
                    if(j==s.length() || s.charAt(j)=='1') break;
                    isCovered[j] = true;
                }
            }
        }
        for(int i=s.length()-1;i>=0;i--) {
            if(s.charAt(i)=='1') {
                isCovered[i] = true;
                for(int j=i-1;j>=i-x;j--) {
                    if(j==-1 ||  s.charAt(j)=='1') break;
                    isCovered[j] = true;
                }
            }
        }
        for(boolean r: isCovered) {
            if(!r) return false;
        }
        return true;
    }
}

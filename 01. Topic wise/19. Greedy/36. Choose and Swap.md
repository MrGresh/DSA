class Solution {
    public String chooseSwap(String s) {
        int n = s.length();
        
        // Store the first occurrence of each character 'a' through 'z'
        int[] firstOccur = new int[26];
        for (int i = 0; i < 26; i++) {
            firstOccur[i] = -1;
        }
        
        for (int i = 0; i < n; i++) {
            int charIdx = s.charAt(i) - 'a';
            if (firstOccur[charIdx] == -1) {
                firstOccur[charIdx] = i;
            }
        }
        
        char charToReplace = ' ';
        char replacementChar = ' ';
        boolean found = false;
        
        // Scan the string from left to right
        for (int i = 0; i < n; i++) {
            char curr = s.charAt(i);
            
            // Look for the smallest available character that appears after index i
            for (char c = 'a'; c < curr; c++) {
                if (firstOccur[c - 'a'] > i) {
                    charToReplace = curr;
                    replacementChar = c;
                    found = true;
                    break;
                }
            }
            if (found) {
                break;
            }
        }
        
        // If no advantageous swap was found, return the original string
        if (!found) {
            return s;
        }
        
        // Rebuild the string with all occurrences of the two characters swapped
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            char curr = s.charAt(i);
            if (curr == charToReplace) {
                sb.append(replacementChar);
            } else if (curr == replacementChar) {
                sb.append(charToReplace);
            } else {
                sb.append(curr);
            }
        }
        
        return sb.toString();
    }
}

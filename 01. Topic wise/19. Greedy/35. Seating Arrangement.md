class Solution {
    public boolean canSeatAllPeople(int k, int[] seats) {
        if(k==0) return true;
        for(int i=0;i<seats.length;i++) {
            if(seats[i]==0) {
                boolean isLeftEmpty = i-1>=0 ? seats[i-1]==0 : true;
                boolean isRightEmpty = i+1<seats.length ? seats[i+1]==0 : true;
                if(isLeftEmpty && isRightEmpty) {
                    k--;
                    seats[i] = 1;
                }
            }
            if(k==0) return true;
        }
        return false;
    }
}

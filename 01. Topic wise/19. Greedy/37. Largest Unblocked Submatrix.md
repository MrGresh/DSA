### DFS
```java
class Solution {
    Set<Integer> blockedRows;
    Set<Integer> blockedCols;
    boolean[][] vis;
    public int largestArea(int n, int m, int k, int[][] arr) {
        blockedRows = new HashSet<>();
        blockedCols = new HashSet<>();
        for(int[] e: arr) {
            blockedRows.add(e[0]-1);
            blockedCols.add(e[1]-1);
        }
        vis = new boolean[n][m];
        int ans = 0;
        for(int r=0;r<n;r++) {
            for(int c=0;c<m;c++) {
                if(!blockedRows.contains(r) && !blockedCols.contains(c) && !vis[r][c]) {
                    ans = Math.max(ans, rec(n, m, r, c));
                }
            }
        }
        return ans;
    }
    int rec(int n, int m, int r, int c) {
        vis[r][c] = true;
        int ans = 1;
        if(r+1<n && !blockedRows.contains(r+1) && !vis[r+1][c]) ans += rec(n, m, r+1, c);
        if(c+1<m && !blockedCols.contains(c+1) && !vis[r][c+1]) ans += rec(n, m, r, c+1);
        if(r-1>=0 && !blockedRows.contains(r-1) && !vis[r-1][c]) ans += rec(n, m, r-1, c);
        if(c-1>=0 && !blockedCols.contains(c-1) && !vis[r][c-1]) ans += rec(n, m, r, c-1);
        return ans;
    }
}
```

### Greedy
```java
class Solution {
    public int largestArea(int n, int m, int k, int[][] arr) {
        List<Integer> rows = new ArrayList<>();
        List<Integer> cols = new ArrayList<>();
        
        // Add virtual boundaries
        rows.add(0);
        rows.add(n + 1);
        cols.add(0);
        cols.add(m + 1);
        
        // Add the blocked rows and columns
        for (int[] cell : arr) {
            rows.add(cell[0]);
            cols.add(cell[1]);
        }
        
        // Sort to find consecutive gaps
        Collections.sort(rows);
        Collections.sort(cols);
        
        // Find the maximum row gap
        int maxRowGap = 0;
        for (int i = 1; i < rows.size(); i++) {
            int gap = rows.get(i) - rows.get(i - 1) - 1;
            maxRowGap = Math.max(maxRowGap, gap);
        }
        
        // Find the maximum column gap
        int maxColGap = 0;
        for (int i = 1; i < cols.size(); i++) {
            int gap = cols.get(i) - cols.get(i - 1) - 1;
            maxColGap = Math.max(maxColGap, gap);
        }
        
        // The largest continuous area is the product of the maximum gaps
        return maxRowGap * maxColGap;
    }
}
```

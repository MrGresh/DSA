### Kadane Algo
```java
class Solution {
    int maxSubstring(String s) {
        int max_sum = Integer.MIN_VALUE;
        int current_sum = 0;
        
        for (int i = 0; i < s.length(); i++) {
            int val = (s.charAt(i) == '0') ? 1 : -1;
            
            current_sum = Math.max(val, current_sum + val);
            
            max_sum = Math.max(max_sum, current_sum);
        }
        
        return max_sum;
    }
}
```

class Solution {
    int transform(String s1, String s2) {
        int n = s1.length();
        if (n != s2.length()) return -1;

        // Check if s1 and s2 are anagrams
        int[] count = new int[256];
        for (int i = 0; i < n; i++) {
            count[s1.charAt(i)]++;
            count[s2.charAt(i)]--;
        }
        for (int c : count) {
            if (c != 0) return -1;
        }

        // Greedily match s2 from the end against s1 from the end
        int j = n - 1;
        int matched = 0;

        for (int i = n - 1; i >= 0; i--) {
            while (j >= 0 && s1.charAt(j) != s2.charAt(i)) {
                j--;
            }
            if (j < 0) break;
            matched++;
            j--;
        }

        return n - matched;
    }
}

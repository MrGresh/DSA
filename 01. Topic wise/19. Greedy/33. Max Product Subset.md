class Solution {
    public int findMaxProduct(int[] arr) {
        int n = arr.length;
        if (n == 1) return arr[0];

        long MOD = 1000000007;
        int maxNegative = Integer.MIN_VALUE;
        int countNeg = 0;
        int countZero = 0;
        long product = 1;
        boolean taken = false; // Tracks if we actually included any element in the product

        for (int i = 0; i < n; i++) {
            if (arr[i] == 0) {
                countZero++;
                continue;
            }

            if (arr[i] < 0) {
                countNeg++;
                maxNegative = Math.max(maxNegative, arr[i]);
            }

            // Multiply positive numbers and all negative numbers for now
            product = (product * arr[i]) % MOD;
            taken = true;
        }

        // Edge Case 1: All elements were zeros
        if (countZero == n) {
            return 0;
        }

        // Edge Case 2: There is an odd number of negatives
        if (countNeg % 2 != 0) {
            if (countNeg == 1 && countZero + countNeg == n) {
                return 0;
            }
            
            product = 1;
            boolean skipped = false;
            for (int i = 0; i < n; i++) {
                if (arr[i] == 0) continue;
                if (arr[i] == maxNegative && !skipped) {
                    skipped = true; // Skip exactly one instance of the maxNegative
                    continue;
                }
                product = (product * arr[i]) % MOD;
            }
        }

        return (int) ((product + MOD) % MOD);
    }
}

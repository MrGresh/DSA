class Solution {
    public int sumDiffPairs(int[] arr, int k) {
        Arrays.sort(arr);
        int ans = 0;
        for(int i=arr.length-1;i>=1;) {
            if(arr[i]-arr[i-1]<k) {
                ans += (arr[i]+arr[i-1]);
                i -= 2;
            } else i--;
        }
        return ans;
    }
}

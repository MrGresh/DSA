class Solution {
    public int countTriplets(int[] arr, int l, int r) {
        Arrays.sort(arr);
        int n = arr.length;
        int cnt = 0;
        for(int i=0;i<n-2;i++) {
            for(int j=i+1;j<n-1;j++) {
                for(int k=j+1;k<n;k++) {
                    int sum = arr[i]+arr[j]+arr[k];
                    if(sum>=l && sum<=r) cnt++;
                    if(sum>r) break;
                }
            }
        }
        return cnt;
    }
}

class Solution {
    public int countTriplets(int[] arr, int l, int r) {
        Arrays.sort(arr);
        return (int)(countLE(arr, r) - countLE(arr, l - 1));
    }
    
    private long countLE(int[] arr, long x) {
        int n = arr.length;
        long count = 0;
        for (int i = 0; i < n - 2; i++) {
            int j = i + 1, k = n - 1;
            while (j < k) {
                if (arr[i] + arr[j] + arr[k] <= x) {
                    // arr[i]+arr[j]+arr[k'] <= x for all k' in [j+1, k]
                    count += (k - j);
                    j++;
                } else {
                    k--;
                }
            }
        }
        return count;
    }
}

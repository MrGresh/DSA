### Brute Force: Try All Possible Ways
```java
class Solution {
    public int maxProfit(int x, int y, int[] a, int[] b) {
        return rec(x, y, a, b, 0);
    }
    int rec(int x, int y, int[] a, int[] b, int i) {
        if(i==a.length) return 0;
        int ans = Integer.MIN_VALUE;
        if(x>0) ans = a[i]+rec(x-1, y, a, b, i+1);
        if(y>0) ans = Math.max(ans, b[i]+rec(x, y-1, a, b, i+1));
        return ans;
    }
}
```

### Sorting
```java
class Solution {
    public int maxProfit(int x, int y, int[] a, int[] b) {
        int n = a.length;
        // Store {abs_diff, index}
        int[][] diff = new int[n][2];
        
        for (int i = 0; i < n; i++) {
            diff[i][0] = Math.abs(a[i] - b[i]);
            diff[i][1] = i;
        }

        // Sort by difference in descending order
        Arrays.sort(diff, (p1, p2) -> Integer.compare(p2[0], p1[0]));

        int totalProfit = 0;
        for (int i = 0; i < n; i++) {
            int idx = diff[i][1];
            
            // If Machine A is more profitable and has space
            if (a[idx] >= b[idx]) {
                if (x > 0) {
                    totalProfit += a[idx];
                    x--;
                } else {
                    // Machine A full, must use B
                    totalProfit += b[idx];
                    y--;
                }
            } 
            // If Machine B is more profitable and has space
            else {
                if (y > 0) {
                    totalProfit += b[idx];
                    y--;
                } else {
                    // Machine B full, must use A
                    totalProfit += a[idx];
                    x--;
                }
            }
        }
        
        return totalProfit;
    }
}
```

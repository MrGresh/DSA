class Solution {
    public int maxArea(int[][] mat) {
        int n = mat.length, m = mat[0].length;
        int[] heights = new int[m];
        int maxArea = 0;

        for (int i = 0; i < n; i++) {
            // update running column heights
            for (int j = 0; j < m; j++) {
                heights[j] = (mat[i][j] == 0) ? 0 : heights[j] + 1;
            }

            // since columns can be freely rearranged, sort this row's heights
            int[] sorted = heights.clone();
            Arrays.sort(sorted); // ascending

            // consider widths from the right (largest heights first)
            for (int k = 0; k < m; k++) {
                int width = m - k;           // number of columns with height >= sorted[k]
                int area = sorted[k] * width;
                maxArea = Math.max(maxArea, area);
            }
        }

        return maxArea;
    }
}

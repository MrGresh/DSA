### BFS
```java
class Solution {
    public int minThrows(int n, int[] lad, int[] san) {
        Map<Integer, Integer> ld = new HashMap<>();
        Map<Integer, Integer> sn = new HashMap<>();
        for(int i=0;i<san.length-1;i+=2) {
            if(san[i]==n*n) return -1;
            sn.put(san[i], san[i+1]);
        }
        for(int i=0;i<lad.length-1;i+=2) {
            ld.put(lad[i], lad[i+1]);
        }
        Queue<int[]> q = new LinkedList<>();
        q.add(new int[]{1, 0});
        boolean[] vis = new boolean[n*n+1];
        vis[1] = true;
        while(!q.isEmpty()) {
            int[] data = q.poll();
            if(data[0]==n*n) return data[1];
            int i = data[0];
            for(int j=1;j<=6;j++) {
                if(ld.containsKey(i+j)) {
                    if(!vis[ld.get(i+j)]) {
                        q.add(new int[]{ ld.get(i+j), data[1]+1 });
                        vis[ld.get(i+j)] = true;
                    }
                }
                else if(sn.containsKey(i+j)) {
                    if(!vis[sn.get(i+j)]) {
                        q.add(new int[]{ sn.get(i+j), data[1]+1 });
                        vis[sn.get(i+j)] = true;
                    }
                } else {
                    if(i+j<=n*n && !vis[i+j]) {
                        q.add(new int[]{ i+j, data[1]+1 });
                        vis[i+j] = true;
                    }
                }
            }
        }
        return -1;
    }
}
```

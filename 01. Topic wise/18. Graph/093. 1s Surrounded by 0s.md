### DFS
```java
class Solution {
    int cntOnes(int[][] grid) {
        for(int i=0;i<grid.length;i++) {
            if(grid[i][0]==1) dfs(grid, i, 0);
            if(grid[i][grid[0].length-1]==1) dfs(grid, i, grid[0].length-1);
        }
        for(int j=0;j<grid[0].length;j++) {
            if(grid[0][j]==1) dfs(grid, 0, j);
            if(grid[grid.length-1][j]==1) dfs(grid, grid.length-1, j);
        }
        int ans = 0;
        for(int i=0;i<grid.length;i++) {
            for(int j=0;j<grid[0].length;j++) {
                if(grid[i][j]==1) ans++;
            }
        }
        return ans;
    }
    int dfs(int[][] grid, int i, int j) {
        grid[i][j] = 0;
        int ans = 1;
        if(i-1>=0 && grid[i-1][j]==1) ans += dfs(grid, i-1, j);
        if(i+1<grid.length && grid[i+1][j]==1) ans += dfs(grid, i+1, j);
        if(j-1>=0 && grid[i][j-1]==1) ans += dfs(grid, i, j-1);
        if(j+1<grid[0].length && grid[i][j+1]==1) ans += dfs(grid, i, j+1);
        return ans;
    }
};
```

### Backtracking
```java
class Solution {
    // Global variable to track the minimum score found in the component
    private int minScore = Integer.MAX_VALUE;

    public int minScore(int n, int[][] roads) {
        // Build the adjacency list graph
        Map<Integer, List<int[]>> graph = new HashMap<>();
        for (int[] road : roads) {
            graph.computeIfAbsent(road[0], k -> new ArrayList<>()).add(new int[] {road[1], road[2]});
            graph.computeIfAbsent(road[1], k -> new ArrayList<>()).add(new int[] {road[0], road[2]});
        }

        // Visited array to prevent infinite cycles
        boolean[] visited = new boolean[n + 1];

        // Start the traversal from city 1
        dfs(1, graph, visited);

        return minScore;
    }

    private void dfs(int node, Map<Integer, List<int[]>> graph, boolean[] visited) {
        // Mark the current city as visited
        visited[node] = true;

        if (graph.containsKey(node)) {
            for (int[] neighbor : graph.get(node)) {
                int nextNode = neighbor[0];
                int distance = neighbor[1];

                // Update the minimum score with every road we encounter
                minScore = Math.min(minScore, distance);

                // Recurse only if the neighbor city hasn't been visited yet
                if (!visited[nextNode]) {
                    dfs(nextNode, graph, visited);
                }
            }
        }
    }
}
```

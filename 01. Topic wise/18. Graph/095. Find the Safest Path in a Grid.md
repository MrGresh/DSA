### BFS + DFS
```java
class Solution {
    int[][] thiefDist;
    public int maximumSafenessFactor(List<List<Integer>> grid) {
        int n = grid.size();
        thiefDist = new int[n][n];
        boolean[][] vis = new boolean[n][n];
        Queue<int[]> q = new LinkedList<>();
        for(int i=0;i<n;i++) {
            for(int j=0;j<n;j++) {
                if(grid.get(i).get(j)==1) {
                    vis[i][j] = true;
                    q.add(new int[] {i, j});
                }
            }
        }
        int dist = -1;
        while(!q.isEmpty()) {
            int size = q.size();
            ++dist;
            while(size-->0) {
                int[] e = q.poll();
                int i = e[0], j=e[1];
                thiefDist[i][j]=dist;
                if(i-1>=0 && !vis[i-1][j]) {
                    vis[i-1][j] = true;
                    q.add(new int[] {i-1, j});
                }
                if(j-1>=0 && !vis[i][j-1]) {
                    vis[i][j-1] = true;
                    q.add(new int[] {i, j-1});
                }
                if(i+1<n && !vis[i+1][j]) {
                    vis[i+1][j] = true;
                    q.add(new int[] {i+1, j});
                }
                if(j+1<n && !vis[i][j+1]) {
                    vis[i][j+1] = true;
                    q.add(new int[] {i, j+1});
                }
            }
        }
        vis = new boolean[n][n];
        return rec(grid, n, vis, 0, 0);
    }
    int rec(List<List<Integer>> grid, int n, boolean[][] vis, int i, int j) {
        if(i==n-1 && j==n-1) return thiefDist[i][j];
        vis[i][j] = true;
        int ans = Integer.MIN_VALUE;
        if(i-1>=0 && !vis[i-1][j]) {
            ans = Math.min(thiefDist[i][j], rec(grid, n, vis, i-1, j));
        }
        if(j-1>=0 && !vis[i][j-1]) {
            ans = Math.max(ans, Math.min(thiefDist[i][j], rec(grid, n, vis, i, j-1)));
        }
        if(i+1<n && !vis[i+1][j]) {
            ans = Math.max(ans, Math.min(thiefDist[i][j], rec(grid, n, vis, i+1, j)));
        }
        if(j+1<n && !vis[i][j+1]) {
            ans = Math.max(ans, Math.min(thiefDist[i][j], rec(grid, n, vis, i, j+1)));
        }
        vis[i][j] = false;
        return ans;
    }
}
```

### BFS + Max-Heap (Dijkstra-like approach)
```java
class Solution {
    int[][] thiefDist;
    public int maximumSafenessFactor(List<List<Integer>> grid) {
        int n = grid.size();
        thiefDist = new int[n][n];
        boolean[][] vis = new boolean[n][n];
        Queue<int[]> q = new LinkedList<>();
        for(int i=0;i<n;i++) {
            for(int j=0;j<n;j++) {
                if(grid.get(i).get(j)==1) {
                    vis[i][j] = true;
                    q.add(new int[] {i, j});
                }
            }
        }
        int dist = -1;
        while(!q.isEmpty()) {
            int size = q.size();
            ++dist;
            while(size-->0) {
                int[] e = q.poll();
                int i = e[0], j=e[1];
                thiefDist[i][j]=dist;
                if(i-1>=0 && !vis[i-1][j]) {
                    vis[i-1][j] = true;
                    q.add(new int[] {i-1, j});
                }
                if(j-1>=0 && !vis[i][j-1]) {
                    vis[i][j-1] = true;
                    q.add(new int[] {i, j-1});
                }
                if(i+1<n && !vis[i+1][j]) {
                    vis[i+1][j] = true;
                    q.add(new int[] {i+1, j});
                }
                if(j+1<n && !vis[i][j+1]) {
                    vis[i][j+1] = true;
                    q.add(new int[] {i, j+1});
                }
            }
        }
        
        vis = new boolean[n][n];
        PriorityQueue<int[]> maxHeap = new PriorityQueue<>((a, b) -> b[0] - a[0]);
        int[] dr = {-1, 1, 0, 0};
        int[] dc = {0, 0, -1, 1};
        maxHeap.add(new int[] {thiefDist[0][0], 0, 0});
        vis[0][0] = true;
        
        while (!maxHeap.isEmpty()) {
            int[] curr = maxHeap.poll();
            int safe = curr[0];
            int r = curr[1];
            int c = curr[2];
            
            if (r == n - 1 && c == n - 1) return safe;
            
            for (int i = 0; i < 4; i++) {
                int nr = r + dr[i];
                int nc = c + dc[i];
                
                if (nr >= 0 && nr < n && nc >= 0 && nc < n && !vis[nr][nc]) {
                    vis[nr][nc] = true;
                    maxHeap.add(new int[] {Math.min(safe, thiefDist[nr][nc]), nr, nc});
                }
            }
        }
        
        return 0;
    }
}
```

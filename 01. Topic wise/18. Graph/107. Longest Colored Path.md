class Solution {
    int ans;
    public int longestPath(String s, int[][] edges) {
        int n = s.length();

        List<List<Integer> > adj = new ArrayList<>();

        for (int i = 0; i < n; i++)
            adj.add(new ArrayList<>());

        for (int[] e : edges) {
            int u = e[0] - 1;
            int v = e[1] - 1;

            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        ans = 1;
        for (int i = 0; i < n; i++) {
            boolean blueStarted = (s.charAt(i) == 'B');

            dfs(adj, s, i, -1, blueStarted, 1);
        }

        return ans;
    }
    void dfs(List<List<Integer> > adj, String s, int node, int par, boolean blueStarted, int len)
    {
        ans = Math.max(ans, len);

        for (int it : adj.get(node)) {
            if (it == par) continue;

            if (blueStarted && s.charAt(it) == 'R') continue;

            boolean nextBlueStarted = blueStarted || (s.charAt(it) == 'B');

            dfs(adj, s, it, node, nextBlueStarted, len + 1);
        }
    }
}

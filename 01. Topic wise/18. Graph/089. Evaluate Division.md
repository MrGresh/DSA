class Solution {
    public double[] calcEquation(List<List<String>> equations, double[] values, List<List<String>> queries) {
        // Map each variable to an index
        Map<String, Integer> id = new HashMap<>();
        for (List<String> eq : equations) {
            for (String s : eq) {
                if (!id.containsKey(s)) id.put(s, id.size());
            }
        }

        int n = id.size();
        double[][] d = new double[n][n];
        for (int i=0;i<d.length;i++) {
            Arrays.fill(d[i], -1.0);
            d[i][i] = 1.0;
        }

        // Direct edges
        for (int i = 0; i < equations.size(); i++) {
            int a = id.get(equations.get(i).get(0));
            int b = id.get(equations.get(i).get(1));
            d[a][b] = values[i];
            d[b][a] = 1.0 / values[i];
        }

        // Floyd-Warshall (multiplicative)
        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (d[i][k]!=-1 && d[k][j]!=-1) {
                        d[i][j] = Math.max(d[i][j], d[i][k] * d[k][j]);
                    }
                }
            }
        }

        // Answer queries
        double[] res = new double[queries.size()];
        for (int i = 0; i < queries.size(); i++) {
            String c = queries.get(i).get(0);
            String e = queries.get(i).get(1);
            if (!id.containsKey(c) || !id.containsKey(e)) {
                res[i] = -1.0;
            } else {
                res[i] = d[id.get(c)][id.get(e)];
            }
        }
        return res;
    }
}

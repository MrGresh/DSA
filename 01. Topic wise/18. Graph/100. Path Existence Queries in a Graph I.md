### Binary Search + DSU
```java
class Solution {
    public boolean[] pathExistenceQueries(int n, int[] nums, int maxDiff, int[][] queries) {
        int[] parent = new int[n];
        for(int i=0;i<n;i++) parent[i] = i;
        for(int i=0;i<n;i++) {
            int leftMost = left(nums, maxDiff, i);
            int rightMost = right(nums, maxDiff, i);
            for(int j=leftMost;j<=rightMost;j++) {
                parent[j] = parent[i];
            }
        }
        boolean[] ans = new boolean[queries.length];
        for(int i=0;i<queries.length;i++) {
            if(parent[queries[i][0]]==parent[queries[i][1]]) ans[i] = true;
        }
        return ans;
    }
    int left(int[] nums, int maxDiff, int i) {
        int l=0, r=i;
        int ans = i;
        while(l<=r) {
            int m=l+(r-l)/2;
            if(Math.abs(nums[m]-nums[i])<=maxDiff) {
                ans = m;
                r = m-1;
            } else l = m+1;
        }
        return ans;
    }
    int right(int[] nums, int maxDiff, int i) {
        int l=i, r=nums.length-1;
        int ans = i;
        while(l<=r) {
            int m=l+(r-l)/2;
            if(Math.abs(nums[m]-nums[i])<=maxDiff) {
                ans = m;
                l = m+1;
            } else r = m-1;
        }
        return ans;
    }
}
```

### Math + DSU
```java
class Solution {
    public boolean[] pathExistenceQueries(int n, int[] nums, int maxDiff, int[][] queries) {
        int[] parent = new int[n];
        int parentId = 0;
        parent[0] = parentId;
        
        for (int i = 1; i < n; i++) {
            if (nums[i] - nums[i - 1] > maxDiff) parentId++;
            parent[i] = parentId;
        }
        
        boolean[] ans = new boolean[queries.length];
        for (int i = 0; i < queries.length; i++) {
            ans[i] = (parent[queries[i][0]] == parent[queries[i][1]]);
        }
        
        return ans;
    }
}
```

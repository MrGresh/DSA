### Dijkstra
```java
class Solution {
    static class Edge {
        int to;
        int weight;

        Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    static class Node implements Comparable<Node> {
        int id;
        int dist;

        Node(int id, int dist) {
            this.id = id;
            this.dist = dist;
        }

        @Override
        public int compareTo(Node other) {
            return Integer.compare(this.dist, other.dist);
        }
    }

    public int minimumEdgeReversal(int[][] edges, int n, int src, int dst) {
        // Step 1: Adjacency list banao
        List<List<Edge>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }

        for (int[] edge : edges) {
            int u = edge[0];
            int v = edge[1];
            
            // Original edge -> cost 0
            adj.get(u).add(new Edge(v, 0));
            // Reversed edge -> cost 1
            adj.get(v).add(new Edge(u, 1));
        }

        // Step 2: Distance array & PriorityQueue setup
        int[] dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);

        PriorityQueue<Node> pq = new PriorityQueue<>();
        
        dist[src] = 0;
        pq.add(new Node(src, 0));

        // Step 3: Dijkstra Process
        while (!pq.isEmpty()) {
            Node current = pq.poll();
            int u = current.id;
            int d = current.dist;

            // Target mil gaya toh direct return karo
            if (u == dst) {
                return d;
            }

            // Agar out-of-date distance wala entry hai toh skip karo
            if (d > dist[u]) {
                continue;
            }

            for (Edge neighbor : adj.get(u)) {
                int v = neighbor.to;
                int weight = neighbor.weight;

                if (dist[u] + weight < dist[v]) {
                    dist[v] = dist[u] + weight;
                    pq.add(new Node(v, dist[v]));
                }
            }
        }

        return dist[dst] == Integer.MAX_VALUE ? -1 : dist[dst];
    }
}
```

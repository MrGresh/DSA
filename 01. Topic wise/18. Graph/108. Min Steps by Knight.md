### BFS
```java
class Solution {
    // Coordinate offsets for all 8 possible knight moves
    private static final int[] dx = {-2, -2, -1, -1, 1, 1, 2, 2};
    private static final int[] dy = {-1, 1, -2, 2, -2, 2, -1, 1};

    public int minStepToReachTarget(int knightPos[], int targetPos[], int n) {
        int startX = knightPos[0];
        int startY = knightPos[1];
        int targetX = targetPos[0];
        int targetY = targetPos[1];

        // Base case: if start position is already the target
        if (startX == targetX && startY == targetY) {
            return 0;
        }

        // Visited array using 1-based indexing
        boolean[][] visited = new boolean[n + 1][n + 1];
        
        // Queue stores array of {x, y, dist}
        Queue<int[]> queue = new ArrayDeque<>();
        
        // Push initial position
        queue.offer(new int[]{startX, startY, 0});
        visited[startX][startY] = true;

        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            int x = current[0];
            int y = current[1];
            int dist = current[2];

            // Try all 8 possible moves
            for (int i = 0; i < 8; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];

                // Check bounds and whether cell is visited
                if (nx >= 1 && nx <= n && ny >= 1 && ny <= n && !visited[nx][ny]) {
                    // Check if target is reached
                    if (nx == targetX && ny == targetY) {
                        return dist + 1;
                    }

                    visited[nx][ny] = true;
                    queue.offer(new int[]{nx, ny, dist + 1});
                }
            }
        }

        return -1; // If target is unreachable
    }
}
```

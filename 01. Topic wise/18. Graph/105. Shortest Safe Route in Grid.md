### BFS
```java
class Solution {
    int shortestPath(int[][] mat) {
        Queue<int[]> q = new LinkedList<>();
        boolean[][] vis = new boolean[mat.length][mat[0].length];
        for(int i=0;i<mat.length;i++) {
            if(isSafe(mat, i, 0)) {
                q.add(new int[] {i, 0, 1});
                vis[i][0] = true;
            }
        }
        while(!q.isEmpty()) {
            int[] e = q.poll();
            int i = e[0], j = e[1], dist = e[2];
            if(j==mat[0].length-1) return dist;
            if(i+1<mat.length && !vis[i+1][j] && isSafe(mat, i+1, j)) {
                q.add(new int[] {i+1, j, dist+1});
                vis[i+1][j] = true;
            }
            if(j+1<mat[0].length && !vis[i][j+1] && isSafe(mat, i, j+1)) {
                q.add(new int[] {i, j+1, dist+1});
                vis[i][j+1] = true;
            }
            if(i-1>=0 && !vis[i-1][j] && isSafe(mat, i-1, j)) {
                q.add(new int[] {i-1, j, dist+1});
                vis[i-1][j] = true;
            }
            if(j-1>=0 && !vis[i][j-1] && isSafe(mat, i, j-1)) {
                q.add(new int[] {i, j-1, dist+1});
                vis[i][j-1] = true;
            }
        }
        return -1;
    }
    boolean isSafe(int[][] mat, int i, int j) {
        if(mat[i][j]==0) return false;
        if(i-1>=0 && mat[i-1][j]==0) return false;
        if(j-1>=0 && mat[i][j-1]==0) return false;
        if(i+1<mat.length && mat[i+1][j]==0) return false;
        if(j+1<mat[0].length && mat[i][j+1]==0) return false;
        return true;
    }
}
```

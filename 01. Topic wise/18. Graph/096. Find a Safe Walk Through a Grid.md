### Backtracking
```java
class Solution {
    public boolean findSafeWalk(List<List<Integer>> grid, int health) {
        return rec(grid, health-grid.get(0).get(0), new boolean[grid.size()][grid.get(0).size()], 0, 0);
    }
    boolean rec(List<List<Integer>> grid, int health, boolean[][] vis, int i, int j) {
        if(health==0) return false;
        int m = grid.size(), n = grid.get(0).size();
        if(i==m-1 && j==n-1) return true;
        vis[i][j] = true;
        if(i+1<m && !vis[i+1][j] && rec(grid, health-grid.get(i+1).get(j), vis, i+1, j)) return true;
        if(j+1<n && !vis[i][j+1] && rec(grid, health-grid.get(i).get(j+1), vis, i, j+1)) return true;
        if(i-1>=0 && !vis[i-1][j] && rec(grid, health-grid.get(i-1).get(j), vis, i-1, j)) return true;
        if(j-1>=0 && !vis[i][j-1] && rec(grid, health-grid.get(i).get(j-1), vis, i, j-1)) return true;
        vis[i][j] = false;
        return false;
    }
}
```

### Dijkstra
```java
class Solution {
    public boolean findSafeWalk(List<List<Integer>> grid, int health) {
        int m = grid.size(), n = grid.get(0).size();
        int[][] dist = new int[m][n];
        for(int[] dis: dist) Arrays.fill(dis, Integer.MAX_VALUE);
        boolean[][] vis = new boolean[m][n];
        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> a[2] - b[2]);
        pq.add(new int[] {0, 0, grid.get(0).get(0)});
        dist[0][0] = grid.get(0).get(0);
        int[] di = new int[] {1, 0, -1, 0};
        int[] dj = new int[] {0, 1, 0, -1};
        while(!pq.isEmpty()) {
            int[] e = pq.poll();
            int i = e[0], j = e[1], cost = e[2];
            if(i==m-1 && j==n-1) return health-cost>0;
            if(vis[i][j]) continue;
            vis[i][j] = true;
            for(int idx=0;idx<4;idx++) {
                int newI = i+di[idx], newJ = j+dj[idx];
                if(newI>=0 && newI<m && newJ>=0 && newJ<n && !vis[newI][newJ] && dist[newI][newJ]>(cost+grid.get(newI).get(newJ))) {
                    dist[newI][newJ] = cost+grid.get(newI).get(newJ);
                    pq.add(new int[] {newI, newJ, dist[newI][newJ]});
                }
            }
        }
        return false;
    }
}
```

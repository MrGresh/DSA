```
Constraints:

1 ≤ n ≤ 105
1 ≤ adj[i][j]
adj.size() = n
```
#### Note: Floyd-Washall is not a good choice here bcoz of time and space complexity
___
### BFS + Diameter
```java
import java.util.*;

class Solution {
    public int partyHouse(ArrayList<ArrayList<Integer>> adj) {
        int n = adj.size();
        
        // Build 0-indexed adjacency list (input values are 1-indexed houses)
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) graph.add(new ArrayList<>());
        for (int u = 0; u < n; u++) {
            for (int v : adj.get(u)) {
                graph.get(u).add(v - 1); // convert 1-indexed neighbor to 0-indexed
            }
        }
        
        // Edge case: single house
        if (n == 1) return 0;
        
        // Step 1: BFS from any node (say 0) to find the farthest node A
        int[] res1 = bfs(0, graph, n);
        int nodeA = res1[0];
        
        // Step 2: BFS from A to find farthest node B; this distance = diameter
        int[] res2 = bfs(nodeA, graph, n);
        int diameter = res2[1];
        
        // Step 3: radius = ceil(diameter / 2)
        return (diameter + 1) / 2;
    }
    
    // Returns {farthestNodeIndex, maxDistance} from BFS starting at src
    private int[] bfs(int src, List<List<Integer>> graph, int n) {
        int[] dist = new int[n];
        Arrays.fill(dist, -1);
        dist[src] = 0;
        
        Queue<Integer> queue = new LinkedList<>();
        queue.add(src);
        
        int farthestNode = src;
        int maxDist = 0;
        
        while (!queue.isEmpty()) {
            int u = queue.poll();
            if (dist[u] > maxDist) {
                maxDist = dist[u];
                farthestNode = u;
            }
            for (int v : graph.get(u)) {
                if (dist[v] == -1) {
                    dist[v] = dist[u] + 1;
                    queue.add(v);
                }
            }
        }
        
        return new int[]{farthestNode, maxDist};
    }
}
```

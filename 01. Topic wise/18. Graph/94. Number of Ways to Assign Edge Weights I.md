class TreeNode {
    int val;
    List<TreeNode> childs;
    TreeNode(int a) {
        val = a;
        childs = new ArrayList<>();
    }
}
class Solution {
    public int assignEdgeWeights(int[][] edges) {
        Map<Integer, TreeNode> map = new HashMap<>();
        for(int[] e: edges) {
            if(!map.containsKey(e[0])) map.put(e[0], new TreeNode(e[0]));
            if(!map.containsKey(e[1])) map.put(e[1], new TreeNode(e[1]));
            TreeNode node1 = map.get(e[0]);
            TreeNode node2 = map.get(e[1]);
            node1.childs.add(node2);
        }
        Queue<TreeNode> q = new LinkedList<>();
        q.add(map.get(1));
        int level = 0;
        while(!q.isEmpty()) {
            level++;
            int size = q.size();
            while(size-->0) {
                TreeNode node = q.poll();
                for(TreeNode child: node.childs) q.add(child);
            }
        }
        return rec(level-1, 0);
    }
    int rec(int curr, int sum) {
        if(curr==0) return (sum%2!=0) ? 1 : 0;
        return rec(curr-1, 1) + rec(curr-1, 2);
    }
}

import java.util.*;

class Solution {
    public int assignEdgeWeights(int[][] edges) {
        int n = edges.length + 1;
        
        // 1. Build an undirected adjacency list
        List<List<Integer>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        for (int[] e : edges) {
            adj.get(e[0]).add(e[1]);
            adj.get(e[1]).add(e[0]);
        }
        
        // 2. BFS to find the maximum depth (number of edges from root to deepest node)
        Queue<Integer> q = new LinkedList<>();
        boolean[] visited = new boolean[n + 1];
        
        q.add(1);
        visited[1] = true;
        
        int maxEdges = -1; // Root node itself represents a path of 0 edges
        
        while (!q.isEmpty()) {
            int size = q.size();
            maxEdges++; // Increment depth level
            
            for (int i = 0; i < size; i++) {
                int curr = q.poll();
                for (int neighbor : adj.get(curr)) {
                    if (!visited[neighbor]) {
                        visited[neighbor] = true;
                        q.add(neighbor);
                    }
                }
            }
        }
        
        // 3. Compute 2^(maxEdges - 1) % (10^9 + 7)
        if (maxEdges == 0) return 0; // Guard rail for an isolated single node tree
        
        return (int) power(2, maxEdges - 1, 1000000007);
    }
    
    // Fast modular exponentiation: O(log exponent)
    private long power(long base, long exp, long mod) {
        long res = 1;
        base = base % mod;
        while (exp > 0) {
            if (exp % 2 == 1) {
                res = (res * base) % mod;
            }
            base = (base * base) % mod;
            exp /= 2;
        }
        return res;
    }
}

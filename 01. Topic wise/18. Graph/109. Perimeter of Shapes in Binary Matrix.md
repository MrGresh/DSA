class Solution {
    static int findPerimeter(int[][] mat) {
        int r = mat.length, c = mat[0].length;
        int ans = 0;
        boolean[][] vis = new boolean[r][c];
        for(int i=0;i<r;i++) {
            for(int j=0;j<c;j++) {
                if(mat[i][j]==1 && !vis[i][j]) {
                    ans += rec(mat, vis, i, j);
                }
            }
        }
        return ans;
    }
    static int rec(int[][] mat, boolean[][] vis, int i, int j) {
        int r = mat.length, c = mat[0].length;
        if(i<0 || j<0 || i==r || j==c || mat[i][j]==0) return 1;
        if(vis[i][j]) return 0;
        vis[i][j] = true;
        return rec(mat, vis, i-1, j) + rec(mat, vis, i+1, j) + rec(mat, vis, i, j-1) + rec(mat, vis, i, j+1);
    }
}

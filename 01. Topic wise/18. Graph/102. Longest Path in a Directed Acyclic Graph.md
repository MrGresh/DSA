### Memoization
```java
class Solution {
    static List<List<int[]>> radj; // reverse adjacency: radj[v] = list of {u, w} for edge u->v
    static int[] dist;
    static boolean[] computed;
    static final int NEG_INF = Integer.MIN_VALUE;

    public int[] maxDistance(int V, int src, ArrayList<ArrayList<Integer>> edges) {
        radj = new ArrayList<>();
        for (int i = 0; i < V; i++) radj.add(new ArrayList<>());

        for (ArrayList<Integer> e : edges) {
            int u = e.get(0), v = e.get(1), w = e.get(2);
            radj.get(v).add(new int[]{u, w}); // reverse edge: v <- u with weight w
        }

        dist = new int[V];
        computed = new boolean[V];

        for (int i = 0; i < V; i++) {
            solve(i, src);
        }

        return dist;
    }

    private int solve(int u, int src) {
        if (computed[u]) return dist[u];
        computed[u] = true; // mark before recursing to be safe (DAG, so no cycles to worry about)

        if (u == src) {
            dist[u] = 0;
            return 0;
        }

        int best = NEG_INF;
        for (int[] edge : radj.get(u)) {
            int p = edge[0], w = edge[1];
            int pd = solve(p, src);
            if (pd != NEG_INF) best = Math.max(best, pd + w);
        }

        dist[u] = best;
        return best;
    }
}
```

### Topo sort
```java
class Solution {
    public int[] maxDistance(int V, int src, ArrayList<ArrayList<Integer>> edges) {
        // Build adjacency list: each entry is {v, w}
        List<List<int[]>> adj = new ArrayList<>();
        for (int i = 0; i < V; i++) adj.add(new ArrayList<>());
        
        int[] indegree = new int[V];
        
        for (ArrayList<Integer> e : edges) {
            int u = e.get(0), v = e.get(1), w = e.get(2);
            adj.get(u).add(new int[]{v, w});
            indegree[v]++;
        }
        
        // Topological sort using Kahn's algorithm
        int[] topoOrder = new int[V];
        int idx = 0;
        Queue<Integer> queue = new LinkedList<>();
        int[] indegreeCopy = indegree.clone();
        
        for (int i = 0; i < V; i++) {
            if (indegreeCopy[i] == 0) queue.add(i);
        }
        
        while (!queue.isEmpty()) {
            int u = queue.poll();
            topoOrder[idx++] = u;
            for (int[] edge : adj.get(u)) {
                int v = edge[0];
                indegreeCopy[v]--;
                if (indegreeCopy[v] == 0) queue.add(v);
            }
        }
        
        // Initialize distances
        int[] dist = new int[V];
        Arrays.fill(dist, Integer.MIN_VALUE);
        dist[src] = 0;
        
        // Relax edges in topological order
        for (int i = 0; i < V; i++) {
            int u = topoOrder[i];
            if (dist[u] == Integer.MIN_VALUE) continue; // unreachable, skip
            
            for (int[] edge : adj.get(u)) {
                int v = edge[0], w = edge[1];
                if (dist[u] + w > dist[v]) {
                    dist[v] = dist[u] + w;
                }
            }
        }
        
        return dist;
    }
}
```

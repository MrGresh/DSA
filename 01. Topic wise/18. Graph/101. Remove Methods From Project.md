class Solution {
    Map<Integer, List<Integer>> graph;
    boolean[] badVis;
    boolean[] vis;
    public List<Integer> remainingMethods(int n, int k, int[][] edges) {
        graph = new HashMap<>();
        for(int[] e: edges) {
            if(!graph.containsKey(e[0])) graph.put(e[0], new ArrayList<>());
            graph.get(e[0]).add(e[1]);
        }
        badVis = new boolean[n];
        dfs(k);
        vis = new boolean[n];
        boolean noNeedToRemove = false;
        for(int i=0;i<n;i++) {
            if(!badVis[i] && !vis[i] && dfs2(i)) {
                noNeedToRemove = true;
                break;
            }
        }
        List<Integer> res = new ArrayList<>();
        for(int i=0;i<n;i++) {
            if(noNeedToRemove) res.add(i);
            else {
                if(!badVis[i]) res.add(i);
            }
        }
        return res;
    }
    void dfs(int node) {
        badVis[node] = true;
        if(graph.containsKey(node)) {
            for(int nei: graph.get(node)) {
                if(!badVis[nei]) dfs(nei);
            }
        }
    }
    boolean dfs2(int node) {
        vis[node] = true;
        if(graph.containsKey(node)) {
            for(int nei: graph.get(node)) {
                if(badVis[nei]) return true;
                if(!vis[nei] && dfs2(nei)) return true;
            }
        }
        return false;
    }
}

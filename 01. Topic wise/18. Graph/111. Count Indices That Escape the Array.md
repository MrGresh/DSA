class Solution {
    int n;
    int[] arr, state;      // state: 0 = unvisited, 1 = in current path, 2 = done
    boolean[] memo;

    public int goodIndices(int[] arr) {
        this.arr = arr;
        n = arr.length;
        state = new int[n];
        memo = new boolean[n];

        int count = 0;
        for (int i = 0; i < n; i++) {
            if (dfs(i)) count++;
        }
        return count;
    }

    private boolean dfs(int i) {
        if (i < 0 || i >= n) return true;      // escaped the array
        if (state[i] == 2) return memo[i];     // already resolved
        if (state[i] == 1) return false;       // back on the current path: cycle

        state[i] = 1;
        boolean res = dfs(i + arr[i]);         // a 0 jump calls dfs(i) -> state 1 -> false
        state[i] = 2;
        memo[i] = res;
        return res;
    }
}

### BFS
```java
class Solution {
    public Node cloneGraph(Node node) {
        if (node == null) return null;

        Map<Node, Node> map = new HashMap<>(); // keep clones of each node
        Queue<Node> queue = new LinkedList<>();

        map.put(node, new Node(node.val));
        queue.add(node);
        boolean[] visited = new boolean[101]; // Not necessarily required
        while (!queue.isEmpty()) {
            Node curr = queue.poll();

            for (Node neighbor : curr.neighbors) {
                if (!visited[neighbor.val]) { // !map.containsKey(neighbor) also work
                    map.put(neighbor, new Node(neighbor.val));
                    visited[neighbor.val] = true;
                    queue.add(neighbor);
                }
                Node cloneOfCurr = map.get(curr);
                Node cloneOfNei = map.get(neighbor);
                cloneOfCurr.neighbors.add(cloneOfNei);
            }
        }

        return map.get(node);
    }
}
```

### DFS
```java
class Solution {
    private Map<Node, Node> visited = new HashMap<>();

    public Node cloneGraph(Node node) {
        if (node == null) return null;

        if (visited.containsKey(node)) return visited.get(node);

        Node cloneNode = new Node(node.val, new ArrayList<>());
        
        visited.put(node, cloneNode);

        for (Node neighbor : node.neighbors) {
            cloneNode.neighbors.add(cloneGraph(neighbor));
        }

        return cloneNode;
    }
}
```

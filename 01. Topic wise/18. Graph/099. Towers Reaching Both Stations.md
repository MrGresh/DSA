class Solution {
    public int countCoordinates(int[][] mat) {
        int r = mat.length, c = mat[0].length;
        boolean[][] vis1 = new boolean[r][c];
        boolean[][] vis2 = new boolean[r][c];
        for(int j=0;j<c;j++) {
            rec(mat, vis1, r, c, 0, j);
            rec(mat, vis2, r, c, r-1, j);
        }
        for(int i=0;i<r;i++) {
            rec(mat, vis1, r, c, i, 0);
            rec(mat, vis2, r, c, i, c-1);
        }
        int cnt = 0;
        for(int i=0;i<r;i++) {
            for(int j=0;j<c;j++) {
                if(vis1[i][j] && vis2[i][j]) cnt++;
            }
        }
        return cnt;
    }
    void rec(int[][] mat, boolean[][] vis, int r, int c, int i, int j) {
        vis[i][j] = true;
        if(i-1>=0 && !vis[i-1][j] && mat[i][j]<=mat[i-1][j]) rec(mat, vis, r, c, i-1, j);
        if(j-1>=0 && !vis[i][j-1] && mat[i][j]<=mat[i][j-1]) rec(mat, vis, r, c, i, j-1);
        if(i+1<r && !vis[i+1][j] && mat[i][j]<=mat[i+1][j]) rec(mat, vis, r, c, i+1, j);
        if(j+1<c && !vis[i][j+1] && mat[i][j]<=mat[i][j+1]) rec(mat, vis, r, c, i, j+1);
    }
}

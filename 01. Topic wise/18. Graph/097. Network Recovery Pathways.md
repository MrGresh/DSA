### Backtracking

```java
class Solution {
    public int findMaxPathScore(int[][] edges, boolean[] online, long k) {
        Map<Integer, List<int[]>> graph = new HashMap<>();
        int lastEdge = 0;
        for(int[] e: edges) {
            lastEdge = Math.max(lastEdge, Math.max(e[0], e[1]));
            if(!graph.containsKey(e[0])) graph.put(e[0], new ArrayList<>());
            graph.get(e[0]).add(new int[] {e[1], e[2]});
        }
        int ans = rec(graph, online, k, online.length - 1, 0);
        return ans == Integer.MIN_VALUE ? -1 : ans;
    }
    int rec(Map<Integer, List<int[]>> graph, boolean[] online, long k, int lastEdge, int edge) {
        if(edge==lastEdge) return Integer.MAX_VALUE;
        int ans = Integer.MIN_VALUE;
        for(int[] e: graph.getOrDefault(edge, new ArrayList<>())) {
            if(online[e[0]] && k-e[1]>=0) {
                ans = Math.max(ans, Math.min(e[1], rec(graph, online, k-e[1], lastEdge, e[0])));
            }
        }
        return ans;
    }
}
```

### Binary Search
```java
import java.util.*;

class Solution {
    public int findMaxPathScore(int[][] edges, boolean[] online, long k) {
        int n = online.length;
        
        // 1. Collect all unique edge costs to define our binary search space
        TreeSet<Integer> weights = new TreeSet<>();
        for (int[] e : edges) {
            weights.add(e[2]);
        }
        
        List<Integer> distinctWeights = new ArrayList<>(weights);
        int low = 0, high = distinctWeights.size() - 1;
        int ans = -1;
        
        // 2. Build adjacency list for efficient traversal
        List<int[]>[] graph = new ArrayList[n];
        for (int i = 0; i < n; i++) graph[i] = new ArrayList<>();
        for (int[] e : edges) {
            graph[e[0]].add(new int[]{e[1], e[2]});
        }
        
        // 3. Find topological order of the DAG
        List<Integer> topoOrder = getTopoOrder(n, graph);
        
        // 4. Binary Search over the possible bottleneck values
        while (low <= high) {
            int mid = low + (high - low) / 2;
            int minEdgeAllowed = distinctWeights.get(mid);
            
            if (isValid(n, graph, online, topoOrder, minEdgeAllowed, k)) {
                ans = minEdgeAllowed; // Found a working bottleneck, try to find a larger one
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }
        
        return ans;
    }
    
    // Helper to find the minimum total cost path using only edges >= minEdgeAllowed
    private boolean isValid(int n, List<int[]>[] graph, boolean[] online, List<Integer> topoOrder, int minEdgeAllowed, long k) {
        long[] dist = new long[n];
        Arrays.fill(dist, Long.MAX_VALUE);
        dist[0] = 0;
        
        for (int u : topoOrder) {
            if (dist[u] == Long.MAX_VALUE || !online[u]) continue;
            
            for (int[] edge : graph[u]) {
                int v = edge[0];
                int weight = edge[1];
                
                // Only traverse if the node is online and edge weight satisfies our binary search guess
                if (online[v] && weight >= minEdgeAllowed) {
                    if (dist[u] + weight < dist[v]) {
                        dist[v] = dist[u] + weight;
                    }
                }
            }
        }
        
        return dist[n - 1] <= k;
    }
    
    // Standard Kahn's algorithm for Topological Sort
    private List<Integer> getTopoOrder(int n, List<int[]>[] graph) {
        int[] inDegree = new int[n];
        for (int u = 0; u < n; u++) {
            for (int[] edge : graph[u]) {
                inDegree[edge[0]]++;
            }
        }
        
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            if (inDegree[i] == 0) queue.add(i);
        }
        
        List<Integer> topo = new ArrayList<>();
        while (!queue.isEmpty()) {
            int u = queue.poll();
            topo.add(u);
            for (int[] edge : graph[u]) {
                inDegree[edge[0]]--;
                if (inDegree[edge[0]] == 0) {
                    queue.add(edge[0]);
                }
            }
        }
        return topo;
    }
}
```

class Solution {
    public int minTime(int[] duration, int[][] dependencies) {
        int n = duration.length;

        // Build graph using adjacency list
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }

        int[] indeg = new int[n];

        for (int[] dep : dependencies) {
            int u = dep[0];
            int v = dep[1];
            graph.get(u).add(v);
            indeg[v]++;
        }

        // finish[i] = earliest time module i can be completed
        long[] finish = new long[n];
        Queue<Integer> queue = new ArrayDeque<>();

        for (int i = 0; i < n; i++) {
            if (indeg[i] == 0) {
                queue.offer(i);
                finish[i] = duration[i];
            }
        }

        int processed = 0;
        long ans = 0;

        while (!queue.isEmpty()) {
            int u = queue.poll();
            processed++;
            ans = Math.max(ans, finish[u]);

            for (int v : graph.get(u)) {
                finish[v] = Math.max(finish[v], finish[u] + duration[v]);
                if (--indeg[v] == 0) queue.offer(v);
            }
        }

        if (processed != n) return -1;
        return (int) ans;
    }
}

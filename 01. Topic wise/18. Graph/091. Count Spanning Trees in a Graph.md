### Backtracking
```java
class Solution {
    int count = 0;

    public int countSpanTree(int n, int[][] edges) {
        if (n <= 1) return 1;
        count = 0;
        
        // We need to pick exactly n-1 edges out of m
        backtrack(edges, 0, new ArrayList<int[]>(), n);
        return count;
    }

    private void backtrack(int[][] edges, int index, List<int[]> currentEdges, int n) {
        // Base case: if we have n-1 edges, check if they form a spanning tree
        if (currentEdges.size() == n - 1) {
            if (isConnected(n, currentEdges)) {
                count++;
            }
            return;
        }

        // If we've run out of edges to pick, stop
        if (index == edges.length) return;

        // Option 1: Pick the current edge
        currentEdges.add(edges[index]);
        backtrack(edges, index + 1, currentEdges, n);
        
        // Option 2: Don't pick the current edge (backtrack)
        currentEdges.remove(currentEdges.size() - 1);
        backtrack(edges, index + 1, currentEdges, n);
    }

    private boolean isConnected(int n, List<int[]> edgeList) {
        List<List<Integer>> adj = new ArrayList<>();
        for (int i = 0; i < n; i++) adj.add(new ArrayList<>());
        
        for (int[] e : edgeList) {
            adj.get(e[0]).add(e[1]);
            adj.get(e[1]).add(e[0]);
        }

        // BFS/DFS to check connectivity
        boolean[] visited = new boolean[n];
        Queue<Integer> q = new LinkedList<>();
        q.add(0);
        visited[0] = true;
        int visitedCount = 1;

        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : adj.get(u)) {
                if (!visited[v]) {
                    visited[v] = true;
                    visitedCount++;
                    q.add(v);
                }
            }
        }

        return visitedCount == n;
    }
}
```

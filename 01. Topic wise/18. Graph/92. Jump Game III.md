### BFS
```java
class Solution {
    public boolean canReach(int[] arr, int start) {
        if(arr[start]==0) return true;
        Queue<Integer> q = new LinkedList<>();
        boolean[] vis = new boolean[arr.length];
        q.add(start);
        vis[start] = true;
        while(!q.isEmpty()) {
            int i = q.poll();
            if((i-arr[i])>=0 && !vis[i-arr[i]]) {
                if(arr[i-arr[i]]==0) return true;
                q.add(i-arr[i]);
                vis[i-arr[i]] = true;
            }
            if((i+arr[i])<arr.length && !vis[i+arr[i]]) {
                if(arr[i+arr[i]]==0) return true;
                q.add(i+arr[i]);
                vis[i+arr[i]] = true;
            }
        }
        return false;
    }
}
```

### DFS
```java
class Solution {
    public boolean canReach(int[] arr, int start) {
        return rec(arr, start, new boolean[arr.length]);
    }
    boolean rec(int[] arr, int i, boolean[] vis) {
        if(arr[i]==0) return true;
        vis[i] = true;
        if((i-arr[i])>=0 && !vis[i-arr[i]] && rec(arr, i-arr[i], vis)) return true;
        if((i+arr[i])<arr.length && !vis[i+arr[i]] && rec(arr, i+arr[i], vis)) return true;
        return false;
    }
}
```

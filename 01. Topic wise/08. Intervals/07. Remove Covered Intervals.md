class Solution {
    public int removeCoveredIntervals(int[][] intervals) {
        Arrays.sort(intervals, (a, b) -> {
            if (a[0] == b[0]) {
                return b[1] - a[1];
            }
            return a[0] - b[0];
        });
        
        int remainingCount = intervals.length;
        int maxRight = 0;
        
        for (int[] interval : intervals) {
            int currentRight = interval[1];
            
            if (currentRight <= maxRight) remainingCount--;
            else maxRight = currentRight;
        }
        
        return remainingCount;
    }
}

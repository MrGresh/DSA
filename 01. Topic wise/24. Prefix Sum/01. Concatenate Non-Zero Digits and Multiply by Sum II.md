class Solution {
    public int[] sumAndMultiply(String s, int[][] queries) {
        int n = s.length();
        long MOD = 1_000_000_007;
        
        // Precompute powers of 10
        long[] pow10 = new long[n + 1];
        pow10[0] = 1;
        for (int i = 1; i <= n; i++) {
            pow10[i] = (pow10[i - 1] * 10) % MOD;
        }
        
        // 1-based indexing for prefix arrays
        long[] P = new long[n + 1]; // Prefix values of non-zero digits
        int[] S = new int[n + 1];    // Prefix sum of digits
        int[] C = new int[n + 1];    // Prefix count of non-zero digits
        
        for (int i = 0; i < n; i++) {
            char ch = s.charAt(i);
            int digit = ch - '0';
            
            if (digit != 0) {
                P[i + 1] = (P[i] * 10 + digit) % MOD;
                C[i + 1] = C[i] + 1;
            } else {
                P[i + 1] = P[i];
                C[i + 1] = C[i];
            }
            S[i + 1] = S[i] + digit;
        }
        
        int[] answer = new int[queries.length];
        
        for (int i = 0; i < queries.length; i++) {
            int l = queries[i][0];
            int r = queries[i][1];
            
            // Convert to 1-based indices for our prefix arrays
            int L = l; 
            int R = r + 1;
            
            int k = C[R] - C[L];
            long x = (P[R] - (P[L] * pow10[k]) % MOD + MOD) % MOD;
            long sum = S[R] - S[L];
            
            answer[i] = (int) ((x * sum) % MOD);
        }
        
        return answer;
    }
}

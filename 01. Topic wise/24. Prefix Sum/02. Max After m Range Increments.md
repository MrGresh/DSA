class Solution {
    public int findMax(int n, int[] a, int[] b, int[] k) {
        long[] diff = new long[n + 1]; // +1 to safely handle b[i]+1 == n
        
        for (int i = 0; i < a.length; i++) {
            diff[a[i]] += k[i];
            if (b[i] + 1 <= n) diff[b[i] + 1] -= k[i];
        }
        
        long max = 0;
        long running = 0;
        for (int j = 0; j < n; j++) {
            running += diff[j];
            max = Math.max(max, running);
        }
        
        return (int) max;
    }
}

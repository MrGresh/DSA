```java
class Solution {
    public int maxSumWithK(int[] arr, int k) {
        int n = arr.length;
        long[] prefix = new long[n + 1];
        for (int i = 0; i < n; i++) {
            prefix[i + 1] = prefix[i] + arr[i];
        }

        long minPrefix = prefix[0];   // smallest prefix[i] among valid i's seen so far
        long ans = Long.MIN_VALUE;

        for (int j = k; j <= n; j++) {
            // window [i, j) has length j - i >= k  =>  i <= j - k
            ans = Math.max(ans, prefix[j] - minPrefix);
            // once j increases, i = j - k + 1 becomes newly valid for the next step
            minPrefix = Math.min(minPrefix, prefix[j - k + 1]);
        }

        return (int) ans;
    }
}
```

### Inefficient DP Memoization
```java
class Solution {
    Integer[][] memo;
    public int maxSumWithK(int[] arr, int k) {
        memo = new Integer[arr.length+1][k];
        int ans = Integer.MIN_VALUE;
        for(int i=0;i<=arr.length-k;i++) {
            ans = Math.max(ans, arr[i]+rec(arr, i+1, k-1));
        }
        return ans;
    }
    int rec(int[] arr, int idx, int k) {
        if(memo[idx][k]!=null) return memo[idx][k];
        if(k==0) {
            int val = 0, curr=0;
            for(int i=idx;i<arr.length;i++) {
                curr += arr[i];
                val = Math.max(val, curr);
            }
            return memo[idx][k] = val;
        } else return memo[idx][k] = arr[idx] + rec(arr, idx+1, k-1);
    }
}
```

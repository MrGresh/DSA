### Recursion
```java
class Solution {
    public int computeValue(int n) {
        return rec(n, 0, 0);
    }
    int rec(int n, int sum1, int sum2) {
        if(n==0) return sum1==sum2 ? 1 : 0;
        int ans = 0;
        ans += rec(n-1, sum1, sum2);
        ans += rec(n-1, sum1+1, sum2);
        ans += rec(n-1, sum1, sum2+1);
        ans += rec(n-1, sum1+1, sum2+1);
        return ans;
    }
}
```

### Efficient
```java
class Solution {
    private static final int MOD = 1000000007;

    public int computeValue(int n) {
        // We need to calculate (2n)! / (n! * n!) % MOD
        long numerator = 1;
        long denominator = 1;

        // Calculate (2n)! and n! in a single loop to save time
        for (int i = 1; i <= n; i++) {
            numerator = (numerator * (n + i)) % MOD; // Represents (2n * (2n-1) * ... * (n+1))
            denominator = (denominator * i) % MOD;   // Represents n!
        }

        // Result = numerator * (denominator)^(-1) % MOD
        long ans = (numerator * power(denominator, MOD - 2)) % MOD;
        return (int) ans;
    }

    // Binary Exponentiation to find (base^exp) % MOD
    private long power(long base, long exp) {
        long res = 1;
        base = base % MOD;
        while (exp > 0) {
            if ((exp & 1) == 1) {
                res = (res * base) % MOD;
            }
            base = (base * base) % MOD;
            exp >>= 1;
        }
        return res;
    }
}
```

class Solution {
    boolean[][] vis;
    public int longestPath(int[][] mat, int xs, int ys, int xd, int yd) {
        if (mat[xs][ys] == 0 || mat[xd][yd] == 0) return -1;
        int r = mat.length, c = mat[0].length;
        vis = new boolean[r][c];
        int ans = rec(mat, r, c, xd, yd, xs, ys);
        return ans == Integer.MIN_VALUE ? -1 : ans;
    }
    int rec(int[][] mat, int r, int c, int xd, int yd, int i, int j) {
        if(i==xd && j==yd) return 0;
        vis[i][j] = true;
        int ans = Integer.MIN_VALUE;
        if(i-1>=0 && !vis[i-1][j] && mat[i-1][j]==1) ans = rec(mat, r, c, xd, yd, i-1, j);
        if(j-1>=0 && !vis[i][j-1] && mat[i][j-1]==1) ans = Math.max(ans, rec(mat, r, c, xd, yd, i, j-1));
        if(i+1<r && !vis[i+1][j] && mat[i+1][j]==1) ans = Math.max(ans, rec(mat, r, c, xd, yd, i+1, j));
        if(j+1<c && !vis[i][j+1] && mat[i][j+1]==1) ans = Math.max(ans, rec(mat, r, c, xd, yd, i, j+1));
        vis[i][j] = false;
        return ans == Integer.MIN_VALUE ? ans : ans + 1;
    }
}

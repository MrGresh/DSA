Remove the **minimum** number of invalid parentheses from s so that the resulting string becomes valid.  Return all the possible distinct valid strings in lexicographically sorted order.
> **Constraints:**
1 ≤ |s| ≤ 20,
s consists of lowercase English letters and parentheses '(' and ')'
```java
class Solution {
    Set<String> res;
    public List<String> validParenthesis(String s) {
        Stack<Character> st = new Stack<>();
        int bad = 0;
        for(int i=0;i<s.length();i++) {
            if(s.charAt(i)=='(') st.push('(');
            else if(s.charAt(i)==')') {
                if(st.isEmpty()) bad++;
                else st.pop();
            }
        }
        int validLength = s.length()-(bad+st.size());
        res = new HashSet<>();
        if(validLength>0) rec(s, validLength, 0, 0, 0, new StringBuilder());
        else res.add("");
        List<String> ans = new ArrayList<>(res);
        Collections.sort(ans);
        return ans;
    }
    void rec(String s, int validLength, int i, int open, int close, StringBuilder curr) {
        if(i==s.length()) {
            if(curr.toString().length()==validLength && open==close) res.add(curr.toString());
            return;
        }
        if(s.charAt(i)=='(') {
            curr.append('(');
            rec(s, validLength, i+1, open+1, close, curr);
            curr.deleteCharAt(curr.length() - 1);
            rec(s, validLength, i+1, open, close, curr);
        } else if(s.charAt(i)==')') {
            if(close<open) {
                curr.append(')');
                rec(s, validLength, i+1, open, close+1, curr);
                curr.deleteCharAt(curr.length() - 1);
            }
            rec(s, validLength, i+1, open, close, curr);
        } else {
            curr.append(s.charAt(i));
            rec(s, validLength, i+1, open, close, curr);
            curr.deleteCharAt(curr.length() - 1);
        }
    }
}
```

```java
        List<String> ans = new ArrayList<>(res);
        Collections.sort(ans);
        return ans;
    }
    void rec(String s, int validLength, int i, int balance, StringBuilder curr) {
        if(i==s.length()) {
            if(curr.toString().length()==validLength && balance==0) res.add(curr.toString());
            return;
        }
        if(s.charAt(i)=='(') {
            curr.append('(');
            rec(s, validLength, i+1, balance+1, curr);
            curr.deleteCharAt(curr.length() - 1);
            rec(s, validLength, i+1, balance, curr);
        } else if(s.charAt(i)==')') {
            if(balance>0) {
                curr.append(')');
                rec(s, validLength, i+1, balance-1, curr);
                curr.deleteCharAt(curr.length() - 1);
            }
            rec(s, validLength, i+1, balance, curr);
        } else {
            curr.append(s.charAt(i));
            rec(s, validLength, i+1, balance, curr);
            curr.deleteCharAt(curr.length() - 1);
        }
    }
}
```

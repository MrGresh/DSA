class Solution {
    public String lexGreaterPermutation(String s, String target) {
        int[] freq = new int[26];
        for(int i=0;i<s.length();i++) freq[s.charAt(i)-'a']++;
        StringBuilder ans = new StringBuilder();
        rec(s, target, ans, freq, 0);
        if(ans.length()==0) return "";
        // Append remaining in sorted order
        for(int i=0;i<26;i++) {
            char ch = (char)('a'+i);
            for(int j=1;j<=freq[i];j++) ans.append(ch);
        }
        return ans.toString();
    }
    boolean rec(String s, String target, StringBuilder ans, int[] freq, int j) {
        if(j==target.length()) return false;
        char ch = target.charAt(j);
        int idx = ch-'a';
        for(int i=idx;i<26;i++) {
            int val = freq[i];
            if(val>0) {
                if(i>idx) {
                    ans.append((char)(i+'a'));
                    freq[i]--;
                    return true;
                } else {
                    ans.append(ch);
                    freq[i]--;
                    if(rec(s, target, ans, freq, j+1)) return true;
                    ans.deleteCharAt(ans.length() - 1);
                    freq[i]++;
                }
            }
        }
        return false;
    }
}

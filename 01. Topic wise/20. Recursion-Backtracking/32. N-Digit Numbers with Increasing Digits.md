class Solution {
    static ArrayList<Integer> res;
    public static ArrayList<Integer> increasingNumbers(int n) {
        res = new ArrayList<>();
        if(n>9) return res;
        if(n==1) res.add(0);
        rec(n, 0, 0);
        return res;
    }
    static void rec(int n, int currNum, int lastDigit) {
        if(n==0) {
            res.add(currNum);
            return;
        }
        for(int i=lastDigit+1;i<=9;i++) {
            if((9-i)<(n-1)) break;
            rec(n-1, currNum*10+i, i);
        }
    }
}

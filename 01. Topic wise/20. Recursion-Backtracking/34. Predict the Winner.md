class Solution {
    public boolean predictTheWinner(int[] nums) {
        return rec(nums, 0, nums.length-1, 1, 0, 0);
    }
    boolean rec(int[] nums, int l, int r, int turn, int s1, int s2) {
        if(l>r) return s1>=s2;
        if(turn==1) {
            if(rec(nums, l+1, r, 2, s1+nums[l], s2)) return true;;
            return rec(nums, l, r-1, 2, s1+nums[r], s2);
        } else {
            if(!rec(nums, l+1, r, 1, s1, s2+nums[l])) return false;
            return rec(nums, l, r-1, 1, s1, s2+nums[r]);
        }
    }
}

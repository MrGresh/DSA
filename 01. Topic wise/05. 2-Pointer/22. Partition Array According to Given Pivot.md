```java
class Solution {
    public int[] pivotArray(int[] nums, int piv) {
        List<Integer> less = new ArrayList<>();
        List<Integer> equal = new ArrayList<>();
        List<Integer> greater = new ArrayList<>();
        for(int x: nums) {
            if(x<piv) less.add(x);
            else if(x==piv) equal.add(x);
            else greater.add(x);
        }
        int idx = -1;
        for(int x: less) nums[++idx] = x;
        for(int x: equal) nums[++idx] = x;
        for(int x: greater) nums[++idx] = x;
        return nums;
    }
}
```

### 3 Pointer
```java
class Solution {
    public int[] pivotArray(int[] nums, int pivot) {
        int n = nums.length;
        int[] result = new int[n];
        
        int lessCount = 0;
        int equalCount = 0;
        
        for (int num : nums) {
            if (num < pivot) {
                lessCount++;
            } else if (num == pivot) {
                equalCount++;
            }
        }
        
        int leftIdx = 0;
        int equalIdx = lessCount;
        int rightIdx = lessCount + equalCount;
        
        for (int num : nums) {
            if (num < pivot) {
                result[leftIdx++] = num;
            } else if (num == pivot) {
                result[equalIdx++] = num;
            } else {
                result[rightIdx++] = num;
            }
        }
        return result;
    }
}
```

class Solution {
    public void mergeTwoParts(int[] arr) {
        int p1=0, p2=arr.length;
        for(int i=1;i<arr.length;i++) {
            if(arr[i]<arr[i-1]) {
                p2 = i;
                break;
            }
        }
        if(p2<arr.length) {
            int idx = -1;
            int[] newArr = new int[arr.length];
            int m = p2-1;
            while(p1<=m && p2<arr.length) {
                if(arr[p1]<=arr[p2]) newArr[++idx] = arr[p1++];
                else newArr[++idx] = arr[p2++];
            }
            while(p1<=m) newArr[++idx] = arr[p1++];
            while(p2<arr.length) newArr[++idx] = arr[p2++];
            for(int i=0;i<arr.length;i++) arr[i] = newArr[i];
        }
    }
}

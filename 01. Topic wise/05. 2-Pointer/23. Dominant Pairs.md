### Sorting + 2 Pointer
```java
class Solution {
    public int dominantPairs(int[] arr) {
        int[] firstHalf = new int[arr.length/2];
        int[] secondHalf = new int[arr.length/2];
        int idx = -1;
        for(int i=0;i<arr.length/2;i++) firstHalf[++idx] = arr[i];
        idx = -1;
        for(int i=arr.length/2;i<arr.length;i++) secondHalf[++idx] = arr[i];
        Arrays.sort(firstHalf);
        Arrays.sort(secondHalf);
        int cnt = 0;
        int i=firstHalf.length-1, j=secondHalf.length-1;
        while(i>=0) {
            while(j>=0) {
                if(firstHalf[i]>=(5*secondHalf[j])) {
                    cnt += (j+1);
                    break;
                } else j--;
            }
            i--;
        }
        return cnt;
    }
}
```

### Brute Force

```java
class Solution {
    public int findCoverage(int[][] mat) {
        int r = mat.length, c = mat[0].length;
        int ans = 0;
        for(int i=0;i<r;i++) {
            for(int j=0;j<c;j++) {
                if(mat[i][j]==0) {
                    if(j>0) {
                        int tJ = j-1;
                        while(tJ>-1) {
                            if(mat[i][tJ]==1) {
                                ans++;
                                break;
                            }
                            else tJ--;
                        }
                    }
                    if(j<c-1) {
                        int tJ = j+1;
                        while(tJ<c) {
                            if(mat[i][tJ]==1) {
                                ans++;
                                break;
                            }
                            else tJ++;
                        }
                    }
                    if(i>0) {
                        int tr = i-1;
                        while(tr>-1) {
                            if(mat[tr][j]==1) {
                                ans++;
                                break;
                            }
                            else tr--;
                        }
                    }
                    if(i<r-1) {
                        int tr = i+1;
                        while(tr<r) {
                            if(mat[tr][j]==1) {
                                ans++;
                                break;
                            }
                            else tr++;
                        }
                    }
                }
            }
        }
        return ans;
    }
}
```

### Efficient
```java
class Solution {
    public int findCoverage(int[][] mat) {
        int r = mat.length;
        int c = mat[0].length;
        
        boolean[][] left = new boolean[r][c];
        boolean[][] right = new boolean[r][c];
        boolean[][] up = new boolean[r][c];
        boolean[][] down = new boolean[r][c];
        
        // 1. Precompute Left and Right visibility
        for (int i = 0; i < r; i++) {
            boolean hasOne = false;
            for (int j = 0; j < c; j++) {
                left[i][j] = hasOne;
                if (mat[i][j] == 1) hasOne = true;
            }
            
            hasOne = false;
            for (int j = c - 1; j >= 0; j--) {
                right[i][j] = hasOne;
                if (mat[i][j] == 1) hasOne = true;
            }
        }
        
        // 2. Precompute Up and Down visibility
        for (int j = 0; j < c; j++) {
            boolean hasOne = false;
            for (int i = 0; i < r; i++) {
                up[i][j] = hasOne;
                if (mat[i][j] == 1) hasOne = true;
            }
            
            hasOne = false;
            for (int i = r - 1; i >= 0; i--) {
                down[i][j] = hasOne;
                if (mat[i][j] == 1) hasOne = true;
            }
        }
        
        // 3. Accumulate coverage for all 0s
        int totalCoverage = 0;
        for (int i = 0; i < r; i++) {
            for (int j = 0; j < c; j++) {
                if (mat[i][j] == 0) {
                    if (left[i][j]) totalCoverage++;
                    if (right[i][j]) totalCoverage++;
                    if (up[i][j]) totalCoverage++;
                    if (down[i][j]) totalCoverage++;
                }
            }
        }
        
        return totalCoverage;
    }
}
```

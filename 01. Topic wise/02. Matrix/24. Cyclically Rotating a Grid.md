class Solution {
    public int[][] rotateGrid(int[][] grid, int k) {
        int m = grid.length, n = grid[0].length;
        int startI = 0, startJ = 0, endI = m-1, endJ = n-1;
        while(startI<endI && startJ<endJ) {
            int totalElements = 2*(endI-startI+1) + 2*(endJ-startJ-1);
            int K = k%totalElements;
            while(K-->0) {
                int t = grid[startI][startJ];
                int currI = startI, currJ = startJ;
                boolean flag = false;
                char movement = 'D';
                while(!(currI==startI && currJ==startJ && flag)) {
                    if(currI==endI && currJ==startJ) movement = 'R';
                    else if(currI==endI && currJ==endJ) movement = 'U';
                    else if(currI==startI && currJ==endJ) movement = 'L';
                    int nextI = getNextI(movement, currI);
                    int nextJ = getNextJ(movement, currJ);
                    int a = grid[nextI][nextJ];
                    grid[nextI][nextJ] = t;
                    t = a;
                    currI = nextI;
                    currJ = nextJ;
                    flag = true;
                }
            }
            startI++;
            startJ++;
            endI--;
            endJ--;
        }
        return grid;
    }
    int getNextI(char movement, int currI) {
        if(movement=='D') return currI+1;
        if(movement=='R') return currI;
        if(movement=='U') return currI-1;
        return currI;
    }
    int getNextJ(char movement, int currJ) {
        if(movement=='D') return currJ;
        if(movement=='R') return currJ+1;
        if(movement=='U') return currJ;
        return currJ-1;
    }
}

class Solution {
    public int[][] rotateGrid(int[][] grid, int k) {
        int m = grid.length, n = grid[0].length;
        int startI = 0, startJ = 0, endI = m - 1, endJ = n - 1;
        
        while (startI < endI && startJ < endJ) {
            List<Integer> layerElements = new ArrayList<>();
            int currI = startI, currJ = startJ;
            char movement = 'D';
            boolean flag = false;

            while (!(currI == startI && currJ == startJ && flag)) {
                layerElements.add(grid[currI][currJ]);
                
                if (currI == endI && currJ == startJ) movement = 'R';
                else if (currI == endI && currJ == endJ) movement = 'U';
                else if (currI == startI && currJ == endJ) movement = 'L';
                
                currI = getNextI(movement, currI);
                currJ = getNextJ(movement, currJ);
                flag = true;
            }

            int totalElements = layerElements.size();
            int K = k % totalElements;

            currI = startI; currJ = startJ;
            movement = 'D';
            int listIdx = 0;

            while (listIdx<totalElements) {
                int targetIdx = listIdx - K;
                if(targetIdx<0) targetIdx += totalElements;
                grid[currI][currJ] = layerElements.get(targetIdx);

                if (currI == endI && currJ == startJ) movement = 'R';
                else if (currI == endI && currJ == endJ) movement = 'U';
                else if (currI == startI && currJ == endJ) movement = 'L';
                
                currI = getNextI(movement, currI);
                currJ = getNextJ(movement, currJ);
                listIdx++;
            }

            startI++; startJ++; endI--; endJ--;
        }
        return grid;
    }

    int getNextI(char movement, int currI) {
        if (movement == 'D') return currI + 1;
        if (movement == 'U') return currI - 1;
        return currI;
    }

    int getNextJ(char movement, int currJ) {
        if (movement == 'R') return currJ + 1;
        if (movement == 'L') return currJ - 1;
        return currJ;
    }
}

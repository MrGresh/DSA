class Solution {
    public List<Integer> exitPoint(int[][] mat) {
        int rows = mat.length, cols = mat[0].length;
        int i=0, j=0;
        char dir = 'R';
        while(true) {
            if(mat[i][j]==0) {
                if(dir=='R') {
                    if(j+1<cols) j+=1;
                    else break;
                } else if(dir=='D') {
                    if(i+1<rows) i+=1;
                    else break;
                } else if(dir=='L') {
                    if(j-1>=0) j-=1;
                    else break;
                } else {
                    if(i-1>=0) i-=1;
                    else break;
                }
            } else {
                mat[i][j]=0;
                if(dir=='R') {
                    dir = 'D';
                    if(i+1<rows) i+=1;
                    else break;
                } else if(dir=='D') {
                    dir = 'L';
                    if(j-1>=0) j-=1;
                    else break;
                } else if(dir=='L') {
                    dir = 'U';
                    if(i-1>=0) i-=1;
                    else break;
                } else {
                    dir = 'R';
                    if(j+1<cols) j+=1;
                    else break;
                }
            }
        }
        return new ArrayList<>(Arrays.asList(i, j));
    }
}

class Solution {
    public char[][] rotateTheBox(char[][] boxGrid) {
        int m = boxGrid.length, n = boxGrid[0].length;
        for(char[] row: boxGrid) {
            int nextEmptyJ = -1;
            for(int j=n-1;j>=0;j--) {
                if(row[j]=='*') nextEmptyJ = -1;
                else if(row[j]=='.' && nextEmptyJ == -1) nextEmptyJ = j;
                else if(row[j]=='#' && nextEmptyJ != -1) {
                    row[nextEmptyJ] = '#';
                    row[j] = '.';
                    nextEmptyJ--;
                }
            }
        }
        char[][] res = new char[n][m];
        for(int i=0;i<m;i++) {
            for(int j=0;j<n;j++) {
                res[j][m-1-i] = boxGrid[i][j];
            }
        }
        return res;
    }
}

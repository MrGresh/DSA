class Solution {
    public int maximumSum(int[][] mat, int k) {
        int n = mat.length;
        
        // 1-based 2D prefix sum array to avoid boundary checks
        int[][] pref = new int[n + 1][n + 1];
        
        // Step 1: Construct the 2D prefix sum array
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                pref[i][j] = mat[i - 1][j - 1] 
                           + pref[i - 1][j] 
                           + pref[i][j - 1] 
                           - pref[i - 1][j - 1];
            }
        }
        
        int maxSum = Integer.MIN_VALUE;
        
        // Step 2: Iterate over all possible k x k sub-grids
        for (int r = k; r <= n; r++) {
            for (int c = k; c <= n; c++) {
                // Calculate sub-grid sum using 2D prefix sum formula
                int currentSum = pref[r][c] 
                               - pref[r - k][c] 
                               - pref[r][c - k] 
                               + pref[r - k][c - k];
                
                maxSum = Math.max(maxSum, currentSum);
            }
        }
        
        return maxSum;
    }
}

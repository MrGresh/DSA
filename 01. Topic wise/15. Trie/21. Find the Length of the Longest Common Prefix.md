class TrieNode {
    TrieNode[] childs = new TrieNode[10];
}
class Solution {
    public int longestCommonPrefix(int[] arr1, int[] arr2) {
        TrieNode root = new TrieNode();
        for(int x: arr1) {
            TrieNode node = root;
            String num = String.valueOf(x);
            for(int i=0;i<num.length();i++) {
                int idx = num.charAt(i)-'0';
                if(node.childs[idx]==null) node.childs[idx] = new TrieNode();
                node = node.childs[idx];
            }
        }
        int ans = 0;
        for(int x: arr2) {
            TrieNode node = root;
            int curr = 0;
            String num = String.valueOf(x);
            for(int i=0;i<num.length();i++) {
                int idx = num.charAt(i)-'0';
                if(node.childs[idx]==null) break;
                else {
                    curr++;
                    node = node.childs[idx];
                }
            }
            ans = Math.max(ans, curr);
        }
        return ans;
    }
}

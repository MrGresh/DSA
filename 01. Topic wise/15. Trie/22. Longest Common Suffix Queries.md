class TrieNode {
    TrieNode[] childs = new TrieNode[26];
    int index = -1;
}
class Solution {
    public int[] stringIndices(String[] wordsContainer, String[] wordsQuery) {
        TrieNode root = new TrieNode();
        int nonAnswerIdx = 0;
        for(int i=0;i<wordsContainer.length;i++) {
            String str = wordsContainer[i];
            if(str.length()<wordsContainer[nonAnswerIdx].length()) nonAnswerIdx = i;
            TrieNode node = root;
            for(int j=str.length()-1;j>=0;j--) {
                int idx = str.charAt(j)-'a';
                if(node.childs[idx] == null) node.childs[idx] = new TrieNode();
                node = node.childs[idx];
                if(node.index == -1 || (wordsContainer[i].length() < wordsContainer[node.index].length())) node.index = i;
            }
        }
        root.index = nonAnswerIdx;
        int[] ans = new int[wordsQuery.length];
        for(int i=0;i<wordsQuery.length;i++) {
            String str = wordsQuery[i];
            TrieNode node = root;
            for(int j=str.length()-1;j>=0;j--) {
                int idx = str.charAt(j)-'a';
                if(node.childs[idx] == null) break;
                node = node.childs[idx];
            }
            ans[i] = node.index;
        }
        return ans;
    }
}

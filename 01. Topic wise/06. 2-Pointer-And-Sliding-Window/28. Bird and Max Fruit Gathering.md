class Solution {
    public int maxFruits(ArrayList<Integer> arr, int m) {
        int n = arr.size();
        m = Math.min(m, n);

        int sum = 0;
        for (int i = 0; i < m; i++) sum += arr.get(i);

        int ans = sum;
        for (int i = 1; i < n; i++) {
            // remove arr[i-1], add the element that newly enters the window
            sum -= arr.get(i - 1);
            sum += arr.get((i + m - 1) % n);
            ans = Math.max(ans, sum);
        }
        return ans;
    }
}

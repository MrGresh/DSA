### Recursion (TLE)
```java
class Solution {
    public int minOperations(int[] nums, int x) {
        int val = rec(nums, 0, nums.length-1, x);
        return val == Integer.MAX_VALUE ? -1 : val;
    }
    int rec(int[] nums, int l, int r, int x) {
        if(x==0) return 0;
        if(l>r) return Integer.MAX_VALUE;
        int steps = Integer.MAX_VALUE;
        if(nums[l]<=x) {
            int val = rec(nums, l+1, r, x-nums[l]);
            if(val != Integer.MAX_VALUE) steps = 1 + val;
        }
        if(nums[r]<=x) {
            int val = rec(nums, l, r-1, x-nums[r]);
            if(val != Integer.MAX_VALUE) steps = Math.min(steps, 1 + val);
        }
        return steps;
    }
}
```

### Sliding-Window
```java
class Solution {
    public int minOperations(int[] nums, int x) {
        int n = nums.length;
        long totalSum = 0;
        for (int v : nums) totalSum += v;

        long target = totalSum - x;
        if (target < 0) return -1;
        if (target == 0) return n; // remove everything

        int left = 0;
        long windowSum = 0;
        int maxLen = -1;

        for (int right = 0; right < n; right++) {
            windowSum += nums[right];

            while (windowSum > target) windowSum -= nums[left++];

            if (windowSum == target) maxLen = Math.max(maxLen, right - left + 1);
        }

        return maxLen == -1 ? -1 : n - maxLen;
    }
}
```

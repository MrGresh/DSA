### Binary Search
```java
class Solution {
    public static int countPairs(int arr[], int k) {
        Arrays.sort(arr);
        int n = arr.length;
        int count = 0;

        for (int i = 0; i < n; i++) {
            // Search right: count j > i such that arr[j] - arr[i] < k
            if (i < n - 1) {
                int lo = i + 1, hi = n - 1, res = i; // res = i means no valid j found
                while (lo <= hi) {
                    int mid = lo + (hi - lo) / 2;
                    if (arr[mid] - arr[i] < k) {
                        res = mid;
                        lo = mid + 1;
                    } else {
                        hi = mid - 1;
                    }
                }
                count += (res - i); // number of valid j's from i+1 to res
            }

            // Search left: count j < i such that arr[i] - arr[j] < k
            if (i > 0) {
                int lo = 0, hi = i - 1, res = i; // res = i means no valid j found
                while (lo <= hi) {
                    int mid = lo + (hi - lo) / 2;
                    if (arr[i] - arr[mid] < k) {
                        res = mid;
                        hi = mid - 1;
                    } else {
                        lo = mid + 1;
                    }
                }
                count += (i - res);
            }
        }

        return count / 2; // each pair counted twice (once from each side)
    }
}
```

### Sliding Window
```java
class Solution {
    public static int countPairs(int arr[], int k) {
        Arrays.sort(arr);
        int start = 0;
        int ans = 0;

        for (int end = 1; end < arr.length; end++) {
            while (arr[end] - arr[start] >= k) start++;
            ans += (end - start); // every index in [start, end-1] pairs validly with end
        }

        return ans;
    }
}
```

class Solution {
    public String shortestBeautifulSubstring(String s, int k) {
        String ans = "";
        int start = 0;
        int cnt = 0;
        
        for (int end = 0; end < s.length(); end++) {
            if (s.charAt(end) == '1') cnt++;
            
            // shrink while we have too many 1's
            while (cnt > k) {
                if (s.charAt(start) == '1') cnt--;
                start++;
            }
            
            // shrink further while cnt == k and leading char is '0'
            // (removing leading zeros keeps cnt the same but shortens window)
            while (cnt == k && s.charAt(start) == '0') {
                start++;
            }
            
            if (cnt == k) {
                String candidate = s.substring(start, end + 1);
                if (ans.isEmpty() 
                    || candidate.length() < ans.length()
                    || (candidate.length() == ans.length() && candidate.compareTo(ans) < 0)) {
                    ans = candidate;
                }
            }
        }
        
        return ans;
    }
}

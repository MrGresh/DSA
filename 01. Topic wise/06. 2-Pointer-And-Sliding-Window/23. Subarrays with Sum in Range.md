class Solution {
    public int countSubarray(int[] arr, int l, int r) {
        return (int)(countAtMost(arr, r) - countAtMost(arr, l - 1));
    }
    
    // Count subarrays with sum <= X
    private long countAtMost(int[] arr, int X) {
        if (X < 0) return 0;
        long sum = 0;
        long count = 0;
        int start = 0;
        for (int end = 0; end < arr.length; end++) {
            sum += arr[end];
            while (sum > X) {
                sum -= arr[start];
                start++;
            }
            // every subarray ending at 'end' starting from start..end has sum <= X
            count += (end - start + 1);
        }
        return count;
    }
}

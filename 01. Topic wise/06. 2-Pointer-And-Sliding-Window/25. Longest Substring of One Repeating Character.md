class Solution {
    public int[] longestRepeating(String s, String queryCharacters, int[] queryIndices) {
        int k = queryIndices.length;
        char[] str = s.toCharArray();
        int[] ans = new int[k];
        for(int i=0;i<k;i++) {
            int len = 0;
            str[queryIndices[i]] = queryCharacters.charAt(i);
            int start = 0;
            Map<Character, Integer> vis = new HashMap<>();
            for(int end=0;end<str.length;end++) {
                vis.put(str[end], vis.getOrDefault(str[end], 0) + 1);
                while(vis.size()>1) {
                    if(vis.get(str[start])==1) vis.remove(str[start]);
                    else vis.put(str[start], vis.get(str[start])-1);
                    start++;
                }
                len = Math.max(len, end-start+1);
            }
            ans[i] = len;
        }
        return ans;
    }
}

class Solution {
    public int maxFrequency(int[] arr, int k) {
        Arrays.sort(arr);

        int l = 0;
        long sum = 0;   // sum of the current window
        int best = 1;

        for (int r = 0; r < arr.length; r++) {
            sum += arr[r];

            // cost to raise every element in [l, r] to arr[r]
            while ((long) arr[r] * (r - l + 1) - sum > k) sum -= arr[l++];

            best = Math.max(best, r - l + 1);
        }

        return best;
    }
}

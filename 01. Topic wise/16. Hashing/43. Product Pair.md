### Hashing
```java
class Solution {
    public boolean isProduct(int[] arr, long target) {
        Set<Integer> set = new HashSet<>();
        
        for (int x : arr) {
            // 1. Handle zero separately to avoid division by zero exceptions
            if (x == 0) {
                // If target is 0 and we've already seen at least one other element, 
                // we have a valid pair because 0 * anything = 0.
                if (target == 0 && !set.isEmpty()) {
                    return true;
                }
                set.add(x);
                continue;
            }
            
            // 2. Check if the target is cleanly divisible by x
            if (target % x == 0) {
                long needed = target / x;
                
                // 3. Prevent truncation bugs: Only check the set if 'needed' 
                // safely fits within the boundaries of a standard 32-bit integer.
                if (needed >= Integer.MIN_VALUE && needed <= Integer.MAX_VALUE) {
                    if (set.contains((int) needed)) {
                        return true;
                    }
                }
            }
            
            set.add(x);
        }
        
        return false;
    }
}
```

### Binary Search
```java
class Solution {
    public boolean isProduct(int[] arr, long target) {
        int n = arr.length;
        // 1. Sort the array in-place: O(N log N) time, O(1) auxiliary space
        Arrays.sort(arr);
        
        for (int i = 0; i < n; i++) {
            long x = arr[i];
            
            // Handle zero explicitly to avoid division by zero
            if (x == 0) {
                if (target == 0 && n > 1) {
                    return true;
                }
                continue;
            }
            
            // Check if cleanly divisible
            if (target % x == 0) {
                long needed = target / x;
                
                // Ensure the needed value fits into a 32-bit integer bounds
                if (needed >= Integer.MIN_VALUE && needed <= Integer.MAX_VALUE) {
                    int targetValue = (int) needed;
                    
                    // 2. Binary search for the complement value: O(log N)
                    int index = Arrays.binarySearch(arr, targetValue);
                    
                    // If found, ensure it isn't the current element matching with itself
                    if (index >= 0) {
                        if (index != i) {
                            return true; 
                        } else {
                            // If it found itself, check if the neighbor has the same value
                            if ((index > 0 && arr[index - 1] == targetValue) || 
                                (index < n - 1 && arr[index + 1] == targetValue)) {
                                return true;
                            }
                        }
                    }
                }
            }
        }
        
        return false;
    }
}
```

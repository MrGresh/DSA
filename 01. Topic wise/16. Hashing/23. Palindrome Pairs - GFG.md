class Solution {
    public boolean palindromePair(String[] arr) {
        Map<String, List<Integer>> map = new HashMap<>();
        for(int i=0;i<arr.length;i++) {
            String rev = new StringBuilder(arr[i]).reverse().toString();
            if(!map.containsKey(rev)) map.put(rev, new ArrayList<>());
            map.get(rev).add(i);
        }
        for(int i=0;i<arr.length;i++) {
            for(int j=0;j<=arr[i].length();j++) {
                String pref = arr[i].substring(0, j);
                String suff = arr[i].substring(j);
                if(isPalindrome(suff)) {
                    if(map.containsKey(pref)) {
                        for(int idx: map.get(pref)) {
                            if(idx!=i) return true;
                        }
                    }
                }
                if(isPalindrome(pref)) {
                    if(map.containsKey(suff)) {
                        for(int idx: map.get(suff)) {
                            if(idx!=i) return true;
                        }
                    }
                }
            }
        }
        return false;
    }
    boolean isPalindrome(String s) {
        int l=0, r=s.length()-1;
        while(l<r) {
            if(s.charAt(l)!=s.charAt(r)) return false;
            l++;
            r--;
        }
        return true;
    }
}

class Solution {
    public int[] findThePrefixCommonArray(int[] A, int[] B) {
        int[] ans = new int[A.length];
        Set<Integer> s1 = new HashSet<>();
        Set<Integer> s2 = new HashSet<>();
        for(int i=0;i<A.length;i++) {
            s1.add(A[i]);
            s2.add(B[i]);
            int count = 0;
            for (int num : s1) {
                if (s2.contains(num)) count++;
            }
            ans[i] = count;
        }
        return ans;
    }
}

class Solution {
    public int[] findThePrefixCommonArray(int[] A, int[] B) {
        int n = A.length;
        int[] ans = new int[n];
        Set<Integer> seen = new HashSet<>();
        int commonCount = 0;
        for (int i = 0; i < n; i++) {
            if (!seen.add(A[i])) commonCount++;
            if (!seen.add(B[i])) commonCount++;
            ans[i] = commonCount;
        }
        return ans;
    }
}

class Solution {
    public int minimumPushes(String word) {
        int[] freq = new int[26];
        for(int i=0;i<word.length();i++) {
            char ch = word.charAt(i);
            freq[ch-'a']++;
        }
        TreeMap<Integer, List<Character>> map = new TreeMap<>(Collections.reverseOrder());
        for(int i=0;i<26;i++) {
            if(freq[i]>0) {
                if(!map.containsKey(freq[i])) map.put(freq[i], new ArrayList<>());
                map.get(freq[i]).add((char)(i+'a'));
            }
        }
        Set<Character> first = new HashSet<>();
        Set<Character> second = new HashSet<>();
        Set<Character> third = new HashSet<>();
        Set<Character> fourth = new HashSet<>();
        int ans = 0;
        for (int key : map.keySet()) {
            for(char ch: map.get(key)) {
                    if(first.contains(ch)) ans+=key;
                    else if(first.size()<8) {
                        ans+=key;
                        first.add(ch);
                    }
                    else if(second.contains(ch)) ans+=(2*key);
                    else if(second.size()<8) {
                        ans+=(2*key);
                        second.add(ch);
                    }
                    else if(third.contains(ch)) ans+=(3*key);
                    else if(third.size()<8) {
                        ans+=(3*key);
                        third.add(ch);
                    }
                    else if(fourth.contains(ch)) ans+=(4*key);
                    else if(fourth.size()<2) {
                        ans+=(4*key);
                        fourth.add(ch);
                    }
            }
        }
        return ans;
    }
}

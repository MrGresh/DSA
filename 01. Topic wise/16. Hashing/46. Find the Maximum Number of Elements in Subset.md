import java.util.*;

class Solution {
    public int maximumLength(int[] nums) {
        Map<Integer, Integer> freq = new HashMap<>();
        for (int x : nums) {
            freq.put(x, freq.getOrDefault(x, 0) + 1);
        }
        
        int ans = 1;
        
        // Special case for 1: We can pick the largest odd number of 1s
        if (freq.containsKey(1)) {
            int count1 = freq.get(1);
            if (count1 % 2 == 0) {
                ans = Math.max(ans, count1 - 1);
            } else {
                ans = Math.max(ans, count1);
            }
        }
        
        for (int key : freq.keySet()) {
            if (key == 1) continue; // Already handled
            
            int cnt = 0;
            long x = key; // Use long to safely square without overflow
            
            while (true) {
                // If the current element doesn't exist in the map, 
                // the previous element must act as the single peak element.
                if (!freq.containsKey((int) x)) {
                    ans = Math.max(ans, (cnt * 2) - 1);
                    break;
                }
                
                if (freq.get((int) x) >= 2) {
                    cnt++;
                } else {
                    // If it only has 1 element, it MUST be the peak element.
                    cnt++;
                    ans = Math.max(ans, (cnt * 2) - 1);
                    break;
                }
                
                x = x * x;
            }
        }
        
        return ans;
    }
}

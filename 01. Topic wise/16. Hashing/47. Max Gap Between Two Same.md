class Solution {
    public int maxCharGap(String s) {
        int[] firstIndex = new int[26];
        Arrays.fill(firstIndex, -1);
        
        int maxGap = -1;
        
        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);
            int idx = ch - 'a';
            
            if (firstIndex[idx] == -1) {
                firstIndex[idx] = i;
            } else {
                int currentGap = i - firstIndex[idx] - 1;
                maxGap = Math.max(maxGap, currentGap);
            }
        }
        
        return maxGap;
    }
}

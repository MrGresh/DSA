class Solution {
    public int maxNumberOfFamilies(int n, int[][] reservedSeats) {
        Map<Integer, Set<Integer>> reserved = new HashMap<>();
        for (int[] r : reservedSeats) {
            reserved.computeIfAbsent(r[0], k -> new HashSet<>()).add(r[1]);
        }

        List<Integer> left = Arrays.asList(2, 3, 4, 5);
        List<Integer> mid = Arrays.asList(4, 5, 6, 7);
        List<Integer> right = Arrays.asList(6, 7, 8, 9);

        long cnt = 0;

        // Only process rows that have reservations
        for (Set<Integer> seats : reserved.values()) {
            boolean leftFree = Collections.disjoint(seats, left);
            boolean rightFree = Collections.disjoint(seats, right);

            if (leftFree && rightFree) {
                cnt += 2;
            } else if (leftFree || rightFree) {
                cnt += 1;
            } else if (Collections.disjoint(seats, mid)) {
                cnt += 1;
            }
        }

        // All remaining rows are completely empty -> 2 groups each
        long emptyRows = n - reserved.size();
        cnt += emptyRows * 2;

        return (int) cnt;
    }
}

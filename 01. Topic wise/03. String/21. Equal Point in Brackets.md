class Solution {
    public int findIndex(String s) {
        int[] closeCountInRight = new int[s.length()];
        int currCloseCnt = 0;
        for(int i=s.length()-1;i>=0;i--) {
            if(s.charAt(i)==')') currCloseCnt++;
            closeCountInRight[i] = currCloseCnt;
        }
        int currOpenCnt = 0;
        for(int i=0;i<s.length();i++) {
            if(currOpenCnt==closeCountInRight[i]) return i;
            if(s.charAt(i)=='(') currOpenCnt++;
        }
        return s.length();
    }
}

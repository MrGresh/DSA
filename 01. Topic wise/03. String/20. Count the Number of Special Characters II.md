class Solution {
    public int numberOfSpecialChars(String word) {
        int[] freqLower = new int[26];
        int[] freqUpper = new int[26];
        int[] idxLower = new int[26];
        int[] idxUpper = new int[26];
        Arrays.fill(idxLower, -1);
        Arrays.fill(idxUpper, -1);
        for(int i=0;i<word.length();i++) {
            if(word.charAt(i)>='a' && word.charAt(i)<='z') {
                freqLower[word.charAt(i)-'a']++;
                idxLower[word.charAt(i)-'a'] = i;
            }
            else {
                freqUpper[word.charAt(i)-'A']++;
                if(idxUpper[word.charAt(i)-'A'] == -1) idxUpper[word.charAt(i)-'A'] = i;
            }
        }
        int ans = 0;
        for(int i=0;i<26;i++) {
            if(freqLower[i]>0 && freqUpper[i]>0 && (idxLower[i]<idxUpper[i])) ans++;
        }
        return ans;
    }
}

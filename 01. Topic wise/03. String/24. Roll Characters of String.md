class Solution {
    public String findRollOut(String s, int[] roll) {
        int n = s.length();
        int[] cnt = new int[n + 1];

        for (int r : roll) cnt[r]++;

        char[] arr = s.toCharArray();
        int running = 0;

        for (int i = n - 1; i >= 0; i--) {
            running = (running + cnt[i + 1]) % 26;
            arr[i] = (char) ('a' + (arr[i] - 'a' + running) % 26);
        }

        return new String(arr);
    }
}

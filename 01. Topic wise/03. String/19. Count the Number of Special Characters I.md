class Solution {
    public int numberOfSpecialChars(String word) {
        int[] freq1 = new int[26];
        int[] freq2 = new int[26];
        for(int i=0;i<word.length();i++) {
            if(word.charAt(i)>='a' && word.charAt(i)<='z') freq1[word.charAt(i)-'a']++;
            else freq2[word.charAt(i)-'A']++;
        }
        int ans = 0;
        for(int i=0;i<26;i++) {
            if(freq1[i]>0 && freq2[i]>0) ans++;
        }
        return ans;
    }
}

class Solution {
    public String smallestPalindrome(String s) {
        int n = s.length();
        int[] freq = new int[26];
        for(int i=0;i<n;i++) {
            char ch = s.charAt(i);
            freq[ch-'a']++;
        }
        char center = '@';
        StringBuilder forward = new StringBuilder();
        StringBuilder backward = new StringBuilder();
        for(int i=0;i<26;i++) {
            char ch = (char)(i+'a');
            if(freq[i]%2!=0) {
                center = ch;
                freq[i]--;
            }
            while(freq[i]>0) {
                forward.append(ch);
                backward.insert(0, ch);
                freq[i]-=2;
            }
        }
        if(center!='@') forward.append(center);
        return forward.append(backward).toString();
    }
}

class Solution {
    public int minimumPushes(String word) {
        Set<Character> first = new HashSet<>();
        Set<Character> second = new HashSet<>();
        Set<Character> third = new HashSet<>();
        Set<Character> fourth = new HashSet<>();
        int ans = 0;
        for(int i=0;i<word.length();i++) {
            char ch = word.charAt(i);
            if(first.contains(ch)) ans++;
            else if(first.size()<8) {
                ans++;
                first.add(ch);
            }
            else if(second.contains(ch)) ans+=2;
            else if(second.size()<8) {
                ans+=2;
                second.add(ch);
            }
            else if(third.contains(ch)) ans+=3;
            else if(third.size()<8) {
                ans+=3;
                third.add(ch);
            }
            else if(fourth.contains(ch)) ans+=4;
            else if(fourth.size()<2) {
                ans+=4;
                fourth.add(ch);
            }
        }
        return ans;
    }
}
